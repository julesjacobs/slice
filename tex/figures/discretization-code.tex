    \centering
    
    % Basic discretization rules
    \begin{align}
        \texttt{discretize}(c) &= 
        \begin{cases}
            c & \text{if } B = \top \\[0.5em]
            \finconst{k}{n+1} \text{ such that } c \in \intervals{B}_k & \text{if } B = \{\clt{b_1}, \ldots, \clt{b_n}\}
        \end{cases}
    \end{align}
    
    \vspace{0.5em}
    
    \begin{align}
        \texttt{discretize}(\letkw\; x = e_1\; \inkw\; e_2) 
            &= \letkw\; x = \texttt{discretize}(e_1)\; \inkw\; \texttt{discretize}(e_2)
    \end{align}
    
    \begin{align}
        \texttt{discretize}(e_1 < e_2) 
            &= \texttt{discretize}(e_1) < \texttt{discretize}(e_2)
    \end{align}
    
    \begin{align}
        \texttt{discretize}(\ifkw\; e_1\; \thenkw\; e_2\; \elsekw\; e_3) 
            &= \ifkw\; \texttt{discretize}(e_1)\; \thenkw\; \texttt{discretize}(e_2) \notag \\
            &\quad \elsekw\; \texttt{discretize}(e_3)
    \end{align}
    
    \begin{align}
        \texttt{discretize}(x) &= x 
    \end{align}
    
    \vspace{1em}
    
    % Uniform distribution discretization
    \begin{align}
        \texttt{discretize}(\uniform(e_1, e_2)) &= 
        \begin{cases}
            \uniform(e_1, e_2) & \text{if } B = \top \\[1em]
            \begin{aligned}
                &\discrete(p_0, \ldots, p_n) \\
                &\text{where } p_k = \text{CDF}(b_{k+1}) - \text{CDF}(b_k)
            \end{aligned}
            & 
            \begin{aligned}
                &\text{if } B = \{\clt{b_1}, \ldots, \clt{b_n}\} \\
                &\text{and } e_1, e_2 \in \mathbb{R}
            \end{aligned} \\[1.5em]
            \begin{aligned}
                &\ifkw\; e_1 = \texttt{discretize}(v_1)\; \thenkw \\
                &\quad \texttt{discretize}(\uniform(v_1, e_2)) \\
                &\elsekw\; \ifkw\; e_1 = \texttt{discretize}(v_2)\; \thenkw \\
                &\quad \texttt{discretize}(\uniform(v_2, e_2)) \\
                &\vdots \\
                &\elsekw\; \texttt{discretize}(\uniform(v_m, e_2))
            \end{aligned}
            &
            \begin{aligned}
                &\text{if } B = \{\clt{b_1}, \ldots, \clt{b_n}\} \\
                &\text{and } V_1 = \{v_1, \ldots, v_m\}
            \end{aligned} \\[5.5em]
            \begin{aligned}
                &\ifkw\; e_2 = \texttt{discretize}(v_1)\; \thenkw \\
                &\quad \texttt{discretize}(\uniform(e_1, v_1)) \\
                &\elsekw\; \ifkw\; e_2 = \texttt{discretize}(v_2)\; \thenkw \\
                &\quad \texttt{discretize}(\uniform(e_1, v_2)) \\
                &\vdots \\
                &\elsekw\; \texttt{discretize}(\uniform(e_1, v_m))
            \end{aligned}
            &
            \begin{aligned}
                &\text{if } B = \{\clt{b_1}, \ldots, \clt{b_n}\} \\
                &\text{and } V_2 = \{v_1, \ldots, v_m\}
            \end{aligned}
        \end{cases}
    \end{align}
