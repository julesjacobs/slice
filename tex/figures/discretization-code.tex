    \centering
    \footnotesize

    % Basic discretization rules
    \begin{align}
        \mathcal{D}\sem{c : \float[B;V]} &=
        \begin{cases}
            c & \text{if } B = \top, \\
            \finconst{k}{n+1} \text{ such that } c \in \intervals{B}_k & \text{if } B \neq \top.
        \end{cases}
    \end{align}

    \vspace{0.5em}

    \begin{align}
        \mathcal{D}\sem{\letkw\; x = e_1 : \tau_1\; \inkw\; e_2 : \tau_2 : \tau_2} &=
        \letkw\; x = \mathcal{D}\sem{e_1 : \tau_1}\; \inkw\; \mathcal{D}\sem{e_2 : \tau_2}.
    \end{align}

    \begin{align}
        \mathcal{D}\sem{(e_1 : \float[B_1;V_1] < e_2 : \float[B_2;V_2]) : \bool} &=
        \mathcal{D}\sem{e_1 : \float[B_1;V_1]} < \mathcal{D}\sem{e_2 : \float[B_2;V_2]}.
    \end{align}

    \begin{align}
        \mathcal{D}\sem{(e_1 : \float[B_1;V_1] \leq e_2 : \float[B_2;V_2]) : \bool} &=
        \mathcal{D}\sem{e_1 : \float[B_1;V_1]} \leq \mathcal{D}\sem{e_2 : \float[B_2;V_2]}.
    \end{align}

    \begin{align}
        \mathcal{D}\sem{(\ifkw\; e_1 : \bool\; \thenkw\; e_2 : \tau\; \elsekw\; e_3 : \tau) : \tau} &=
        \ifkw\; \mathcal{D}\sem{e_1 : \bool}\; \thenkw\; \mathcal{D}\sem{e_2 : \tau} \notag \\
        &\quad \elsekw\; \mathcal{D}\sem{e_3 : \tau}.
    \end{align}

    \begin{align}
        \mathcal{D}\sem{x : \tau} &= x.
    \end{align}

    \vspace{1em}

    % Uniform distribution discretization
    \begin{align}
        \mathcal{D}\sem{\uniform(e_1 : \float[B_1;V_1], e_2 : \float[B_2;V_2]) : \float[B;V]} &
    \end{align}

    \begin{align}
        =
        \begin{cases}
            \texttt{Error: Uniform $e_1 > e_2$} 
            & \text{if } e_1, e_2 \in \mathbb{R}, e_1 > e_2 \\[1em]
            \uniform(\mathcal{D}\sem{e_1 : \float[B_1;V_1]}, \mathcal{D}\sem{e_2 : \float[B_2;V_2]}) 
            & \text{if } B = \top, \\[1.5em]
            \begin{aligned}
                &\discrete(p_0, \ldots, p_n) \\
                &\text{ where } p_k = \text{CDF}(b_{k+1}) - \text{CDF}(b_k)
            \end{aligned}
            & 
            \begin{aligned}
                &\text{if } B \neq \top, \\
                &B = \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\},\\
                &e_1, e_2 \in \mathbb{R}
            \end{aligned} \\[2.5em]
            \begin{aligned}
                &\letkw \; v_1' = \mathcal{D}\sem{e_1 : \float[B_1; V_1]} \; \inkw \\
                &\letkw \; v_2' = \mathcal{D}\sem{e_2 : \float[B_2; V_2]} \; \inkw \\
                &\ifkw\; v_1' == 0\; \&\& \; v_2' == 0\; \thenkw \\
                &\quad \mathcal{D}\sem{\uniform(v_1, w_1)} \\
                &\vdots \\
                &\elsekw\; \ifkw\; v_1' == i\; \&\& \; v_2' == j\; \thenkw \\
                &\quad \mathcal{D}\sem{\uniform(v_{i+1}, w_{j+1})} \\
                &\vdots \\
                &\elsekw \\
                &\quad \mathcal{D}\sem{\uniform(v_n, w_m)}
            \end{aligned}
            &
            \begin{aligned}
                &\text{if } B \neq \top, \\
                &V_1 = \{v_1, \ldots, v_n\}, \\
                &V_2 = \{w_1, \ldots, w_m\}
            \end{aligned}
        \end{cases}
    \end{align}

\end{figure}
