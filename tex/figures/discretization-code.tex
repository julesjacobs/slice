\begin{figure}[t]
    \centering
    \tiny
    
    % Basic discretization rules
    \begin{align}
        \texttt{discretize}(c : \float[B;V]) &= 
        \begin{cases}
            c & \text{if } B = \top \\[0.5em]
            \finconst{k}{n+1} \text{ such that } c \in \intervals{B}_k & \text{if } B \neq \top
        \end{cases}
    \end{align}
    
    \vspace{0.5em}
    
    \begin{align}
        \texttt{discretize}((\letkw\; x = e_1 : \tau_1\; \inkw\; e_2 : \tau_2) : \tau_2) 
            &= \letkw\; x = \texttt{discretize}(e_1 : \tau_1)\; \inkw\; \texttt{discretize}(e_2 : \tau_2)
    \end{align}
    
    \begin{align}
        \texttt{discretize}((e_1 : \float[B_1; V_1] < e_2 : \float[B_2; V_2]) : \bool) 
            &= \texttt{discretize}(e_1 : \float[B_1; V_1]) < \texttt{discretize}(e_2 : \float[B_2; V_2])
    \end{align}

    \begin{align}
        \texttt{discretize}((e_1 : \float[B_1; V_1] \leq e_2 : \float[B_2; V_2]) : \bool) 
            &= \texttt{discretize}(e_1 : \float[B_1; V_1]) \leq \texttt{discretize}(e_2 : \float[B_2; V_2])
    \end{align}
    
    \begin{align}
        \texttt{discretize}((\ifkw\; e_1 : \bool\; \thenkw\; e_2 : \tau\; \elsekw\; e_3 : \tau) : \tau) 
            &= \ifkw\; \texttt{discretize}(e_1 : \bool)\; \thenkw\; \texttt{discretize}(e_2 : \tau) \notag \\
            &\quad \elsekw\; \texttt{discretize}(e_3 : \tau)
    \end{align}
    
    \begin{align}
        \texttt{discretize}(x : \tau) &= x 
    \end{align}
    
    \vspace{1em}
    
    % Uniform distribution discretization
    \begin{align}
        \texttt{discretize}(\uniform(e_1 : \float[B_1;V_1], e_2 : \float[B_2;V_2]) : \float[B;V]) &=
        \begin{cases}
            \uniform(\texttt{discretize}(e_1 : \float[B_1;V_1]), \\
            \texttt{ discretize}(e_2 : \float[B_2;V_2])) & \text{if } B = \top \\[2.5em]
            \begin{aligned}
                &\discrete(p_0, \ldots, p_n) \\
                &\text{ where } p_k = \text{CDF}(b_{k+1}) - \text{CDF}(b_k)
            \end{aligned}
            & 
            \begin{aligned}
                &\text{if } B \neq \top \\
                &\text{and } e_1, e_2 \in \mathbb{R}
            \end{aligned} \\[2.5em]
            \begin{aligned}
                & \letkw \; \text{fresh1} = \texttt{discretize}(e_1 : \float[B_1; V_1]) \; \inkw \\
                &\ifkw\; \text{fresh1} == 0\; \thenkw \\
                &\quad \texttt{discretize}(\uniform(v_1, e_2 : \float[B_2; V_2])) \\
                &\elsekw\; \ifkw\; \text{fresh1} == 1\; \thenkw \\
                &\quad \texttt{discretize}(\uniform(v_2, e_2 : \float[B_2; V_2])) \\
                &\vdots \\
                &\elsekw \\
                &\quad \texttt{discretize}(\uniform(v_n, e_2))
            \end{aligned}
            &
            \begin{aligned}
                &\text{if } B \neq \top \\
                &\text{and } V_1 = \{v_1, \ldots, v_n\}
            \end{aligned} \\[6.5em]
            \begin{aligned}
                & \letkw \; \text{fresh2} = \texttt{discretize}(e_2 : \float[B_2; V_2]) \; \inkw \\
                &\ifkw\; \text{fresh2} == 0\; \thenkw \\
                &\quad \texttt{discretize}(\uniform(e_1 : \float[B_1; V_1], v_1)) \\
                &\elsekw\; \ifkw\; \text{fresh2} == 1\; \thenkw \\
                &\quad \texttt{discretize}(\uniform(e_1 : \float[B_1; V_1], v_2)) \\
                &\vdots \\
                &\elsekw \\
                &\quad \texttt{discretize}(\uniform(e_1, v_n))
            \end{aligned}
            &
            \begin{aligned}
                &\text{if } B \neq \top \\
                &\text{and } V_2 = \{v_1, \ldots, v_n\}
            \end{aligned}
        \end{cases}
    \end{align}

    \label{fig:discretization-code}
\end{figure}
