    \begin{align}
        \texttt{discretize}(c) &= 
        \begin{cases}
            c, & \text{if } B = \top \\
            \finconst{k}{n+1} \text{ such that } c \in \intervals{B}_k, & \text{if } B = \{\clt{b_1}, \ldots, \clt{b_n}\}
        \end{cases} \\
        \texttt{discretize}(\letkw\; x = e_1\; \inkw\; e_2) 
            &= \letkw\; x = \texttt{discretize}(e_1)\; \inkw\; \texttt{discretize}(e_2) \\
        \texttt{discretize}(e_1 < e_2) 
            &= \texttt{discretize}(e_1) < \texttt{discretize}(e_2) \\
        \texttt{discretize}(\ifkw\; e_1\; \thenkw\; e_2\; \elsekw\; e_3) 
            &= \ifkw\; \texttt{discretize}(e_1) \notag \\ 
            &\quad \thenkw\; \texttt{discretize}(e_2)\; \elsekw\; \texttt{discretize}(e_3) \\
        \texttt{discretize}(x) &= x 
    \end{align}
