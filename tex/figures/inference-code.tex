    let infer (e : expr) : texpr =
        let rec aux (env : ty StringMap.t) (ExprNode e_node : expr) : texpr =
            match e_node with
            | Const f ->
                (TFloat($\emptyset$, f), TAExprNode (Const f))

            | Let (x, e1, e2) ->
                let t1, a1 = aux env e1 in
                let env' = StringMap.add x t1 env in
                let t2, a2 = aux env' e2 in
                (t2, TAExprNode (Let (x, (t1,a1), (t2,a2))))

            | Var x ->
                t = StringMap.find x env in 
                (t, TAExprNode (Var x))

            | Less (e1, e2) ->
                let t1, a1 = aux env e1 in (* Assume t1 = TFloat(B1, V1) *)
                let t2, a2 = aux env e2 in (* Assume t2 = TFloat(B2, V2) *)
                let B =
                    match V1, V2 with
                    | T, T -> T
                    | Finite _, Finite V2 -> V2
                    | Finite V1, T -> V1
                    | T, Finite V2 -> V2
                in
                (TBool, TAExprNode (Less (TFloat(B,V1), a1), (TFloat(B,V2), a2)))

            | If (e1, e2, e3) ->
                let t1, a1 = aux env e1 in (* Assume t1 = TBool *)
                let t2, a2 = aux env e2 in (* Assume t2 = TFloat(B2, V2) *)
                let t3, a3 = aux env e3 in (* Assume t3 = TFloat(B3, V3) *)
                let t = TFloat(B2, V2 $\latticesup$ V3) in (* Assume B1 = B2, t1 <: t, and t2 <: t *)
                (t, TAExprNode (If ((t1, a1), (t2, a3), (t3, a3))))

            | Uniform (e1, e2) ->
                let t1, a1 = aux env e1 in (* Assume t1 = TFloat(B1, V1) *)
                let t2, a2 = aux env e2 in (* Assume t2 = TFloat(B2, V2) *)
                let t = TFloat($\emptyset$, T) in 
                add_floats_to_cutbag V1 B1;
                add_floats_to_cutbag V2 B2;
                make_output_top_if_input_cutbag_is_top B1;
                make_output_top_if_input_cutbag_is_top B2;
                make_input_top_if_output_cutbag_is_top B1 t;
                make_input_top_if_output_cutbag_is_top B2 t;
                (t, TAExprNode (Uniform ((t1, a1), (t2, a2))))

            | (* Other Slice constructs... *)
            
        in aux StringMap.empty e
