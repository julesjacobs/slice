  \node[text=green] at (5.8, -0.3) {$+\infty$};

  \node[text=green] at (0.1, -0.5) {0.1};
  \node[text=green] at (0.8, -0.5) {0.8};
  \node[text=green] at (2.0, -0.5) {2.0};

  % Gaussian curve N(0,1)
  \draw[thick, domain=-4:6, smooth, variable=\x]
      plot ({\x},{2.2*exp(-(\x*\x)/2)});

  % Vertical dotted lines for cuts
  \draw[dotted, thick, gray] (0.1,2.4) -- (0.1,0);
  \draw[dotted, thick, gray] (0.8,2.4) -- (0.8,0);
  \draw[dotted, thick, gray] (2.0,2.4) -- (2.0,0);

  % Shaded continuous intervals 
  \fill[blue,   opacity=0.5] (-4,0) -- plot[domain=-4:0.1] (\x,{2.2*exp(-(\x*\x)/2)}) -- (0.1,0) -- cycle;
  \fill[red,    opacity=0.5] (0.1,0) -- plot[domain=0.1:0.8] (\x,{2.2*exp(-(\x*\x)/2)}) -- (0.8,0) -- cycle;
  \fill[purple, opacity=0.5] (0.8,0) -- plot[domain=0.8:2.0] (\x,{2.2*exp(-(\x*\x)/2)}) -- (2.0,0) -- cycle;
  \fill[olive, opacity=0.5] (2.0,0) -- plot[domain=2.0:6] (\x,{2.2*exp(-(\x*\x)/2)}) -- (6,0) -- cycle;

  % Discrete distribution (equal-width bars)
  \def\shift{0.7}
  \def\barW{1.4}
  \def\barY{-3.2}

  % TOPS of bars (with areas proprotional to true CDF masses)
  \def\hA{-1.6}        % 1.600 = 0.5398 * k
  \def\hB{-2.464}      % 0.736 = 0.2483 * k
  \def\hC{-2.639}      % 0.561 = 0.1891 * k
  \def\hD{-3.132}      % 0.068 = 0.0228 * k

  % Left edges
  \def\xA{-2*\barW + \shift}
  \def\xB{-1*\barW + \shift}
  \def\xC{0 + \shift}
  \def\xD{\barW + \shift}

  % Bars
  \draw (\xA,\hA) rectangle ++(\barW,\barY-\hA);
  \fill[blue, opacity=0.5] (\xA,\hA) rectangle ++(\barW,\barY-\hA);

  \draw (\xB,\hB) rectangle ++(\barW,\barY-\hB);
  \fill[red, opacity=0.5] (\xB,\hB) rectangle ++(\barW,\barY-\hB);

  \draw (\xC,\hC) rectangle ++(\barW,\barY-\hC);
  \fill[purple, opacity=0.5] (\xC,\hC) rectangle ++(\barW,\barY-\hC);

  \draw (\xD,\hD) rectangle ++(\barW,\barY-\hD);
  \fill[olive, opacity=0.5] (\xD,\hD) rectangle ++(\barW,\barY-\hD);

  % Bar labels
  \node[text=orange] at (\xA+0.5*\barW,\barY-0.3) {0};

  % (-∞) → top of bar 0 left edge

  \draw[dotted, thick, gray]

  \draw[dotted, thick, gray]

  \draw[dotted, thick, gray]
