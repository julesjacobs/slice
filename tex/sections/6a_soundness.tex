\begin{definition}
	\label{def:smallstep_space_and_monad}
	Let $\tau$ be a type.
 	Given a suitable measurable space, we denote by $\mathcal{D}(Expr_{\tau} + \bot)$ the set of sub-probability measures over $Expr_{\tau} + \bot$. Moreover, for all $\tau,\tau'$, we define the  monad
 	%
 	\begin{align*}
 		 & \text{return} \colon Expr_{\tau} \to \mathcal{D}(Expr_{\tau} + \bot), \quad \text{return}(e) = \delta_e \\
 		 %
 		 & \text{bind} \colon \mathcal{D}(Expr_{\tau} + \bot) \to (Expr_{\tau} \to \mathcal{D}(Expr_{\tau'} + \bot)) \to \mathcal{D}(Expr_{\tau'} + \bot), \\
 		 %& \text{bind}(\mu)(f)(y) := \sum_{x \in support(\mu)} \mu(x) \cdot f_{\bot}(x)(y) \\
 		 &\text{bind}(\mu)(f)(A) = \int_{x \in Expr + \bot} f_{\bot}(x)(A) \cdot \mu(dx)~,
 	\end{align*}
 	%
 	where $f_\bot \colon Expr_{\tau} + \bot \to \mathcal{D}(Expr_{\tau'} + \bot)$ is obtained from $f$ by 
 	%
 	\[
 		f_\bot(e) =
 		\begin{cases}
 			\delta_\bot & \text{if } e =\bot \\
 			%
 			f(e) & \text{otherwise}~.
 		\end{cases}
 	\]
\end{definition}
%
%
%
%
Equipped with these notions, we define the small-step semantics of our language:
%
\begin{definition}
	The small-step semantics $\sem{e :\tau} \in \mathcal{D}(Expr_{\tau} + \bot)$ is defined recursively on the structure of $e:\tau$ by the rules in \Cref{tab:smallstep}. 
\end{definition}
%
The missing probability mass in $\sem{e :\tau}$ is the probability of encountering an error due to ill-formed parameters fed into distributions (e.g., $\uniform(0.8,0.1)$). $\sem{e :\tau}(\bot)$ is the probability of encountering an observation failure.
It follows by induction on the type derivation of $e:\tau$ that $\sem{e:\tau}$ is well-defined, i.e., the small-step semantics is type-preserving.\\

Define $P_\tau = \{ (e:\tau, \discretize{e} : \discretize{\tau}) ~|~ e:\tau \}$.

\begin{theorem}
	Let $e:\tau$. We have:
	%
	\begin{enumerate}
		\item $\sem{e:\tau}(\{\bot\}) = \sem{e:\tau, \discretize{e} : \discretize{\tau}}(\{\bot\}) = \sem{\discretize{e} : \discretize{\tau}}(\{\bot\})$.
		%
		\item For all measurable $E \subseteq Expr_\tau$, we have \[\sem{e:\tau}(E) = \sem{e:\tau, \discretize{e} : \discretize{\tau}}( E \times Expr_{\discretize{\tau}} )~. \]
		%
		%
		\item For all measurable $E \subseteq Expr_{\discretize{\tau}}$, we have \[\sem{\discretize{e} : \discretize{\tau}}(E) = \sem{e:\tau, \discretize{e} : \discretize{\tau}}( Expr_\tau \times E )~. \]
		%
		\item For all measurable $E\subseteq Expr_\tau \times Expr_{\discretize{\tau}}$, we have 
		%
		\[
			\sem{e:\tau, \discretize{e} : \discretize{\tau}}(E) = \sem{e:\tau, \discretize{e} : \discretize{\tau}}(E \cap P_\tau)~.
		\]
	\end{enumerate}
\end{theorem}
%
\begin{proof}
	By induction on the type derivation of $e:\tau$.
\end{proof}

\subsection{Big-Step Semantics and Correctness of Discretization}
%
We extend both of the small-step semantics canoncially to an $n$-step semantics by%
\begin{align*}
		\bigsemstep{e : \tau}{n} &{}= 
		\begin{cases}
			\delta_{e : \tau} & \text{if $n = 0$} \\
			%
			\bigsemstep{e : \tau}{n-1} \monbind  \lambda e'. \sem{e' : \tau} & \text{if $n > 0$}~.
		\end{cases} \\
		%
		%
		\bigsemstep{e : \tau, \discretize{e} : \discretize{\tau}}{n} &{}= 
		\begin{cases}
			\delta_{(e : \tau, \discretize{e} : \discretize{\tau})} & \text{if $n = 0$} \\
			%
			\bigsemstep{e : \tau, \discretize{e} : \discretize{\tau}}{n-1} \monbind  \lambda (e', e''). \sem{e' : \tau, e'' : \discretize{\tau}} & \text{if $n > 0$}~.
		\end{cases}
\end{align*}
%
%
For every type $\tau$, we denote by $Vals_\tau$ the set of values of type $\tau$ and equip it with a suitable measurable sub-space of the measurable space over $Expr_\tau$.
%
\begin{definition}
	Let $e:\tau$. The \emph{(big-step) semantics} $\bigsem{e:\tau} \in \mathcal{D}(Vals_\tau + \bot)$ and $\bigsem{e:\tau} \in \mathcal{D}(Vals_\tau \times Vals_{\discretize{\tau}} + \bot)$ are respectively defined as 
	%
	\begin{align*}
		\bigsem{e:\tau} ={}& \lambda A. \sup_n \bigsemstep{e:\tau}{n}(A) \\
		%
		\bigsem{e:\tau,  \discretize{e} : \discretize{\tau}} ={}& \lambda A. \sup_n \bigsemstep{e:\tau,  \discretize{e} : \discretize{\tau}}{n}(A)
	\end{align*}
	%
\end{definition}
%
%
