Equipped with these notions, we define the small-step semantics of our language:
%
\begin{definition}
	The small-step semantics $\sem{e :\tau} \in \mathcal{D}(Expr_{\tau} + \bot)$ is defined recursively on the structure of $e:\tau$ by the rules in \Cref{tab:smallstep}. 
\end{definition}
%
The missing probability mass in $\sem{e :\tau}$ is the probability of encountering an error due to ill-formed parameters fed into distributions (e.g., $\uniform(0.8,0.1)$). $\sem{e :\tau}(\bot)$ is the probability of encountering an observation failure.
It follows by induction on the type derivation of $e:\tau$ that $\sem{e:\tau}$ is well-defined, i.e., the small-step semantics is type-preserving.\\

We now define suitable projection functions on measures $\mu \in \mathcal{D}(P_{\tau} + \bot)$ by 
%
\begin{align*}
	 \pi_1(\mu) &{}={} \mu \gg\!= \lambda(e_1,e_2). \delta_{e_1} \\
	 %
	 \pi_1(\mu) 2&{}={} \mu \gg\!= \lambda(e_1,e_2). \delta_{e_2}~.
\end{align*}
%
%
We have the following key property:
%
\begin{lemma}
	\label{lem:small_step_coupling_props}
	Let $e:\tau$. We have:
	%
	\begin{enumerate}
		\item\label{lem:small_step_coupling_props1} $\pi_1(\sem{e:\tau, \discretize{e} : \discretize{\tau}}) = \sem{e:\tau}$.
		%
		\item\label{lem:small_step_coupling_props2} $\pi_2(\sem{e:\tau, \discretize{e} : \discretize{\tau}}) = \sem{\discretize{e}:\discretize{\tau}}$.
	\end{enumerate}
\end{lemma}
%
\begin{proof}
	By induction on the type derivation of $e:\tau$. The most interesting case is 
	\[
	    e:\tau = \uniform(c_1 : \floattype{B_1}{V_1}, c_2 : \floattype{B_2}{V_2}) : \floattype{B}{V}~,
	\]
	%
	where
	%
	\begin{enumerate}
		\item $c_1 \leq c_2$,
		%
		\item $B =  \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\}$,
		%
		\item $V=\top$, 
		%
		\item $V_1 = \{v_1,\ldots, v_{n_1}\}$ with $\distinguishes{B_1}{V_1}$ and $c_1 \in V_1$ due to $\has{V_1}{c_1}$ by I.H. 
		%
		\item $V_2 = \{w_1,\ldots, w_{n_2}\}$ with $\distinguishes{B_2}{V_2}$ and  $c_2 \in V_2$ due to $\has{V_2}{c_2}$ by I.H. 
	\end{enumerate}
	%
	First observe that we have 
	%
	\begin{align*}
		&\discretize{ e:\tau} : \discretize{\tau} \\
		%
		{}={}& \dc(\discretize{c_1 : \floattype{B_1}{V_1}}, \discretize{c_2 :  \floattype{B_2}{V_2}}) : \fin{|B| + 1}\\
		%
		{}={}&\dc( k_1 : \fin{|B_1| + 1}, k_2:\fin{|B_2|+1} ) : \fin{|B|+1}~,
	\end{align*}
	%
	where $k_i$ is the unique number such that $c_i \in \intervals{B_i}_{k_i}$. By \Cref{tab:smallstep} and \Cref{fig:discretization-code}, this yields
	%
	\begin{align*}
		 & \sem{\discretize{e:\tau}} \\
		 %
		 {}={}& \sem{\discrete(p_0,\ldots,p_n)}~\text{where}~p_k = \text{CDF}(b_{k+1}) - \text{CDF}(b_k) = \frac{b_{k+1} - c_1}{c_2-c_1} - \frac{b_{k} - c_1}{c_2-c_1}  =  \frac{b_{k+1} - b_k}{c_2-c_1}~. 
	\end{align*}
	%
	
	For \Cref{lem:small_step_coupling_props}.\ref{lem:small_step_coupling_props2}, consider the following:
	%
	\begin{align*}
		 & \pi_1(\sem{e:\tau, \discretize{e} : \discretize{\tau}}) \\
		 %
		 ~{}={}~ &  \sem{e:\tau, \discretize{e} : \discretize{\tau}} \gg\!= \lambda(e_1,e_2). \delta_{e_1}  
		 \tag{definition} \\
		 %
		 ~{}={}~ & (\text{Uniform}(c_1, c_2) \gg\!= \lambda v.\delta_{(v : \floattype{B}{V}, v_i : \fin{|B| + 1})})\gg\!= \lambda(e_1,e_2). \delta_{e_1}  
		 \tag{\Cref{tab:coupling_smallstep}, $v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$} \\
		 %
		 ~{}={}~ & \text{Uniform}(c_1, c_2) \gg\!= \lambda v.(\delta_{(v : \floattype{B}{V}, v_i : \fin{|B| + 1})} \gg\!= \lambda(e_1,e_2). \delta_{e_1}  )
		 \tag{monad assoc., $v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$} \\
		 %
		 ~{}={}~ & \text{Uniform}(c_1, c_2) \gg\!= \lambda v. \delta_{v : \floattype{B}{V}}  
		 \\
		 %
		 ~{}={}~ & \text{Uniform}(c_1, c_2) \\
		 %
		 ~{}={}~ & \sem{e:\tau}~.
		 \tag{\Cref{tab:smallstep}}
		 %
		 %
	\end{align*}
	
	For \Cref{lem:small_step_coupling_props}.\ref{lem:small_step_coupling_props2}, consider the following:
	%
	\begin{align*}
		& \pi_2(\sem{e:\tau, \discretize{e} : \discretize{\tau}}) \\
		%
		~{}={}~ &  \sem{e:\tau, \discretize{e} : \discretize{\tau}} \gg\!= \lambda(e_1,e_2). \delta_{e_2}  
		\tag{definition} \\
		%
		~{}={}~ & (\text{Uniform}(c_1, c_2) \gg\!= \lambda v.\delta_{(v : \floattype{B}{V}, v_i : \fin{|B| + 1})})\gg\!= \lambda(e_1,e_2). \delta_{e_2}  
		\tag{\Cref{tab:coupling_smallstep}, $v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$} \\
		%
		~{}={}~ & \text{Uniform}(c_1, c_2) \gg\!= \lambda v.(\delta_{(v : \floattype{B}{V}, v_i : \fin{|B| + 1})} \gg\!= \lambda(e_1,e_2). \delta_{e_2}  )
		\tag{monad assoc., $v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$} \\
		%
		~{}={}~ & \text{Uniform}(c_1, c_2) \gg\!= \lambda v. \delta_{v_i : \fin{|B| + 1}}  
		\tag{$v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$}
		\\
		%
		~{}={}~ & \sem{\discrete(p_0,\ldots,p_n)}~\text{where}~p_k = \text{CDF}(b_{k+1}) - \text{CDF}(b_k) = \frac{b_{k+1} - c_1}{c_2-c_1} - \frac{b_{k} - c_1}{c_2-c_1}  =  \frac{b_{k+1} - b_k}{c_2-c_1} 
		\tag{by construction} \\
		%
		~{}={}~ & \sem{\discretize{e}:\discretize{\tau}}~.
		\tag{\Cref{tab:smallstep}}
		%
		%
	\end{align*}
\end{proof}



\subsection{Big-Step Semantics and Correctness of Discretization}
%
We extend both of the small-step semantics canoncially to an $n$-step semantics by%
\begin{align*}
		\bigsemstep{e : \tau}{n} &{}= 
		\begin{cases}
			\delta_{e : \tau} & \text{if $n = 0$} \\
			%
			\bigsemstep{e : \tau}{n-1} \monbind  \lambda e'. \sem{e' : \tau} & \text{if $n > 0$}~.
		\end{cases} \\
		%
		%
		\bigsemstep{e : \tau, \discretize{e} : \discretize{\tau}}{n} &{}= 
		\begin{cases}
			\delta_{(e : \tau, \discretize{e} : \discretize{\tau})} & \text{if $n = 0$} \\
			%
			\bigsemstep{e : \tau, \discretize{e} : \discretize{\tau}}{n-1} \monbind  \lambda (e', e''). \sem{e' : \tau, e'' : \discretize{\tau}} & \text{if $n > 0$}~.
		\end{cases}
\end{align*}
%
%

%
%
For every type $\tau$, we denote by $Vals_\tau$ the set of values of type $\tau$ and equip it with a suitable measurable sub-space of the measurable space over $Expr_\tau$. Moreover, let $V_\tau = \{ (v:\tau, \discretize{v} : \discretize{\tau}) ~|~ v:\tau \}$.
%
\begin{definition}
	Let $e:\tau$. The \emph{(big-step) semantics} $\bigsem{e:\tau} \in \mathcal{D}(Vals_\tau + \bot)$ and $\bigsem{e:\tau} \in \mathcal{D}(V_\tau + \bot)$ are respectively defined as 
	%
	\begin{align*}
		\bigsem{e:\tau} ={}& \lambda A. \lim_{n\to\infty} \bigsemstep{e:\tau}{n}(A) \\
		%
		\bigsem{e:\tau,  \discretize{e} : \discretize{\tau}} ={}& \lambda A. \lim_{n\to\infty} \bigsemstep{e:\tau,  \discretize{e} : \discretize{\tau}}{n}(A)
	\end{align*}
	%
\end{definition}
%
%
Well-definedness of these objects follows from the monotone convergence theorem: The probability of evaluating to some set of values or an error is montonically increasing in $n$ and upper-bounded by $1$. The following will imply our desired equivalence claim:

\begin{lemma}
	\label{lem:big_step_coupling_props}
	Let $e:\tau$. We have:
	%
	\begin{enumerate}
		\item\label{lem:big_step_coupling_props1} $\pi_1(\sem{e:\tau, \discretize{e} : \discretize{\tau}}) = \bigsem{e:\tau}$.
		%
			\item\label{lem:big_step_coupling_props2} $\pi_2(\sem{e:\tau, \discretize{e} : \discretize{\tau}}) = \bigsem{\discretize{e}:\discretize{\tau}}$.
	\end{enumerate}
\end{lemma}

\begin{theorem}
	Let $e:\bool$. Then $\bigsem{e:bool} = \bigsem{\discretize{e:bool} : \bool}$.
\end{theorem}
%
\begin{proof}
	Since $\mathcal{D}(Vals_\tau + \bot)$ contains only discrete distributions, we proceed by caste distinction on the probality of the final outcome $v\in\{\true,\false,\bot\}$.\\
	%
	
	\emph{The case $v=\true$.} We have 
	%
	\begin{align*}
		&  \bigsem{e : \bool}(\true) \\
		%
		{}={}~& \pi_1 (\bigsem{e : \bool, \discretize{e}:\discretize{\bool}})(\true)
		\tag{\Cref{lem:big_step_coupling_props}.\ref{lem:big_step_coupling_props1}} \\
		%
		{}={}~& \bigsem{e : \bool, \discretize{e}:\discretize{\bool}} \gg\!= \lambda(e_1,e_2). \delta_{e_1}
		\tag{definition} \\
		%
		{}={}~& \bigsem{e : \bool, \discretize{e}:\discretize{\bool}}((\true,\true))\cdot \delta_{\true}(\true) \\
		&{}+ 
		\bigsem{e : \bool, \discretize{e}:\discretize{\bool}}((\true,\true))\cdot \delta_{\false}(\true) \\
		&{}+ 
		\bigsem{e : \bool, \discretize{e}:\discretize{\bool}}((\true,\true))\cdot \delta_{\bot}(\true)

		%
		{}={}~& \bigsem{e : \bool, \discretize{e}:\discretize{\bool}}((\true,\true))
		\tag{definition of $\delta$} \\
		%
		{}={}~& \pi_2(\bigsem{e : \bool, \discretize{e}:\discretize{\bool}})(\true)
		\tag{reasoning completely analogous} \\
		%

	\end{align*}
	
	The cases $v=\false$ and $v=\bot$ are completely analogous.
