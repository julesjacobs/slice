\subsubsection{Brief recap of measure theory} 
We first give a brief overiew of basic measure theory terms.

\paragraph{Topological space} A topological space is a pair $(X, \mathcal{T})$ where $\mathcal{T}$ is a collection of subsets of $X$, called open sets, such that $\emptyset, X \in \mathcal{T}$, and $\mathcal{T}$ is closed under arbitrary unions and finite intersections.

\paragraph{$\sigma$-algebras} A $\sigma$-algebra on a set $X$ is a collection $\Sigma$ of subsets of $X$ that includes $X$ itself and is closed under complement and under countable unions. By implication, a $\sigma$-algebra includes the empty set and is closed under countable intersections. 
Given a topological space $(X, \mathcal{T})$, the Borel $\sigma$-algebra $\mathcal{B}(X)$ is the smallest $\sigma$-algebra containing all open sets in $\mathcal{T}$.

\paragraph{Measurable spaces} A measurable space $(X, \Sigma)$ consists of an underlying set and its $\sigma$-algebra.  
When $\Sigma$ is the Borel $\sigma$-algebra on a topological space $X$, we call $(X, \Sigma)$ a Borel measurable space.

\paragraph{Measures} Let $(X,\Sigma)$ be a measurable space. A function $\mu : \Sigma \to \R^+$ is called a measure if it satisfies nonnegativity, null empty set, and countable additivity. 
When $\Sigma$ is a Borel $\sigma$-algebra, $\mu$ is called a Borel measure.

\paragraph{Measure spaces} The triple $(X, \Sigma, \mu)$ is called a measure space.
 When $\Sigma$ is the Borel $\sigma$-algebra and $\mu$ is the Borel measure, $(X, \Sigma, \mu)$ is called a Borel measure space.

\paragraph{(Sub)probability measures} When $\mu(X) \leq 1$, it is called a (Borel) subprobability measure. When $\mu(X) = 1$, it is called a (Borel) probability measure and $(X, \Sigma, \mu)$ a (Borel) probability space.

\paragraph{Product measure} The product of two measurable spaces $(X_1, \Sigma_1)$ and $(X_2, \Sigma_2)$ is also a measurable space, namely $(X_1 \times X_2, \Sigma_1 \otimes \Sigma_2)$.

\begin{definition}[Support]
	We define the support of a measure separately for discrete and continuous cases:
	\begin{enumerate}
		\item Let $(X, \Sigma, \mu)$ be a measure space where $\mu$ is a discrete measure. The support of $\mu$ is the set of all points with positive measure:
		\[
			{support}(\mu) \triangleq \{ x \in X \mid \mu(x) > 0 \}.
		\]

		\item Let $(X, \Sigma, \mu)$ be a Borel measure space where $X$ is a topological space and $\mu$ is a continuous measure. The support of $\mu$ is:
		\[
			{support}(\mu) \triangleq \{ x \in X \mid \forall U \in \Sigma, \; (U \text{ open and } x \in U) \implies \mu(U) > 0 \}.
		\]
	\end{enumerate}
\end{definition}

\begin{definition}[Projection with error]
	Let $(X + \bot, \Sigma_{X + \bot})$ and $(Y + \bot, \Sigma_{Y + \bot})$ be measurable spaces, with $(X, \Sigma_X)$ and $(Y, \Sigma_Y)$ measurable spaces as well. Let $\mu \in \mathcal{D}((X \times Y) + \bot)$. We define the first and second projection measures as:
	\begin{enumerate}
	\item $\pi_1(\mu)(A) := \mu(A \times Y)$ for every measurable $A \in \Sigma_X$, and $\pi_1(\mu)({\bot}) := \mu({\bot})$
	\item $\pi_2(\mu)(B) := \mu(X \times B)$ for every measurable $B \in \Sigma_Y$, and $\pi_2(\mu)({\bot}) := \mu({\bot})$
	\end{enumerate}
	Then, $\pi_1(\mu)$ is a measure on $(X + \bot, \Sigma_{X + \bot})$ and $\pi_2(\mu)$ is a measure on $(Y + \bot, \Sigma_{Y + \bot})$.
\end{definition}


\subsection{Measure space on expressions}
To make the probabilistic semantics precise, we define an appropriate measure space structure on the set of expressions. We construct a Borel measurable space on $Expr_{\tau}$ by factoring out real numbers using the notion of what we call \emph{expression skeletons}.

\begin{definition}[Expression Skeleton]
	An expression skeleton $s : \tau$ is a well-typed expression $e : \tau$ where each occurrence of a float constant is replaced by a hole $\square$, generated as follows:
	\begin{align*}
		s ::= \; 
		&\square \mid x \mid \true \mid \false \mid \finconst{k}{n} \mid \letkw \; x = s_1 \; \inkw \; s_2 \mid \ifkw \; s_1 \; \thenkw \; s_2 \; \elsekw \; s_3 \mid s_1 < s_2 \mid \uniform(s_1, s_2)
	\end{align*}
	with the understanding that each hole $\square$ has type $\floattype{B}{V}$, and that all other types appearing in the original expression are preserved in its corresponding skeleton.
	
	We denote by $\text{Skel}_{\tau}$ the set of all expression skeletons, and by $\holes(s)$ the number of distinct holes in skeleton $s$.
\end{definition}

\begin{example}
	The expression $\uniform(0.0, 1.0) < 0.5$ has the skeleton $\uniform(\square, \square) < \square$ with three holes. 
\end{example}

\begin{lemma}[Skel is countable]\label{lem:skel-countable}
	The set $\text{Skel}_{\tau}$ is countable.
\end{lemma}

\begin{proof}
	Each skeleton is a finite syntactic tree where:
	\begin{itemize}
	\item Internal nodes are labeled by constructors from a finite set of ${\letkw, \ifkw, <, \uniform}$
	\item Leaves are labeled by literals or variables (countable) or holes $\square$ (countable)
	\end{itemize}
	The set of all finite trees over countable labels is countable.
\end{proof}

\begin{definition}[Measurable space on well-typed expressions]
	For a given $s : \tau \in \text{Skel}_{\tau}$, define the subset:
	\[
		A_{s : \tau} := \{f(e : \tau)_2 \mid f(e : \tau)_1 = s : \tau \text{ and } e : \tau \in Expr_{\tau}\}.
	\]
	Let $s[\vec{r}] : \tau$ denote the (unique) well-typed expression of type $\tau$ obtained by filling the holes of $s$ with the coordinates of $\vec{r}$. We then define the measurable space of well-typed expressions as the union:
	\[
		Expr_{\tau} := \bigcup_{s : \tau \in \text{Skel}_{\tau}} \{s[\vec{r}] : \tau \mid \vec{r} \in A_{s : \tau}\}.
	\]
\end{definition}

\begin{example}
	Given $s : \tau = (\square : \floattype{\clt{3.5}, \clt{4.5}}{\top} < \square : \floattype{\clt{3.5}, \clt{4.5}}{3.5,4.5}) : \bool$, we compute $A_{s : \tau}$.

	\begin{align*}
		f(e : \bool)_1 
			&= (\square : \floattype{\clt{3.5}, \clt{4.5}}{\top} 
				< \square : \floattype{\clt{3.5}, \clt{4.5}}{3.5,4.5}) : \bool \\[6pt]
		\iff\;& 
			e = e_1 < e_2 \\[3pt]
		&\quad 
			f(e_1 : \floattype{\clt{3.5}, \clt{4.5}}{\top})_1 
			= \square : \floattype{\clt{3.5}, \clt{4.5}}{\top}, \\[3pt]
		&\quad 
			f(e_2 : \floattype{\clt{3.5}, \clt{4.5}}{3.5,4.5})_1 
				= \square : \floattype{\clt{3.5}, \clt{4.5}}{3.5,4.5} \\[6pt]
		\iff\;& 
			e_1 = c \in \R, \quad e_2 \in \{3.5, 4.5\}.
	\end{align*}
	
	Thus,
	\begin{align*}
		f(c : \floattype{\clt{3.5}, \clt{4.5}}{\top})_2 &= \R, \\[3pt]
		f(3.5 : \floattype{\clt{3.5}, \clt{4.5}}{3.5,4.5})_2 &= \{3.5\}, \\[3pt]
		f(4.5 : \floattype{\clt{3.5}, \clt{4.5}}{3.5,4.5})_2 &= \{4.5\}.
	\end{align*}
	
	Hence,
	\begin{align*}
		A_{s : \tau} 
			&= \{(x, 3.5) \mid x \in \R\} \cup \{(x, 4.5) \mid x \in \R\} \\[3pt]
			&= \R \times \{3.5, 4.5\} \subseteq \R^2.
	\end{align*}	
\end{example}

\begin{lemma}
	$(Expr_{\tau}, \mathcal{F}_{Expr_{\tau}})$ is a Borel measurable space.
\end{lemma}

\begin{proof}
	Follows by structural recursion on expressions. 
	
	Fix an $s \in \text{Skel}$. At the base cases, we either get a finite value set $V$ (a finite union of Borel singletons, which is Borel), $V = \top$ (the entire $\R$, which is Borel), or $\emptyset$ (Borel). Each construct combines its subexpressions through Cartesian products (Borel sets are closed under finite products).

	Since $\text{Skel}$ is countable by Lemma~\ref{lem:skel-countable}, and since Borel sets are closed under countable unions, $Expr_{\tau}$ forms a Borel measurable space.
\end{proof}

\begin{definition}[Extension with error element]\label{def:meas-space-with-error}
	We extend the Borel measurable space of expressions to include a distinguished error element $\bot$, defining the measurable space of extended expressions as
	\[
		Expr_{\tau} + \bot := Expr_{\tau} \cup \{\bot\},
	\]
	with the topology $\mathcal{T}_{Expr_{\tau} + \bot} = \{U, U \cup \{\bot\} \mid U \in \mathcal{T}_{Expr_{\tau}}\}$, equipped with the Borel $\sigma$-algebra $\mathcal{F}_{Expr_{\tau} + \bot} := \sigma(\mathcal{T}_{Expr_{\tau} + \bot})$.
\end{definition}

\begin{lemma}
	$(Expr_{\tau} + \bot, \mathcal{F}_{Expr_{\tau} + \bot})$ is a Borel measurable space.
\end{lemma}

\begin{proof}
	$Expr_{\tau}$ is itself a Borel set, and $\{\bot\}$ is an open singleton in the extended topology. The union of a Borel set with an open set is Borel. 
\end{proof}

% Intuitively, this means that measurable subsets of $Expr_{\tau}$ can be described by first fixing a finite syntactic skeleton and then selecting a measurable subset of real assignments to its float constants.

We can equip this measurable space with a (sub)probability measure such that $\mu(Expr_{\tau} + \bot) \leq 1$. And in particular, any (sub)probability measure $\mu$ on $\R^{\holes(s)}$ for a skeleton $s$ can be lifted to a (sub)probability measure on $Expr_{\tau} + \bot$, as follows: 
\[
	\mu_{Expr_{\tau} + \bot}(A) = \mu(\{\vec{r} \in \R^{\holes(s)} \mid s[\vec{r}] \in A\}).
\]

We give some basic lemmas:

\begin{lemma} 
	Let $(Expr_{\tau} + \bot, \mathcal{F}_{Expr_{\tau} + \bot})$ be the measurable space of expressions with error from Definition~\ref{def:meas-space-with-error}. Let $V \subset Expr_{\tau}$ be the set of all value expressions. Let $N \subset Expr_{\tau}$, defined as $N = Expr_{\tau} \setminus V$, be the set of non-value expressions. Let $\{\bot\}$ be the singleton set containing the error element $\bot$. The following are true:
	\begin{enumerate}
		\item The set $\vals$ is measurable.
		\item The set $\{\bot\}$ is measurable.
		\item The set $\nonvals$ is measurable.
	\end{enumerate}
\end{lemma}

\begin{proof}
	We prove the measurability of $\vals$, $\{\bot\}$, and $N$ in separate parts.
	\begin{enumerate}
		\item \textbf{(Measurability of $\vals$):} By the construction of $\mathcal{F}_{Expr_{\tau}}$, for each skeleton $s$ that represents a value, the set $\{\vec{r} \in \R^{\holes(s)} \mid \vdash s[\vec{r}] : \tau\}$ is a measurable set in the union. The set $V$ is the countable union of all such sets corresponding to value skeletons. Since $\text{Skel}$ is countable, $V$ is a countable union of measurable sets and is therefore itself measurable, e.g. $V \in \mathcal{F}_{Expr_{\tau}}$. As $\mathcal{F}_{Expr_{\tau}} \subset \mathcal{F}_{Expr_{\tau} + \bot}$, it follows that $V \in \mathcal{F}_{Expr_{\tau} + \bot}$.
		
		\item \textbf{(Measurability of $\{\bot\}$):} By Definition~\ref{def:meas-space-with-error}, the topology $\mathcal{T}_{Expr_{\tau} + \bot}$ contains all sets of the form $U \cup \{\bot\}$, where $U \in \mathcal{T}_{Expr_{\tau}}$. In particular, taking $U = \emptyset$ gives $\emptyset \cup \{\bot\} = \{\bot\} \in \mathcal{T}_{Expr_{\tau} + \bot}$. Since the Borel $\sigma$-algebra $\mathcal{F}_{Expr_{\tau} + \bot}$ is generated by $\mathcal{T}_{Expr_{\tau} + \bot}$, it follows that $\{\bot\} \in \mathcal{F}_{Expr_{\tau} + \bot}$.

		\item \textbf{(Measurablity of $N$):} By definition, $N$ is the set of non-value expressions, e.g. $N = Expr_{\tau} \setminus V$. From part i), we have established that $V \in \mathcal{F}_{Expr_{\tau}}$. Since $\mathcal{F}_{Expr_{\tau}}$ is a $\sigma$-algebra and is closed under complement, the complement of $V$ within $Expr_{\tau}$, which is precisely $N$, is also measurable, e.g. $N \in \mathcal{F}_{Expr_{\tau}}$. As $\mathcal{F}_{Expr_{\tau}} \subset \mathcal{F}_{Expr_{\tau} + \bot}$, it follows that $N \in \mathcal{F}_{Expr_{\tau} + \bot}$.
	\end{enumerate}
\end{proof}

Note that in general, given $(X, \Sigma_X)$ and $(Y, \Sigma_Y)$ to be measurable spaces, $((X \times Y) + \bot)$ forms a measurable space, since the product $(X \times Y)$ is measurable and for any measurable space $(Z, \Sigma_Z)$, the space $(Z + \bot, \Sigma_{Z + \bot})$ with $\Sigma_{Z + \bot}$ generated by $\Sigma_Z \cup {{\bot}}$ is measurable.

		%
		\item $V_2 = \{w_1,\ldots, w_{n_2}\}$ with $\distinguishes{B_2}{V_2}$ and  $c_2 \in V_2$ due to $\has{V_2}{c_2}$ by I.H. 
	\end{enumerate}
	%
	First observe that we have 
	%
	\begin{align*}
		&\discretize{ e:\tau} : \discretize{\tau} \\
		%
		{}={}& \dc(\discretize{c_1 : \floattype{B_1}{V_1}}, \discretize{c_2 :  \floattype{B_2}{V_2}}) : \fin{|B| + 1}\\
		%
		{}={}&\dc( k_1 : \fin{|B_1| + 1}, k_2:\fin{|B_2|+1} ) : \fin{|B|+1}~,
	\end{align*}
	%
	where $k_i$ is the unique number such that $c_i \in \intervals{B_i}_{k_i}$. By \Cref{tab:smallstep} and \Cref{fig:discretization-code}, this yields
	%
	\begin{align*}
		 & \sem{\discretize{e:\tau}} \\
		 %
		 {}={}& \sem{\discrete(p_0,\ldots,p_n)}~\text{where}~p_k = \text{CDF}(b_{k+1}) - \text{CDF}(b_k) = \frac{b_{k+1} - c_1}{c_2-c_1} - \frac{b_{k} - c_1}{c_2-c_1}  =  \frac{b_{k+1} - b_k}{c_2-c_1}~. 
	\end{align*}
	%
	
	For \Cref{lem:small_step_coupling_props}.\ref{lem:small_step_coupling_props2}, consider the following:
	%
	\begin{align*}
		 & \pi_1(\sem{e:\tau, \discretize{e} : \discretize{\tau}}) \\
		 %
		 ~{}={}~ &  \sem{e:\tau, \discretize{e} : \discretize{\tau}} \gg\!= \lambda(e_1,e_2). \delta_{e_1}  
		 \tag{definition} \\
		 %
		 ~{}={}~ & (\text{Uniform}(c_1, c_2) \gg\!= \lambda v.\delta_{(v : \floattype{B}{V}, v_i : \fin{|B| + 1})})\gg\!= \lambda(e_1,e_2). \delta_{e_1}  
		 \tag{\Cref{tab:coupling_smallstep}, $v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$} \\
		 %
		 ~{}={}~ & \text{Uniform}(c_1, c_2) \gg\!= \lambda v.(\delta_{(v : \floattype{B}{V}, v_i : \fin{|B| + 1})} \gg\!= \lambda(e_1,e_2). \delta_{e_1}  )
		 \tag{monad assoc., $v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$} \\
		 %
		 ~{}={}~ & \text{Uniform}(c_1, c_2) \gg\!= \lambda v. \delta_{v : \floattype{B}{V}}  
		 \\
		 %
		 ~{}={}~ & \text{Uniform}(c_1, c_2) \\
		 %
		 ~{}={}~ & \sem{e:\tau}~.
		 \tag{\Cref{tab:smallstep}}
		 %
		 %
	\end{align*}
	
	For \Cref{lem:small_step_coupling_props}.\ref{lem:small_step_coupling_props2}, consider the following:
	%
	\begin{align*}
		& \pi_2(\sem{e:\tau, \discretize{e} : \discretize{\tau}}) \\
		%
		~{}={}~ &  \sem{e:\tau, \discretize{e} : \discretize{\tau}} \gg\!= \lambda(e_1,e_2). \delta_{e_2}  
		\tag{definition} \\
		%
		~{}={}~ & (\text{Uniform}(c_1, c_2) \gg\!= \lambda v.\delta_{(v : \floattype{B}{V}, v_i : \fin{|B| + 1})})\gg\!= \lambda(e_1,e_2). \delta_{e_2}  
		\tag{\Cref{tab:coupling_smallstep}, $v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$} \\
		%
		~{}={}~ & \text{Uniform}(c_1, c_2) \gg\!= \lambda v.(\delta_{(v : \floattype{B}{V}, v_i : \fin{|B| + 1})} \gg\!= \lambda(e_1,e_2). \delta_{e_2}  )
		\tag{monad assoc., $v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$} \\
		%
		~{}={}~ & \text{Uniform}(c_1, c_2) \gg\!= \lambda v. \delta_{v_i : \fin{|B| + 1}}  
		\tag{monad return, $v_i$ is unique number s.t. $v \in \intervals{B}_{v_i}$}
		\\
		%
		~{}={}~ & \sem{\discrete(p_0,\ldots,p_n)}~\text{where}~p_k = \text{CDF}(b_{k+1}) - \text{CDF}(b_k) = \frac{b_{k+1} - c_1}{c_2-c_1} - \frac{b_{k} - c_1}{c_2-c_1}  =  \frac{b_{k+1} - b_k}{c_2-c_1} 
		\tag{by construction} \\
		%
		~{}={}~ & \sem{\discretize{e}:\discretize{\tau}}~.
		\tag{\Cref{tab:smallstep}}
		%
		%
	\end{align*}
	
	Now, the remaining cases are either (i) trivial base cases or (ii) immediately follow from the I.H.\ and \Cref{lem:bind_project} further below. We demonstrate this for the case 
	%
	\[
	e:\tau = \uniform(e_1 : \floattype{B_1}{V_1}, e_2 : \floattype{B_2}{V_2}) : \floattype{B}{V}~,
	\]
	%
	where
	%
	\begin{enumerate}
		\item $e_1 = c_1$ is a value but $e_2$ is no value,
		%
		\item $B =  \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\}$,
		%
		\item $V=\top$, 
		%
		\item $V_1 = \{v_1,\ldots, v_{n_1}\}$ with $\distinguishes{B_1}{V_1}$ and $c_1 \in V_1$ due to $\has{V_1}{c_1}$ by I.H. 
		%
		\item $V_2 = \{w_1,\ldots, w_{n_2}\}$ with $\distinguishes{B_2}{V_2}$ and  $c_2 \in V_2$ due to $\has{V_2}{c_2}$ by I.H. 
	\end{enumerate}
	%
	For \Cref{lem:small_step_coupling_props}.\ref{lem:small_step_coupling_props2}, consider the following:
	%
	\begin{align*}
		& \pi_1(\sem{e:\tau, \discretize{e} : \discretize{\tau}}) \\
		%
		~{}={}~ &  \sem{e:\tau, \discretize{e} : \discretize{\tau}} \gg\!= \lambda(e_1,e_2). \delta_{e_1}  
		\tag{definition} \\
		%
		~{}={}~ & \big( \sem{e_2:\tau_2, \discretize{e_2} : \discretize{\tau_2}} \\
		& \gg\!= \lambda (f,f'). \delta_{(\uniform(v_1:\tau_1, f:\tau_2 ) : \tau, \dc(\discretize{v_1}:\discretize{\tau_1}, f': \discretize{\tau_2} ) : \discretize{\tau})} \big)
		 \gg\!= \lambda(e_1,e_2). \delta_{e_1}   \\
		 %
		 ~{}={}~ &  \sem{e_2:\tau_2, \discretize{e_2} : \discretize{\tau_2}} \\
		 & \gg\!= \lambda (f,f'). \big(\delta_{(\uniform(v_1:\tau_1, f:\tau_2 ) : \tau, \dc(\discretize{v_1}:\discretize{\tau_1}, f': \discretize{\tau_2} ) : \discretize{\tau})} 
		 \gg\!= \lambda(e_1,e_2). \delta_{e_1} \big)
		 \tag{monad assoc.} \\
		 %
		 ~{}={}~ &  \sem{e_2:\tau_2, \discretize{e_2} : \discretize{\tau_2}} 
		  \gg\!= \lambda (f,f'). \delta_{\uniform(v_1:\tau_1, f:\tau_2 ) : \tau} 
		 \tag{monad return} \\
		 %
		 ~{}={}~ &  \pi_1(\sem{e_2:\tau_2, \discretize{e_2} : \discretize{\tau_2}})
		 \gg\!= \lambda f. \delta_{\uniform(v_1:\tau_1, f:\tau_2 ) : \tau} 
		 \tag{\Cref{lem:bind_project}.\ref{lem:bind_project1}} \\
		 %
		 ~{}={}~ &  \sem{e_2:\tau_2}
		 \gg\!= \lambda f. \delta_{\uniform(v_1:\tau_1, f:\tau_2 ) : \tau} ~.
		 \tag{$ \sem{e_2:\tau_2, \discretize{e_2} : \discretize{\tau_2}}  =  \sem{e_2:\tau_2}$ by I.H.} \\
		 %
		 ~{}={}~ &  \sem{e : \tau}
		 \tag{\Cref{tab:smallstep}} 
	\end{align*}
\end{proof}

\begin{lemma}
	\label[lemma]{lem:bind_project}
	Let $F\colon P_\tau \to \mathcal{D}(P_\tau + \bot)$, let $G\colon Expr_\tau \to \mathcal{D}(Expr_\tau + \bot)$, and let $G' \colon Expr_\discretize{\tau} \to \mathcal{D}(Expr_\discretize{\tau} + \bot)$. We have:
	%
	\begin{enumerate}
		\item\label{lem:bind_project1} If  $\forall (f,f') \in P_\tau\colon F(f,f') = G(f)$, then for all $\mu \in \mathcal{D}(P_\tau + \bot)$,
		%
		\[
		\pi_1(\mu) \gg\!= \lambda f. G(f)
		~{}={}~
		\mu \gg\!= \lambda (f,f'). F(f,f')~.
		\]
		%
		%
		\item\label{lem:bind_project2} If  $\forall (f,f') \in P_\tau\colon F(f,f') = G'(f)$, then for all $\mu \in \mathcal{D}(P_\tau + \bot)$,
		%
		\[
		\pi_2(\mu) \gg\!= \lambda f. G'(f)
		~{}={}~
		\mu \gg\!= \lambda (f,f'). F(f,f')~.
		\]
	\end{enumerate}
\end{lemma}
%
%
\begin{proof}
	We only prove \Cref{lem:bind_project}.\ref{lem:bind_project1} because the reasoning for \Cref{lem:bind_project}.\ref{lem:bind_project2} is completely analogous. We have
	%
	\begin{align*}
		& \pi_1(\mu) \gg\!= \lambda f. G(f) \\
		%
		{}={}~& (\mu \gg\!= \lambda (f,f'). \delta_f) \gg\!= \lambda f. G(f) 
		\tag{definition} \\
		%
		{}={}~& \mu \gg\!= \lambda (f,f'). (\delta_f \gg\!= \lambda f. G(f))
		\tag{monad assoc.} \\
		%
		{}={}~& \mu \gg\!= \lambda (f,f'). G(f)
		\tag{monad neutral} \\
		%
		{}={}~& \mu \gg\!= \lambda (f,f'). F(f,f')~.
		\tag{assumption} 
	\end{align*}
	%
\end{proof}

\subsection{Big-Step Semantics and Correctness of Discretization}
%
We extend both of the small-step semantics canoncially to an $n$-step semantics by%
\begin{align*}
		\bigsemstep{e : \tau}{n} &{}= 
		\begin{cases}
			\delta_{e : \tau} & \text{if $n = 0$} \\
			%
			\bigsemstep{e : \tau}{n-1} \monbind  \lambda e'. \sem{e' : \tau} & \text{if $n > 0$}~.
		\end{cases} \\
		%
		%
		\bigsemstep{e : \tau, \discretize{e} : \discretize{\tau}}{n} &{}= 
		\begin{cases}
			\delta_{(e : \tau, \discretize{e} : \discretize{\tau})} & \text{if $n = 0$} \\
			%
			\bigsemstep{e : \tau, \discretize{e} : \discretize{\tau}}{n-1} \monbind  \lambda (e', e''). \sem{e' : \tau, e'' : \discretize{\tau}} & \text{if $n > 0$}~.
		\end{cases}
\end{align*}
%
%

%
%
For every type $\tau$, we denote by $Vals_\tau$ the set of values of type $\tau$ and equip it with a suitable measurable sub-space of the measurable space over $Expr_\tau$. Moreover, let $V_\tau = \{ (v:\tau, \discretize{v} : \discretize{\tau}) ~|~ v:\tau \}$.
%
\begin{definition}
	Let $e:\tau$. The \emph{(big-step) semantics} $\bigsem{e:\tau} \in \mathcal{D}(Vals_\tau + \bot)$ and $\bigsem{e:\tau} \in \mathcal{D}(V_\tau + \bot)$ are respectively defined as 
	%
	\begin{align*}
		\bigsem{e:\tau} ={}& \lambda A. \lim_{n\to\infty} \bigsemstep{e:\tau}{n}(A) \\
		%
		\bigsem{e:\tau,  \discretize{e} : \discretize{\tau}} ={}& \lambda A. \lim_{n\to\infty} \bigsemstep{e:\tau,  \discretize{e} : \discretize{\tau}}{n}(A)
	\end{align*}
	%
\end{definition}
%
%
Well-definedness of these objects follows from the monotone convergence theorem: The probability of evaluating to some set of values or an error is montonically increasing in $n$ and upper-bounded by $1$. The following will imply our desired equivalence claim:

%\kevin{the folllowing must be measurable} \\

\begin{lemma}
	\label{lem:big_step_coupling_props}
	Let $e:\tau$. We have:
	%
	\begin{enumerate}
		\item\label{lem:big_step_coupling_props1} $\pi_1(\bigsem{e:\tau, \discretize{e} : \discretize{\tau}}) = \bigsem{e:\tau}$.
		%
			\item\label{lem:big_step_coupling_props2} $\pi_2(\bigsem{e:\tau, \discretize{e} : \discretize{\tau}}) = \bigsem{\discretize{e}:\discretize{\tau}}$.
	\end{enumerate}
\end{lemma}
%
\begin{proof}
	Follows immeditaley from \Cref{lem:small_step_coupling_props}.
\end{proof}
