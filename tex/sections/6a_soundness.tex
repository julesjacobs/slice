\begin{definition}
	Let $\tau$ be a type.
 	Given a suitable measurable space, we denote by $\mathcal{D}(Expr_{\tau} + \bot)$ the set of sub-probability measures over $Expr_{\tau} + \bot$. Moreover, for all $\tau,\tau'$, we define the following monad:
 	%
 	\begin{align*}
 		 & \text{return} \colon Expr_{\tau} \to \mathcal{D}(Expr_{\tau} + \bot), \quad \text{return}(e) = \delta_e \\
 		 %
 		 & \text{bind} \colon \mathcal{D}(Expr_{\tau} + \bot) \to (Expr_{\tau} \to \mathcal{D}(Expr_{\tau'} + \bot)) \to \mathcal{D}(Expr_{\tau'} + \bot), \\
 		 & \text{bind}(\mu)(f)(y) := \sum_{x \in support(\mu)} \mu(x) \cdot f_{\bot}(x)(y) \\
 		 &\text{bind}(\mu)(f)(A) := \int_{x \in support(\mu)} f_{\bot}(x)(A) \cdot d\mu(x)~,
 	\end{align*}
 	%
 	where $f_\bot \colon Expr_{\tau} + \bot \to \mathcal{D}(Expr_{\tau'} + \bot)$ is obtained from $f$ by 
 	%
 	\[
 		f_\bot(e) =
 		\begin{cases}
 			\delta_\bot & \text{if } e =\bot \\
 			%
 			f(e) & \text{otherwise}~.
 		\end{cases}
 	\]
\end{definition}
