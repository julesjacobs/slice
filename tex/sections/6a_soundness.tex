\subsection{Preliminaries}
We consider a subset of our language, focusing only on let statements, conditionals, less than comparisons, and uniform and discrete distributions. The other constructs follow by analogous arguments.

\subsubsection{Brief recap of measure theory} 
We first give a brief overiew of basic measure theory terms.

\paragraph{Topological space} A topological space is a pair $(X, \mathcal{T})$ where $\mathcal{T}$ is a collection of subsets of $X$, called open sets, such that $\emptyset, X \in \mathcal{T}$, and $\mathcal{T}$ is closed under arbitrary unions and finite intersections.

\paragraph{$\sigma$-algebras} A $\sigma$-algebra on a set $X$ is a collection $\Sigma$ of subsets of $X$ that includes $X$ itself and is closed under complement and under countable unions. By implication, a $\sigma$-algebra includes the empty set and is closed under countable intersections. 
Given a topological space $(X, \mathcal{T})$, the Borel $\sigma$-algebra $\mathcal{B}(X)$ is the smallest $\sigma$-algebra containing all open sets in $\mathcal{T}$.

\paragraph{Measurable spaces} A measurable space is a pair $(X,\Sigma)$ consisting of a set $X$ and a $\sigma$-algebra $\Sigma$ on $X$.
When $\Sigma$ is the Borel $\sigma$-algebra on a topological space $X$, we call $(X, \Sigma)$ a Borel measurable space.

\paragraph{Measures} Let $(X,\Sigma)$ be a measurable space. A function $\mu : \Sigma \to \R^+$ is called a measure if it satisfies nonnegativity, null empty set, and countable additivity. 
When $\Sigma$ is a Borel $\sigma$-algebra, $\mu$ is called a Borel measure.

\paragraph{Measure spaces} The triple $(X, \Sigma, \mu)$ is called a measure space.
 When $\Sigma$ is the Borel $\sigma$-algebra and $\mu$ is the Borel measure, $(X, \Sigma, \mu)$ is called a Borel measure space.

\paragraph{(Sub)probability measures} When $\mu(X) \leq 1$, it is called a (Borel) subprobability measure. When $\mu(X) = 1$, it is called a (Borel) probability measure and $(X, \Sigma, \mu)$ a (Borel) probability space.

\paragraph{Product measure} The product of two measurable spaces $(X_1, \Sigma_1)$ and $(X_2, \Sigma_2)$ is also a measurable space, namely $(X_1 \times X_2, \Sigma_1 \otimes \Sigma_2)$. 

\begin{definition}[Subspace $\sigma$-algebra]\label{def:subspace-sigma-algebra}
	Let $(X, \Sigma)$ be a Borel measurable space and let $Y \subseteq X$. 
	The subspace (or trace) $\sigma$-algebra on $Y$ is defined as
	\[
	\Sigma_Y = \{ Y \cap A \mid A \in \Sigma \}.
	\]
	Then $(Y, \Sigma_Y)$ is a Borel measurable space.
\end{definition}

\begin{definition}[Pullback $\sigma$-algebra and induced isomorphism]\label{def:pullback}
	Let $f \colon X \to Y$ be a function and let $(Y, \Sigma_Y)$ be a measurable space. The pullback $\sigma$-algebra on $X$ along $f$ is defined by:

	\[
		f^{-1}(\Sigma_Y) := \{f^{-1}(A) \mid A \in \Sigma_Y\}.
	\]

	We equip $X$ with this $\sigma$-algebra and write $(X, f^{-1}(\Sigma_Y))$.

	If $f$ is a bijection, then $f$ induces an isomorphism of measurable spaces

	\[
		(X, f^{-1}(\Sigma_Y)) \cong (Y, \Sigma_Y)
	\]

	i.e.:

	\begin{itemize}
		\item $f$ is measurable, and
		\item $f^{-1}$ is measurable
	\end{itemize}
\end{definition}


\begin{definition}[Support]
	We define the support of a measure separately for discrete and continuous cases:
	\begin{enumerate}
		\item Let $(X, \Sigma, \mu)$ be a measure space where $\mu$ is a discrete measure. The support of $\mu$ is the set of all points with positive measure:
		\[
			{support}(\mu) \triangleq \{ x \in X \mid \mu(x) > 0 \}.
		\]

		\item Let $(X, \Sigma, \mu)$ be a Borel measure space where $X$ is a topological space and $\mu$ is a continuous measure. The support of $\mu$ is:
		\[
			{support}(\mu) \triangleq \{ x \in X \mid \forall U \in \Sigma, \; (U \text{ open and } x \in U) \implies \mu(U) > 0 \}.
		\]
	\end{enumerate}
\end{definition}

\begin{definition}[Markov kernels]
	Given measurable spaces $(X, \Sigma_X)$ and $(Y, \Sigma_Y)$, a Markov kernel with source $X$ and target $Y$ is a function $P : X \times \Sigma_Y \to [0,1]$ (or equivalently, $X \to \Sigma_Y \to [0,1]$) that maps each source state $x \in X$ to a distribution over target states $P(x, -) : \Sigma_Y \to [0,1]$.

	A Markov kernel $P$ must satisfy two conditions:

	\begin{itemize}
		\item \emph{Probability measure:} For each source state $x \in X$, the map $A \mapsto P(x, A)$ must be a probability measure on the target space.
		\item \emph{Measurable function:} For each event $A \in \Sigma_Y$ in the target space, the map $x \mapsto P(x,A)$ must be a measurable function.
	\end{itemize}
	A kernel $P$ is called deterministic if $P(a,-)$ is a Dirac measure for each $a$. 
\end{definition}

\begin{definition}[Projection with error]
	Let $(X + \bot, \Sigma_{X + \bot})$ and $(Y + \bot, \Sigma_{Y + \bot})$ be measurable spaces, with $(X, \Sigma_X)$ and $(Y, \Sigma_Y)$ measurable spaces as well. Let $\mu \in \mathcal{D}((X \times Y) + \bot)$. We define the first and second projection measures as:
	\begin{enumerate}
	\item $\pi_1(\mu)(A) := \mu(A \times Y)$ for every measurable $A \in \Sigma_X$, and $\pi_1(\mu)({\bot}) := \mu({\bot})$
	\item $\pi_2(\mu)(B) := \mu(X \times B)$ for every measurable $B \in \Sigma_Y$, and $\pi_2(\mu)({\bot}) := \mu({\bot})$
	\end{enumerate}
	Then, $\pi_1(\mu)$ is a measure on $(X + \bot, \Sigma_{X + \bot})$ and $\pi_2(\mu)$ is a measure on $(Y + \bot, \Sigma_{Y + \bot})$.
\end{definition}

\subsection{Measure space on expressions}
To make the probabilistic semantics precise, we define an appropriate measure space structure on the set of expressions. We construct a Borel measurable space on $Expr_{\tau}$, the set of well-typed expressions in our langauge, by factoring out real numbers using the notion of what we call \emph{expression skeletons}.

\begin{definition}[Expression Skeleton]
	An expression skeleton $s : \tau$ is a well-typed expression $e : \tau$ where each occurrence of a float constant is replaced by a hole $\square$, generated as follows:
	\begin{align*}
		s ::= \; 
		&\square \mid x \mid \true \mid \false \mid \finconst{k}{n} \mid \letkw \; x = s_1 \; \inkw \; s_2 \\
		&\mid \ifkw \; s_1 \; \thenkw \; s_2 \; \elsekw \; s_3 \mid s_1 < s_2 \mid \uniform(s_1, s_2) \mid \discrete(p_0, \ldots, p_{n-1})
	\end{align*}
	with the understanding that each hole $\square$ has type $\floattype{B}{V}$, and that all other types appearing in the original expression are preserved in its corresponding skeleton.
	
	We denote by $\text{Skel}_{\tau}$ the set of all expression skeletons, and by $\holes(s)$ the number of distinct holes in skeleton $s$.
\end{definition}

\begin{example}
	The expression $\uniform(0.0, 1.0) < 0.5$ has the skeleton $\uniform(\square, \square) < \square$ with three holes. 
\end{example}

\begin{lemma}[Skel is countable]\label{lem:skel-countable}
	The set $\text{Skel}_{\tau}$ is countable.
\end{lemma}

\begin{proof}
	Each skeleton is a finite syntactic tree where:
	\begin{itemize}
	\item Internal nodes are labeled by constructors from a finite set of ${\letkw, \ifkw, <, \uniform}$
	\item Leaves are labeled by variables (countable), literals $\true,\false,\finconst{k}{n}$ (countable), holes $\square$ (countable), or $\discrete$ (where the parameters $p_i$ are fixed).
	\end{itemize}
	The set of all finite trees over countable labels is countable.
\end{proof}

\begin{definition}[Filling a skeleton]
	Let $s : \tau \in \text{Skel}_{\tau}$ and let $\vec{r} = (r_1, \ldots, r_{\holes(s)}) \in \mathbb{R}^{\holes(s)}$. Define the expression $s[\vec{r}] : \tau \in Expr_{\tau}$ to be the result of replacing each hole $\square_i$ in $s$ with the corresponding real constant $r_i$, for $1 \le i \le \holes(s)$.
\end{definition}

\begin{definition}[Decomposing an expression]
	The decomposition function
	\[
		\decompose \colon Expr_{\tau}
		\;\longrightarrow\;
		\bigcup_{s \in Skel_{\tau}} \left( \{s\} \times \mathbb{R}^{\holes(s)} \right)
	\]
	is defined as the inverse of filling a skeleton, i.e., for every expression $e : \tau$,
	\[
		\decompose(e) = (s, \vec{r})
		\quad\text{such that}\quad
		e = s[\vec{r}].
	\]
	The explicit recursive definition of $\decompose$ is shown in Figure~\ref{fig:decompose-definition}.
\end{definition}

\begin{definition}[Hole assignments of a skeleton]
	For a given $s : \tau \in \text{Skel}_{\tau}$, define the set of hole assignments of $s$ as:
	\[
		A_{s : \tau} := 
		\bigcup_{e : \tau \in Expr_{\tau}} \left\{ \decompose(e : \tau)_2 \mid \decompose(e : \tau)_1 = s : \tau \right\}
		\subseteq \R^{\holes(s)}
	\]

	Intuitively, $A_{s;\tau}$ is the set of all real tuples that can be substituted into the holes of $s$ to yield a well-typed expression of type $\tau$.
\end{definition}

\begin{example}
	Consider the skeleton
	\[
		s : \tau 
		=
		(\square : \floattype{\clt{3.5}, \clt{4.5}}{\top}
		\;<\;
		\square : \floattype{\clt{3.5}, \clt{4.5}}{3.5,4.5})
		: \bool.
	\]
	
	This skeleton has two holes:
	\begin{itemize}
		\item The left hole has type 
		$\floattype{\clt{3.5}, \clt{4.5}}{\top}$,
		so it may be instantiated by any real number.
		\item The right hole has type 
		$\floattype{\clt{3.5}, \clt{4.5}}{3.5,4.5}$,
		so it may only be instantiated by one of the values 
		$3.5$ or $4.5$.
	\end{itemize}
	
	An expression $e : \bool$ has skeleton $s$ iff it is of the form
	\[
		e = c_1 < c_2
	\]
	where $c_1 \in \mathbb{R}$ and 
	$c_2 \in \{3.5,4.5\}$.
	
	Therefore, the admissible parameter tuples are precisely
	\[
		A_{s:\tau}
		=
		\{ (c_1,c_2) \mid c_1 \in \mathbb{R},\; c_2 \in \{3.5,4.5\} \}.
	\]
	
	Equivalently,
	\[
		A_{s:\tau}
		=
		\mathbb{R} \times \{3.5,4.5\}
		\subseteq
		\mathbb{R}^2.
	\]
	
	Thus, expressions with this syntactic shape are parameterized by a two-dimensional space whose first coordinate ranges over $\mathbb{R}$ and whose second coordinate ranges over a finite set.
\end{example}

\begin{definition}[$\sigma$-algebra construction on $A_{s;\tau}$]\label{def:sigma-algebra-construction-holes}
	Let $s : \tau \in \text{Skel}_{\tau}$. Endow $A_{s;\tau}$ with the subspace $\sigma$-algebra inherited from the standard Borel $\sigma$-algebra on $\R^{\holes(s)}$:

	\[
		\mathcal{F}_{s;\tau} := \left\{A_{s;\tau} \cap M \mid M \in \mathcal{B}(\R^{\holes(s)})\right\}.
	\]
\end{definition}

\begin{lemma}\label{lem:measurable-a-st}
	$(A_{s;\tau}, \mathcal{F}_{s;\tau})$ is a Borel measurable space.
\end{lemma}

\begin{proof}
	For a fixed $s : \tau \in \text{Skel}_{\tau}$, $(\R^{\holes(s)}, \mathcal{B}(\R^{\holes(s)}))$ is a Borel measurable space and $A_{s; \tau} \subseteq \R^{\holes(s)}$. Thus, by Definition~\ref{def:subspace-sigma-algebra}, $(A_{s;\tau}, \{A_{s;\tau} \cap M \mid M \in \mathcal{B}(\R^{\holes(s)})\}) = (A_{s;\tau}, \mathcal{F}_{s;\tau})$ is a Borel measurable space.
\end{proof}


Now, we map this measurable space to the space of actual expressions that share that skeleton. 

\begin{definition}[Expressions of a skeleton]
	For a given $s : \tau \in \text{Skel}_{\tau}$, define the set of well-typed expressions having skeleton $s$ as:
	\[
		Expr_{s;\tau}
		:= 
		\left\{\, e : \tau \in Expr_{\tau} \mid \decompose(e : \tau)_2 \in A_{s;\tau} \,\right\}
		\subseteq
		Expr_\tau.
	\]
\end{definition}

Intuitively, $Expr_{s;\tau}$ consists exactly of those expressions whose syntactic shape is fixed by $s$, and whose real constants vary according to the admissible hole assignments in $A_{s;\tau}$.

\begin{example}
	Consider the skeleton
	\[
		s : \tau 
		=
		(\square : \floattype{\clt{3.5}, \clt{4.5}}{\top}
		\;<\;
		\square : \floattype{\clt{3.5}, \clt{4.5}}{3.5,4.5})
		: \bool.
	\]
	
	An expression $e : \bool$ belongs to $Expr_{s;\tau}$ iff it is of the form
	\[
		e = c_1 < c_2
	\]
	where $c_1 \in \mathbb{R}$ and $c_2 \in \{3.5,4.5\}$.
	
	Therefore,
	\[
		Expr_{s;\tau}
		=
		\left\{
			c_1 < c_2
			\;\middle|\;
			c_1 \in \mathbb{R},\; c_2 \in \{3.5,4.5\}
		\right\}.
	\]
\end{example}

\begin{definition}[$\sigma$-algebra construction on $Expr_{s;\tau}$]\label{def:sigma-algebra-construction-skeletons}
	Let $s : \tau \in \text{Skel}_{\tau}$. Endow $Expr_{s;\tau}$ with the following $\sigma$-algebra:

	\[
		\mathcal{F}_{Expr_{s;\tau}}
		:=
		\left\{
			\{e : \tau \in Expr_{s;\tau} \mid \decompose(e : \tau)_2 \in B\} \mid B \in 
			\mathcal{F}_{s;\tau}
		\right\}.
	\]
\end{definition}


\begin{lemma}\label{lem:decompose-restriction-bijection}
	For a fixed $s : \tau \in \text{Skel}_{\tau}$, the map 
	\[
		\decompose_{s;\tau} \colon Expr_{s;\tau} \to A_{s;\tau},
		\qquad
		e \mapsto \decompose(e)_2.
	\]
	is a bijection.
\end{lemma}

\begin{proof}
	Let $s : \tau \in \text{Skel}_{\tau}$. 

	\paragraph{(i) Injectivity:}

	Suppose $e_1, e_2 \in Expr_{s;\tau}$ satisfy $\decompose_{s;\tau}(e_1) = \decompose_{s;\tau}(e_2)$. By definition of $\decompose_{s;\tau}$, $\decompose_{s;\tau}(e_1) = \decompose_{s;\tau}(e_2) = \vec{r}$ for some $\vec{r} \in A_{s;\tau}$. Since both $e_1$ and $e_2$ have skeleton $s$ by assumption, we have $e_1 = s[\vec{r}]$ and $e_2 = s[\vec{r}]$, hence $e_1 = e_2$.

	\paragraph{(ii) Surjectivity:}

	Let $\vec{r} \in A_{s;\tau}$. By definition of $A_{s;\tau}$, there exists some expression $e \in Expr_{\tau}$ such that $\decompose(e) = (s,\vec{r})$. Then $e \in Expr_{s;\tau}$ by construction of $Expr_{s;\tau}$, and $\decompose_{s;\tau}(e) = \vec{r}$.

	\medskip

	$\decompose_{s;\tau}$ is injective and surjective, hence a bijection.
\end{proof}

\begin{lemma}[Isomorphism property]\label{lem:isomorphism}
	For $s \in \text{Skel}_{\tau}$,
	\[
		(Expr_{s;\tau}, \mathcal{F}_{Expr_{s;\tau}})
		\cong
		(A_{s;\tau}, \mathcal{F}_{s;\tau}).
	\]
\end{lemma}

\begin{proof}
	By Lemma~\ref{lem:decompose-restriction-bijection}, the restricted decomposition map $\decompose_{s;\tau} \colon Expr_{s;\tau} \to A_{s;\tau}$ is a bijection.  We know $(A_{s;\tau}, \mathcal{F}_{s;\tau})$ is a Borel measurable space by Lemma~\ref{lem:measurable-a-st}. By Definition~\ref{def:pullback}, since $\decompose_{s;\tau}$ is a bijection, it induces an isomorphism $(Expr_{s;\tau}, \mathcal{F}_{Expr_{s;\tau}}) \cong (A_{s;\tau}, \mathcal{F}_{s;\tau})$.
\end{proof}

\begin{lemma}\label{lem:measurable-expr-st}
	$(Expr_{s;\tau}, \mathcal{F}_{Expr_{s;\tau}})$ is a Borel measurable space.
\end{lemma}

\begin{proof}
	By Lemma~\ref{lem:isomorphism}.
\end{proof}


Now we combine the measurable spaces for each distinct skeleton into a single measurable space of all well-typed expressions of type $\tau$. The set of all well-typed expressions of type $\tau$ is thus the disjoint union:
\[
	Expr_{\tau} = \bigsqcup_{s : \tau \in \text{Skel}_{\tau}} Expr_{s;\tau}.
\]
This union is disjoint because every typed expression $e : \tau$ has a unique skeleton, namely $\decompose(e)_1$, so $e$ belongs to exactly one set $Expr_{s;\tau}$.

\begin{definition}[$\sigma$-algebra construction on $Expr_{\tau}$]\label{def:sigma-algebra-construction-expressions}
	Endow $Expr_\tau$ with the following $\sigma$-algebra: 
	\[
		\mathcal{F}_{Expr_{\tau}}
		:= \{A \subseteq Expr_{\tau} \mid \forall s \in \text{Skel}_{\tau} .\; A \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}}\}
	\]
\end{definition}

\begin{lemma}\label{lem:measurable-expr-t}
	$(Expr_{\tau}, \mathcal{F}_{Expr_{\tau}})$ is a Borel measurable space.
\end{lemma}

\begin{proof}
	We first verify that $\mathcal{F}_{Expr_{\tau}}$ is indeed a $\sigma$-algebra on $Expr_{\tau}$.

	\paragraph{(i) Contains the empty set:}
	
	We show $\emptyset \in \mathcal{F}_{Expr_\tau}$. By Definition~\ref{def:sigma-algebra-construction-expressions}, it suffices to show that $\forall s \in \text{Skel}_\tau$, $\emptyset \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}}$. But $\emptyset \cap Expr_{s;\tau} = \emptyset$, and since $\mathcal{F}_{Expr_{s;\tau}}$ is a $\sigma$-algebra, $\emptyset \in \mathcal{F}_{Expr_{s;\tau}}$. Therefore, $\emptyset \in \mathcal{F}_{Expr_\tau}$.

	\paragraph{(ii) Closure under complements:}

	Let $A \in \mathcal{F}_{Expr_\tau}$. We show that $Expr_\tau \setminus A \in \mathcal{F}_{Expr_\tau}$. By Definition~\ref{def:sigma-algebra-construction-expressions}, it suffices to show that for all $s \in \text{Skel}_\tau$,
	\[
		(Expr_\tau \setminus A) \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}}.
	\]
	Fix $s \in \text{Skel}_\tau$.  Observe that:

	\begin{align*}
		&(Expr_\tau \setminus A) \cap Expr_{s;\tau} \\
		=\; &(Expr_\tau \cap Expr_{s;\tau}) \setminus (A \cap Expr_{s;\tau}) \tag{$\cap$ distributes over $\setminus$} \\
		=\; &Expr_{s;\tau} \setminus (A \cap Expr_{s;\tau}). \tag{since $Expr_{s;\tau} \subseteq Expr_\tau$, so $Expr_\tau \cap Expr_{s;\tau} = Expr_{s;\tau}$}
	\end{align*}

	Since $A \in \mathcal{F}_{Expr_\tau}$, by Definition~\ref{def:sigma-algebra-construction-expressions}, we have $A \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}}$, and because $\mathcal{F}_{Expr_{s;\tau}}$ is a $\sigma$-algebra, it is closed under complements relative to $Expr_{s;\tau}$. Hence
	\[
		Expr_{s;\tau} \setminus (A \cap Expr_{s;\tau}) \in \mathcal{F}_{Expr_{s;\tau}}.
	\]
	Therefore $(Expr_\tau \setminus A) \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}}$ for all $s \in \text{Skel}_{\tau}$. Therefore, $Expr_\tau \setminus A \in \mathcal{F}_{Expr_\tau}$.

	\paragraph{(iii) Closure under countable unions:}

	Let $(A_n)_{n \in \Nats}$ be a sequence with $A_n \in \mathcal{F}_{Expr_\tau}$ for all $n$. We show that $\bigcup_{n \in \Nats} A_n \in \mathcal{F}_{Expr_\tau}$. By Definition~\ref{def:sigma-algebra-construction-expressions}, it suffices to show that for all $s \in \text{Skel}_\tau$,
	\[
		\left(\bigcup_{n \in \Nats} A_n\right) \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}}.
	\]
	Fix $s \in \text{Skel}_\tau$. Using distributivity of $\cap$ over $\cup$, we have
	\[
		\left(\bigcup_{n \in \Nats} A_n\right) \cap Expr_{s;\tau}
		=
		\bigcup_{n \in \Nats} \left(A_n \cap Expr_{s;\tau}\right).
	\]
	Since each $A_n \in \mathcal{F}_{Expr_\tau}$, by Definition~\ref{def:sigma-algebra-construction-expressions}, we have $A_n \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}}$ for all $n$. Because $\mathcal{F}_{Expr_{s;\tau}}$ is a $\sigma$-algebra, it is closed under countable unions, hence
	\[
		\bigcup_{n \in \Nats} \left(A_n \cap Expr_{s;\tau}\right) \in \mathcal{F}_{Expr_{s;\tau}}.
	\]
	Therefore $\left(\bigcup_{n \in \Nats} A_n\right) \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}}$ for all $s$, and thus $\bigcup_{n \in \Nats} A_n \in \mathcal{F}_{Expr_\tau}$.

	\medskip

	It remains to show that $\mathcal{F}_{Expr_{\tau}}$ is a Borel $\sigma$-algebra, i.e. that it equals the Borel $\sigma$-algebra of some topology on $Expr_\tau$.

	\paragraph{Topology on $Expr_\tau$.}
	Via the bijection $\decompose_{s;\tau}$, equip each $Expr_{s;\tau}$ with the topology on $A_{s;\tau} \subseteq \mathbb{R}^{\holes(s)}$. Then equip $Expr_\tau = \bigsqcup_{s \in \mathrm{Skel}_\tau} Expr_{s;\tau}$ with the disjoint union topology $\mathcal{T}$: a set $U \subseteq Expr_\tau$ is open if and only if $U \cap Expr_{s;\tau}$ is open in $Expr_{s;\tau}$ for every $s \in \mathrm{Skel}_\tau$.

	\paragraph{Claim.} $\mathcal{F}_{Expr_\tau} = \mathcal{B}(Expr_\tau, \mathcal{T})$.

	\paragraph{($\supseteq$)}
	Let $U \in \mathcal{T}$ be open. By definition of $\mathcal{T}$, $U \cap Expr_{s;\tau}$ is open in $Expr_{s;\tau}$ for every $s$, and hence Borel in $Expr_{s;\tau}$, so $U \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}}$ for all $s \in \mathrm{Skel}_\tau$. By Definition~\ref{def:sigma-algebra-construction-expressions}, $U \in \mathcal{F}_{Expr_\tau}$. Since $\mathcal{F}_{Expr_\tau}$ is a $\sigma$-algebra containing all open sets of $\mathcal{T}$, $\mathcal{B}(Expr_\tau, \mathcal{T}) \subseteq \mathcal{F}_{Expr_{\tau}}$.

	\paragraph{($\subseteq$)}
	Let $A \in \mathcal{F}_{Expr_\tau}$. By Definition~\ref{def:sigma-algebra-construction-expressions}, for each $s \in \mathrm{Skel}_\tau$ we have $A \cap Expr_{s;\tau} \in \mathcal{F}_{Expr_{s;\tau}} = \mathcal{B}(Expr_{s;\tau})$. Since the open sets of $Expr_{s;\tau}$ are also open in $\mathcal{T}$, we have $\mathcal{B}(Expr_{s;\tau}) \subseteq \mathcal{B}(Expr_\tau, \mathcal{T})$, and thus $A \cap Expr_{s;\tau} \in \mathcal{B}(Expr_\tau, \mathcal{T})$ for each $s$. Since $\mathrm{Skel}_\tau$ is countable by Lemma~\ref{lem:skel-countable}, we have
	\[
		A = \bigsqcup_{s \in \mathrm{Skel}_\tau} \left(A \cap Expr_{s;\tau}\right),
	\]
	a countable disjoint union of sets in $\mathcal{B}(Expr_\tau, \mathcal{T})$. Hence $A \in \mathcal{B}(Expr_\tau, \mathcal{T})$.

	\medskip
	Therefore $\mathcal{F}_{Expr_\tau} = \mathcal{B}(Expr_\tau, \mathcal{T})$, and $(Expr_\tau, \mathcal{F}_{Expr_\tau})$ is a Borel measurable space.

\end{proof}

\begin{lemma} 
	Let $(Expr_{\tau} + \bot, \mathcal{F}_{Expr_{\tau} + \bot})$ be the measurable space of expressions with error from Definition~\ref{def:meas-space-with-error}. Let $\{\bot\}$ be the singleton set containing the error element $\bot$. The following are true:
	\begin{enumerate}
		\item The set $\vals$ is measurable.
		\item The set $\{\bot\}$ is measurable.
		\item The set $\nonvals$ is measurable.
	\end{enumerate}
\end{lemma}

\begin{proof}
	We prove the measurability of $\vals$, $\{\bot\}$, and $N$ in separate parts.
	\begin{enumerate}
		\item \textbf{(Measurability of $\vals$):} By the construction of $\mathcal{F}_{Expr_{\tau}}$, for each skeleton $s$ that represents a value, the set $\{\vec{r} \in \R^{\holes(s)} \mid \vdash s[\vec{r}] : \tau\}$ is a measurable set in the union. The set $V$ is the countable union of all such sets corresponding to value skeletons. Since $\text{Skel}$ is countable, $V$ is a countable union of measurable sets and is therefore itself measurable, e.g. $V \in \mathcal{F}_{Expr_{\tau}}$. As $\mathcal{F}_{Expr_{\tau}} \subset \mathcal{F}_{Expr_{\tau} + \bot}$, it follows that $V \in \mathcal{F}_{Expr_{\tau} + \bot}$.
		
		\item \textbf{(Measurability of $\{\bot\}$):} By Definition~\ref{def:meas-space-with-error}, the topology $\mathcal{T}_{Expr_{\tau} + \bot}$ contains all sets of the form $U \cup \{\bot\}$, where $U \in \mathcal{T}_{Expr_{\tau}}$. In particular, taking $U = \emptyset$ gives $\emptyset \cup \{\bot\} = \{\bot\} \in \mathcal{T}_{Expr_{\tau} + \bot}$. Since the Borel $\sigma$-algebra $\mathcal{F}_{Expr_{\tau} + \bot}$ is generated by $\mathcal{T}_{Expr_{\tau} + \bot}$, it follows that $\{\bot\} \in \mathcal{F}_{Expr_{\tau} + \bot}$.

		\item \textbf{(Measurablity of $N$):} By definition, $N$ is the set of non-value expressions, e.g. $N = Expr_{\tau} \setminus V$. From part i), we have established that $V \in \mathcal{F}_{Expr_{\tau}}$. Since $\mathcal{F}_{Expr_{\tau}}$ is a $\sigma$-algebra and is closed under complement, the complement of $V$ within $Expr_{\tau}$, which is precisely $N$, is also measurable, e.g. $N \in \mathcal{F}_{Expr_{\tau}}$. As $\mathcal{F}_{Expr_{\tau}} \subset \mathcal{F}_{Expr_{\tau} + \bot}$, it follows that $N \in \mathcal{F}_{Expr_{\tau} + \bot}$.
	\end{enumerate}
\end{proof}

\begin{lemma}[Dirac return is a Markov kernel]\label{lem:dirac-return-kernel}
	For any measurable space $X$, the map $return: X \to \mathcal{D}(X + \bot)$, $x \mapsto \dirac{x}$ is a Markov kernel.
\end{lemma}

\begin{proof}
	For any measurable set $A \subseteq X + \bot$, the map: 
	\[
		x \mapsto \dirac{x}(A) = \mathds{1}_A(x)
	\]
	is precisely the indicator function of $A$. Since $A$ is measurable, $\mathbf{1}_A$ is a measurable function $X \to [0,1]$. Moreover, each $\dirac{x}$ is a Dirac measure, hence a probability measure on $X + \bot$ concentrated at $x$. Thus $x \mapsto \dirac{x}$ satisfies the definition of a Markov kernel.
\end{proof}

\begin{lemma}[$f_{\bot}$ is a Markov kernel]\label{lem:bot-extension-kernel}
	Given a Markov kernel $f : X \to \mathcal{D}(Y + \bot)$, the extension $f_{\bot} : (X + \bot) \to \mathcal{D}(Y + \bot)$ is also a Markov kernel.
\end{lemma}
	
\begin{proof}
	We verify the two conditions for $f_{\bot}$ to be a Markov kernel.
	
	\paragraph{(1) Probability measure:}
	Fix $e \in X + \bot$. We must show that the map 
	\[
	A \mapsto f_{\bot}(e)(A)
	\]
	is a probability measure on $Y + \bot$.
	
	\begin{itemize}
	\item If $e = \bot$, then $f_{\bot}(\bot) = \delta_{\bot}$, the Dirac measure at $\bot$, which is a probability measure on $Y + \bot$.
	
	\item If $e \in X$, then $f_{\bot}(e) = f(e)$, and since $f$ is a Markov kernel, $f(e)$ is a probability measure on $Y + \bot$.
	\end{itemize}
	
	Thus, for all $e \in X + \bot$, $f_{\bot}(e)$ is a probability measure.
	
	\paragraph{(2) Measurable function:}
	Let $A \subseteq Y + \bot$ be measurable. We must show that the function
	\[
	e \mapsto f_{\bot}(e)(A)
	\]
	is measurable on $X + \bot$.
	
	We consider two cases:
	
	\begin{itemize}
	\item If $e = \bot$, then
	\[
	f_{\bot}(\bot)(A) = \delta_{\bot}(A) =
	\begin{cases}
	1 & \text{if } \bot \in A, \\
	0 & \text{if } \bot \notin A.
	\end{cases}
	\]
	Thus, the function is constant at $\bot$, hence measurable.
	
	\item If $e \in X$, then
	\[
	f_{\bot}(e)(A) = f(e)(A),
	\]
	and the map $x \mapsto f(x)(A)$ is measurable by assumption since $f$ is a Markov kernel.
	\end{itemize}
	
	Therefore, the function $e \mapsto f_{\bot}(e)(A)$ is measurable on $X + \bot$.
	
	\medskip
	
	Since both conditions hold, $f_{\bot}$ is a Markov kernel.
\end{proof}

\begin{lemma}[Composition of Markov kernels~\cite{kallenberg1997foundations}, Lemma~1.38]\label{lem:kernel-composition}
	Let $(X, \Sigma_X)$, $(Y, \Sigma_Y)$, and $(Z, \Sigma_Z)$ be measurable spaces.
	If 
	\[
		P : X \to \mathcal{D}(Y)
		\quad\text{and}\quad
		Q : Y \to \mathcal{D}(Z)
	\]
	are Markov kernels, then their composition
	\[
		(P \monbind Q)(x, A)
		\;:=\;
		\int_Y P(x, dy)\, Q(y, A)
	\]
	defines a Markov kernel
	\[
		P \monbind Q : X \to \mathcal{D}(Z).
	\]
\end{lemma}

\begin{lemma}[Measurability of the constant extension map]\label{lem:constant-extension-measurable}
	Let $A \subseteq \R^n$ be equipped with the subspace $\sigma$-algebra 
	$\mathcal{F}_A := \{A \cap M \mid M \in \mathcal{B}(\R^n)\}$, and let 
	$B \subseteq \R^{n+m}$ be equipped with the subspace $\sigma$-algebra 
	$\mathcal{F}_B := \{B \cap N \mid N \in \mathcal{B}(\R^{n+m})\}$, where $n,m \in \Nats$.
	
	Fix $\vec{c} \in \R^m$ and define
	\[
		\phi_{\vec{c}} \colon A \to B,
		\qquad
		\phi_{\vec{c}}(\vec{r}) := (\vec{r},\vec{c}).
	\]
	Then $\phi_{\vec{c}}$ is measurable.
	\end{lemma}
	
	\begin{proof}
	Let $S \in \mathcal{F}_B$. By definition of the subspace $\sigma$-algebra, there exists
	$N \in \mathcal{B}(\R^{n+m})$ such that $S = B \cap N$.
	We show that $\phi_{\vec{c}}^{-1}(S) \in \mathcal{F}_A$.
	
	Define the (total) constant-extension map
	\[
		g_{\vec{c}} \colon \R^n \to \R^{n+m},
		\qquad
		g_{\vec{c}}(\vec{r}) := (\vec{r},\vec{c}).
	\]
	The map $g_{\vec{c}}$ is continuous since each output coordinate is either the identity (continuous) or a constant (continuous), hence Borel measurable. Therefore:
	\[
	M' := g_{\vec{c}}^{-1}(N) = \{\vec{r} \in \R^n \mid (\vec{r},\vec{c}) \in N\}
	\in \mathcal{B}(\R^n).
	\]
	
	Now compute the preimage:
	\begin{align*}
	\phi_{\vec{c}}^{-1}(S)
	&= \{\vec{r} \in A \mid \phi_{\vec{c}}(\vec{r}) \in B \cap N\} \\
	%
	&= \{\vec{r} \in A \mid \phi_{\vec{c}}(\vec{r}) \in B \cap \phi_{\vec{c}}(\vec{r}) \in N\} \\
	%
	&= \{\vec{r} \in A \mid (\vec{r},\vec{c}) \in N\} 
	\qquad (\text{since } \phi_{\vec c}(\vec r)\in B \text{ for all } \vec r\in A)\\
	%
	&= A \cap \{\vec{r} \in \R^n \mid (\vec{r},\vec{c}) \in N\} \\
	%
	&= A \cap M'.
	\end{align*}
	Since $M' \in \mathcal{B}(\R^n)$, we have $A \cap M' \in \mathcal{F}_A$ by definition. Hence
	$\phi_{\vec{c}}^{-1}(S) \in \mathcal{F}_A$, so $\phi_{\vec{c}}$ is measurable.
\end{proof}

\begin{lemma}[Diverge is a Markov kernel]\label{lem:diverge-kernel}
	For each type $\tau$, define
	\[
		K_{\diverge} \colon Expr_{\tau} + \bot \to \mathcal{D}(Expr_{\tau} + \bot),
		\qquad
		K_{\diverge}(e) := \lambda A.\,0.
	\]
	Then $K_{\diverge}$ is a Markov kernel (into subprobability measures).
\end{lemma}

\begin{proof}
	We verify the two kernel conditions.

	\paragraph{(1) (Sub)probability measure in the codomain.}
	Fix any $e \in Expr_{\tau} + \bot$. Then $K_{\diverge}(e)$ is the zero measure:
	\[
		K_{\diverge}(e)(A)=0
		\quad\text{for all measurable }A \subseteq Expr_{\tau}+\bot.
	\]
	It satisfies the measure laws:
	\[
		K_{\diverge}(e)(\emptyset)=0,
		\qquad
		K_{\diverge}(e)\!\left(\bigcup_{i} A_i\right)=0=\sum_i 0=\sum_i K_{\diverge}(e)(A_i)
	\]
	for every pairwise disjoint family $(A_i)_i$. Also
	\[
		K_{\diverge}(e)(Expr_{\tau}+\bot)=0\le 1.
	\]
	Hence $K_{\diverge}(e)$ is a subprobability measure, i.e.\ an element of $\mathcal{D}(Expr_{\tau}+\bot)$.

	\paragraph{(2) Measurability in the source argument.}
	Let $A \subseteq Expr_{\tau}+\bot$ be measurable. The map
	\[
		e \mapsto K_{\diverge}(e)(A)
	\]
	is the constant-zero function on $Expr_{\tau}+\bot$, therefore measurable.

	Thus both conditions hold, so $K_{\diverge}$ is a Markov kernel.
\end{proof}

\begin{lemma}[Monadic bind is a Markov kernel]\label{lem:bind-kernel}
	For every occurrence of a monadic bind $\mu \monbind f$ given in the small-step operational semantics in~\Cref{tab:smallstep-1,tab:smallstep-2}, $f : (Expr_{\tau_1} + \bot) \to \mathcal{D}(Expr_{\tau_2} + \bot)$ is a Markov kernel.
\end{lemma}

\begin{proof}
	We first prove that $f : Expr_{\tau_1} \to \mathcal{D}(Expr_{\tau_2} + \bot)$ is a Markov kernel. We show the proof for the let statement case; all other constructs are analogous.

	This means proving that $\lambda x.\, \dirac{\letkw\; y = x\; \inkw\; e_2}$ is a Markov kernel. Formally, define
	\[
		f \colon Expr_{\tau_1} \to \mathcal{D}(Expr_{\tau_2} + \bot),
		\qquad
		f(x) = \dirac{\letkw\; y = x\; \inkw\; e_2 : \tau_2}.
	\]
	We verify the two conditions for $f$ to be a Markov kernel.

	\paragraph{(i) Probability measure:} We show for each $x \in Expr_{\tau_1}$, the map $A \mapsto f(x)(A)$ is a probability measure.

	\medskip

	Fix $x \in Expr_{\tau_1}$. The map $A \mapsto f(x)(A)$ equals $A \mapsto \dirac{\letkw\; y = x\; \inkw\; e_2}(A)$, which satisfies:
	\[
		f(x)(A) =
		\begin{cases}
			1 & \text{if } \letkw\; y = x\; \inkw\; e_2 \in A, \\
			0 & \text{otherwise.}
		\end{cases}
	\]
	A Dirac measure is a probability measure, so this condition holds.

	\paragraph{(ii) Measurable function:} 

	We show for each $A \in \mathcal{F}_{Expr_{\tau_2} + \bot}$, the map $x \mapsto f(x)(A)$ is a measurable function.

	\medskip

	Fix a measurable set $A \in \mathcal{F}_{Expr_{\tau_2} + \bot}$. We must show that the map
	\[
		g \colon Expr_{\tau_1} \to \{0,1\},
		\qquad
		g(x) := f(x)(A) = \dirac{\letkw\; y = x\; \inkw\; e_2}(A)
	\]
	is measurable. Since $g$ takes values only in $\{0,1\} \subset \R$, and the Borel $\sigma$-algebra on $\R$ is generated by open intervals $(a, \infty)$, it suffices to check that $g^{-1}((a,\infty))$ is measurable for all $a \in \R$:
	\begin{itemize}
		\item If $a \geq 1$: $g^{-1}((a,\infty)) = \emptyset \in \mathcal{F}_{Expr_{\tau_1}}$.
		\item If $a < 0$: $g^{-1}((a,\infty)) = Expr_{\tau_1} \in \mathcal{F}_{Expr_{\tau_1}}$.
		\item If $0 \leq a < 1$: $g^{-1}((a,\infty)) = \{x \in Expr_{\tau_1} \mid g(x) = 1\}$.
	\end{itemize}
	The only nontrivial case is the last. Define
	\[
		B := \{x \in Expr_{\tau_1} \mid \letkw\; y = x\; \inkw\; e_2 \in A\}.
	\]
	We must show $B \in \mathcal{F}_{Expr_{\tau_1}}$. 

	\begin{align*}
		&B \in \mathcal{F}_{Expr_{\tau_1}} \\
		\iff \quad&B \in \{E \subseteq Expr_{\tau_1} \mid \forall s \in \text{Skel}_{\tau_1},\; E \cap Expr_{s;\tau_1} \in \mathcal{F}_{Expr_{s;\tau_1}}\} \tag{Def.~\ref{def:sigma-algebra-construction-expressions}} \\
		\iff \quad&\forall s \in \text{Skel}_{\tau_1},\; B \cap Expr_{s;\tau_1} \in \mathcal{F}_{Expr_{s;\tau_1}}. \tag{set comprehension}
	\end{align*}
	Fix $s \in \text{Skel}_{\tau_1}$. We prove $B \cap Expr_{s;\tau_1} \in \mathcal{F}_{Expr_{s;\tau_1}}$:
	\begin{align*}
		&B \cap Expr_{s;\tau_1} \in \mathcal{F}_{Expr_{s;\tau_1}} \\
		\iff \quad&\{x \in Expr_{\tau_1} \mid \letkw\; y = x\; \inkw\; e_2 \in A\} \cap Expr_{s;\tau_1} \in \mathcal{F}_{Expr_{s;\tau_1}} \tag{def.\ of $B$}\\
		\iff \quad&\{s[\vec{r}] \mid \vec{r} \in A_{s;\tau_1},\; \letkw\; y = s[\vec{r}]\; \inkw\; e_2 \in A\} \in \mathcal{F}_{Expr_{s;\tau_1}} \tag{$Expr_{s;\tau_1} = \{s[\vec{r}] \mid \vec{r} \in A_{s;\tau_1}\}$}\\
		\iff \quad&\{\vec{r} \in A_{s;\tau_1} \mid \letkw\; y = s[\vec{r}]\; \inkw\; e_2 \in A\} \in \mathcal{F}_{s;\tau_1} \tag{Lem.~\ref{lem:isomorphism}}
	\end{align*}
	

	\medskip
	Let $s_2$ be the skeleton of $e_2$, with $m := \holes(s_2)$, and let $\vec{c} \in A_{s_2;\tau_2} \subseteq \R^m$ be the unique tuple such that $e_2 = s_2[\vec{c}]$. Define the combined skeleton
	\[
		s' := \letkw\; y = s\; \inkw\; s_2 \in \text{Skel}_{\tau_2},
	\]
	whose holes consist of those from $s$ (filled by $\vec{r} \in \R^n$) followed by those from $s_2$ (filled by $\vec{c} \in \R^m$), so that $A_{s';\tau_2} \subseteq \R^{n+m}$. For every $\vec{r} \in A_{s;\tau_1}$:
	\[
		\letkw\; y = s[\vec{r}]\; \inkw\; e_2 = s'[(\vec{r}, \vec{c})].
	\]

	\medskip
	Define the measurable set of hole-assignments corresponding to $A$ at skeleton $s'$:
	\[
		A_{s'} := \{(\vec{r}, \vec{r}\,') \in A_{s';\tau_2} \mid s'[\vec{r}, \vec{r}\,'] \in A\} \in \mathcal{F}_{s';\tau_2}.
	\]
	This set is measurable because $A \in \mathcal{F}_{Expr_{\tau_2} + \bot}$ and the isomorphism of Lemma~\ref{lem:isomorphism} identifies measurable subsets of $Expr_{s';\tau_2}$ with measurable subsets of $A_{s';\tau_2}$.

	\medskip
	Define the constant-extension map
	\[
		\phi_{\vec{c}} \colon A_{s;\tau_1} \to A_{s';\tau_2},
		\qquad
		\phi_{\vec{c}}(\vec{r}) := (\vec{r}, \vec{c}).
	\]
	By Lemma~\ref{lem:constant-extension-measurable}, $\phi_{\vec{c}}$ is measurable. Then:
	\begin{align*}
		\{\vec{r} \in A_{s;\tau_1} \mid \letkw\; y = s[\vec{r}]\; \inkw\; e_2 \in A\}
		&= \{\vec{r} \in A_{s;\tau_1} \mid s'[\phi_{\vec{c}}(\vec{r})] \in A\} \\
		&= \{\vec{r} \in A_{s;\tau_1} \mid \phi_{\vec{c}}(\vec{r}) \in A_{s'}\} \\
		&= \phi_{\vec{c}}^{-1}(A_{s'}).
	\end{align*}
	Since $A_{s'} \in \mathcal{F}_{s';\tau_2}$ and $\phi_{\vec{c}}$ is measurable, we conclude
	\[
		\phi_{\vec{c}}^{-1}(A_{s'}) \in \mathcal{F}_{s;\tau_1},
	\]
	and therefore $B \cap Expr_{s;\tau_1} \in \mathcal{F}_{Expr_{s;\tau_1}}$ for all $s \in \text{Skel}_{\tau_1}$, giving $B \in \mathcal{F}_{Expr_{\tau_1}}$ as required.

\end{proof}

\begin{lemma}[Small-step semantics is a Markov kernel]\label{lem:small-step-kernel}
	For each type $\tau$, the small-step operational semantics $\sem{e : \tau} \colon Expr_{\tau} \to \mathcal{D}(Expr_{\tau} + \bot)$ defines a Markov kernel.
\end{lemma}

\begin{proof}
	By induction on the typing derivation of $e : \tau$. We show in each case that the map $e \mapsto \sem{e : \tau}$ is a Markov kernel.

	Consider the let case. 

	\begin{itemize}
		\item If $e_1 = v$, By~\Cref{lem:dirac-return-kernel} and~\Cref{lem:subst-measurable}, $e \mapsto \dirac{e_2[v/y] : \tau_2}$ is a Markov kernel.
		\item If $e_1 \neq v$, By the I.H., $\sem{e_1 : \tau_1} : Expr_{\tau_1} \to \mathcal{D}(Expr_{\tau_1} + \bot)$ is a Markov kernel. By~\Cref{lem:bind-kernel}, $\lambda g. \dirac{\letkw \; x = g \; \inkw \; e_2 : \tau_2 : \tau_2} \colon (Expr_{\tau_1} + \bot) \to \mathcal{D}(Expr_{\tau_2} + \bot)$ is a Markov kernel. Thus, by~\Cref{lem:kernel-composition}, $\sem{\letkw \; x = e_1 \; \inkw \; e_2 : \tau_2 : \tau_2}$ is a Markov kernel. 
	\end{itemize}

	All other cases are analogous.

\end{proof}

\begin{lemma}
	\label{lem:bind_project}
	Let $F\colon P_\tau \to \mathcal{D}(P_\tau + \bot)$, let $G\colon Expr_\tau \to \mathcal{D}(Expr_\tau + \bot)$, and let $G' \colon Expr_\discretize{\tau} \to \mathcal{D}(Expr_\discretize{\tau} + \bot)$. We have:
	%
	\begin{enumerate}
		\item\label{lem:bind_project1} If  $\forall (f,f') \in P_\tau\colon F(f,f') = G(f)$, then for all $\mu \in \mathcal{D}(P_\tau + \bot)$,
		%
		\[
		\pi_1(\mu) \gg\!= \lambda f. G(f)
		~{}={}~
		\mu \gg\!= \lambda (f,f'). F(f,f')~.
		\]
		%
		%
		\item\label{lem:bind_project2} If  $\forall (f,f') \in P_\tau\colon F(f,f') = G'(f)$, then for all $\mu \in \mathcal{D}(P_\tau + \bot)$,
		%
		\[
		\pi_2(\mu) \gg\!= \lambda f. G'(f)
		~{}={}~
		\mu \gg\!= \lambda (f,f'). F(f,f')~.
		\]
	\end{enumerate}
\end{lemma}
%
%
\begin{proof}
	We only prove \Cref{lem:bind_project}.\ref{lem:bind_project1} because the reasoning for \Cref{lem:bind_project}.\ref{lem:bind_project2} is completely analogous. We have
	%
	\begin{align*}
		& \pi_1(\mu) \gg\!= \lambda f. G(f) \\
		%
		{}={}~& (\mu \gg\!= \lambda (f,f'). \delta_f) \gg\!= \lambda f. G(f) 
		\tag{definition} \\
		%
		{}={}~& \mu \gg\!= \lambda (f,f'). (\delta_f \gg\!= \lambda f. G(f))
		\tag{monad assoc.} \\
		%
		{}={}~& \mu \gg\!= \lambda (f,f'). G(f)
		\tag{monad neutral} \\
		%
		{}={}~& \mu \gg\!= \lambda (f,f'). F(f,f')~.
		\tag{assumption} 
	\end{align*}
	%
\end{proof}

\subsection{Big-Step Semantics and Correctness of Discretization}
%
We extend both of the small-step semantics canoncially to an $n$-step semantics by%
\begin{align*}
		\bigsemstep{e : \tau}{n} &{}= 
		\begin{cases}
			\delta_{e : \tau} & \text{if $n = 0$} \\
			%
			\bigsemstep{e : \tau}{n-1} \monbind  \lambda e'. \sem{e' : \tau} & \text{if $n > 0$}~.
		\end{cases} \\
		%
		%
		\bigsemstep{e : \tau, \discretize{e} : \discretize{\tau}}{n} &{}= 
		\begin{cases}
			\delta_{(e : \tau, \discretize{e} : \discretize{\tau})} & \text{if $n = 0$} \\
			%
			\bigsemstep{e : \tau, \discretize{e} : \discretize{\tau}}{n-1} \monbind  \lambda (e', e''). \sem{e' : \tau, e'' : \discretize{\tau}} & \text{if $n > 0$}~.
		\end{cases}
\end{align*}
%
%

%
%
As a sanity check, we show that the $n$-step semantics defines a Markov kernel.

\begin{lemma}\label{lem:big-step-n-kernel}
	For all $n \in \Nats$, $\bigsemstep{\cdot}{n} : Expr_{\tau} \to \mathcal{D}(Expr_{\tau} + \bot)$ is a Markov kernel.
\end{lemma}

\begin{proof}
	By induction on $n$. 

	Base case: $n = 0$. By definition, $\bigsemstep{e : \tau}{n} = \dirac{e : \tau}$, i.e. the map $e \mapsto \dirac{e}$. By~\Cref{lem:dirac-return-kernel}, this is a Markov kernel.

	Inductive step: $n > 0$. By definition, $\bigsemstep{e : \tau}{n-1} \monbind  \lambda e'. \sem{e' : \tau}$. By the I.H., $\bigsemstep{\cdot}{n-1} \colon Expr_{\tau} \to \mathcal{D}(Expr_{\tau} + \bot)$ is a Markov kernel. By~\Cref{lem:small-step-kernel}, $\lambda e' .\sem{e' : \tau} : Expr_{\tau} \to \mathcal{D}(Expr_{\tau} + \bot)$ is a Markov kernel. By~\Cref{lem:bot-extension-kernel}, its $\bot$-extension $(Expr_{\tau} + \bot)$ to $\mathcal{D}(Expr_{\tau} + \bot)$ is also a Markov kernel. By~\Cref{lem:kernel-composition}, $\bigsemstep{e : \tau}{n-1} \monbind  \lambda e'. \sem{e' : \tau}$ is a Markov kernel.
\end{proof}

\begin{lemma}\label{lem:big-step-n-coupling-kernel}
	For all $n \in \Nats$, $\bigsemstep{\cdot, \discretize{\cdot} : \discretize{\tau}}{n} : P_{\tau} \to \mathcal{D}(P_{\tau} + \bot)$ is a Markov kernel.
\end{lemma}

\begin{proof}
	Analogous to proof of~\Cref{lem:big-step-n-kernel}.
\end{proof}


For every type $\tau$, we denote by $Vals_\tau$ the set of values of type $\tau$ and equip it with a suitable measurable sub-space of the measurable space over $Expr_\tau$. Moreover, let $V_\tau = \{ (v:\tau, \discretize{v} : \discretize{\tau}) ~|~ v:\tau \}$.


\begin{lemma}[Basic Properties of $\bigsemstep{\cdot}{n}$]
	\label{lem:bigstepsem}
	Let $e : \tau \in Expr_{\tau}$ and $n \in \mathbb{N}$. Let $V$ be the measurable set of values. We have:
	\begin{enumerate}
		\item\label{lem:bigstepsem1} $\bigsemstep{e : \tau}{n}(V) \leq \bigsemstep{e : \tau}{n+1}(V)$
		\item\label{lem:bigstepsem2} $\bigsemstep{e : \tau}{n}(\bot) \leq \bigsemstep{e : \tau}{n+1}(\bot)$
	\end{enumerate}
\end{lemma}
%
\begin{proof}
	By induction on $n$.

	\underline{Base case}: $n=0$.

	We must show $\bigsemstep{e : \tau}{0}(V) \leq \bigsemstep{e : \tau}{1}(V)$.

	By definition,
	\[
		\bigsemstep{e : \tau}{0} = \delta_{e : \tau}.
	\]

	Therefore,
	\[
		\bigsemstep{e : \tau}{0}(V) =
		\begin{cases}
			1 &\text{ if } e \in V \\
			0 &\text{ if } e \notin V
		\end{cases}
	\]

	Now, for $n=1$, by defintiion,
	\[
		\bigsemstep{e : \tau}{1} = \delta_{e : \tau} \gg\!= \lambda e'.\sem{e'} = \sem{e : \tau}.
	\]

	We analyze the two cases for $e$:

	\textbf{Case 1:} $e$ is a value $v$. Then, $\bigsemstep{e : \tau}{0}(V) = 1$.

	For $n=1$:
	\[
		\bigsemstep{e : \tau}{1} = \delta_{v : \tau} \gg\!= \lambda e'.\sem{e'} = \sem{v : \tau}.
	\]
	From the semantics, $\sem{v : \tau} = \delta_{v : \tau}$, which means $\bigsemstep{e : \tau}{1} = \delta_{v : \tau}$, and thus
	\[
		\bigsemstep{e : \tau}{1}(V) = 1,
	\]
	and thus $\bigsemstep{e : \tau}{0}(V) \leq \bigsemstep{e : \tau}{1}(V)$.

	\textbf{Case 2:} $e$ is an expression. Then, $\bigsemstep{e : \tau}{0}(V) = 0$.

	For $n=1$:
	\[
		\bigsemstep{e : \tau}{1} = \delta_{e : \tau} \gg\!= \lambda e'.\sem{e'} = \sem{e : \tau}.
	\]
	From the semantics, $\sem{e : \tau}$ may step to either a value, which means $\sem{e : \tau}(V) \geq 0$, or an expression, which means $\sem{e : \tau}(V) = 0$. In either case, 
	\[
		\bigsemstep{e : \tau}{1}(V) \geq 0,
	\]
	and thus $\bigsemstep{e : \tau}{0}(V) \leq \bigsemstep{e : \tau}{1}(V)$.

	\underline{Inductive step}:
	We want to show that for $n = k+1$:
	\begin{align*}
		\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V).
	\end{align*}

	By the recursive definition:
	\begin{align*}
		&\bigsemstep{e : \tau}{k}, \\
		&\bigsemstep{e : \tau}{k+1} = \bigsemstep{e : \tau}{k} \gg\!= \lambda e'.\sem{e'}, \\
		&\bigsemstep{e : \tau}{k+2} = \bigsemstep{e : \tau}{k+1} \gg\!= \lambda e''.\sem{e''} = (\bigsemstep{e : \tau}{k} \gg\!= \lambda e'.\sem{e'}) \gg\!= \lambda e''.\sem{e''}
	\end{align*}

	By the IH,
	\begin{align*}
		\bigsemstep{e : \tau}{k}(V) \leq \bigsemstep{e : \tau}{k+1}(V).
	\end{align*}

	We do a case analysis on $e'$ and $e''$, whether they are in $N$ or $V$.

	\textbf{Case} $e' \in V$, $e'' \in V$.

	Then, from the semantics, $\sem{e'}$ and $\sem{e''}$ are both Diracs over a value. Combined with the IH and the fact that $\sem{\cdot}$ does not change the probability mass of $\cdot$ if it is a value, 
	\[
		\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V).
	\]

	\textbf{Case} $e' \in V$, $e'' \in N$.
	
	Then, from the semantics, $\sem{e'}$ is a Dirac over a value and $\sem{e''}$ is a distribution over a non-value expression. Combined with the IH and the fact that $\sem{\cdot}$ does not change the probability mass of $\cdot$ if it is a value and that $\sem{\cdot}$ does not add any probability mass to $V$ if $\cdot$ is a non-value expression,
	\[
		\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V).
	\]

	\textbf{Case} $e' \in V$, $e'' \in N$.

	Analogous to the previous case, but with $e'$ and $e''$ switched.

	\textbf{Case} $e' \in N$, $e'' \in N$.

	Then, from the semantics,  $\sem{e'}$ and $\sem{e''}$ are both distributions over a non-value expression. Combined with the IH and the fact that $\sem{\cdot}$ does not add any probability mass to $V$ if $\cdot$ is a non-value expression, 
	\[
		\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V).
	\]

	In all cases, $\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V)$.
\end{proof}

Given $e : \tau \in Expr_{\tau}$ and $n\in\Nats$, define $\bigsemstepterm{e : \tau}{n} = \lambda \measset. \bigsemstep{e : \tau}{n}(\measset \setminus N)$.

	Let $(e : \tau, e' : \tau') \in \relation_{\tau}$ and $n \in \mathbb{N}$. Let $V \times V$ be the measurable set of value pairs. We have:
	\begin{enumerate}

		\item\label{lem:bigstepsem2-pairing} $\bigsemstep{e : \tau, e' : \tau'}{n}(\bot) \leq \bigsemstep{e : \tau, e' : \tau'}{n+1}(\bot)$
	\end{enumerate}
\end{lemma}

\begin{proof}
	By induction on $n$. Similar to proof for single case.
\end{proof}

\begin{definition}
	Let $e:\tau$. The \emph{(big-step) semantics} $\bigsem{e:\tau} \in \mathcal{D}(Vals_\tau + \bot)$ and $\bigsem{e:\tau} \in \mathcal{D}(V_\tau + \bot)$ are respectively defined as

		\bigsem{e:\tau} ={}& \lambda A. \lim_{n\to\infty} \bigsemstep{e:\tau}{n}(A) \\
		%
		\bigsem{e:\tau,  \discretize{e} : \discretize{\tau}} ={}& \lambda A. \lim_{n\to\infty} \bigsemstep{e:\tau,  \discretize{e} : \discretize{\tau}}{n}(A)
	\end{align*}
	%
\end{definition}

% Well-definedness of these objects follows from the monotone convergence theorem: The probability of evaluating to some set of values or an error is montonically increasing in $n$ and upper-bounded by $1$. 
By~\Cref{lem:bigstepsem1,lem:bigstepsem2}, the sequence $\bigsemstep{e : \tau}{n}$ is monotonically increasing for values and error. Moreover, $\bigsemstep{e : \tau}{n}$ defines a subprobability measure, and thus $\bigsemstep{e : \tau}{n} \leq 1$ for every $n \in \Nats$. Thus, by the monotone convergence theorem, the pointwise limit exists.

\begin{lemma}[Continunity of projections]
	For all $n \in \Nats$,
	\begin{enumerate}
		\item $\pi_1(\lim_{n \to \infty} \bigsemstep{e,e'}{n}) = \lim_{n \to \infty} \pi_1(\bigsemstep{e,e'}{n})$
		\item $\pi_2(\lim_{n \to \infty} \bigsemstep{e,e'}{n}) = \lim_{n \to \infty} \pi_2(\bigsemstep{e,e'}{n})$ 
	\end{enumerate}
\end{lemma}

\begin{proof}
	We prove part i) in detail; the proof of part ii) is analogous. Fix an aribitrary measurable $A \in \Sigma_1$:
	\begin{align*}
		&\pi_1(\lim_{n \to \infty} \bigsemstep{e,e'}{n})(A) \\
		&= \lim_{n \to \infty} \bigsemstep{e,e'}{n}(A \times X_2) \quad \text{(by defintion of $\pi_1$)}\\
		&= \lim_{n \to \infty} \pi_1(\bigsemstep{e,e'}{n})(A) \quad \text{(by defintion of $\pi_1$)}
	\end{align*}
	Since $A$ was arbitrary, this implies:
	\[
		\pi_1(\lim_{n \to \infty} \bigsemstep{e,e'}{n}) =  \lim_{n \to \infty} \pi_1(\bigsemstep{e,e'}{n}),
	\]
	as desired.
\end{proof}

\begin{lemma}[n-step projection]
	For all $n \in \Nats$, 
	\begin{enumerate}
		\item $\pi_1(\bigsemstep{e : \tau,e' : \tau'}{n}) = \bigsemstep{e : \tau}{n}$
		\item $\pi_2(\bigsemstep{e : \tau,e' : \tau'}{n}) = \bigsemstep{e' : \tau'}{n}$
	\end{enumerate}
\end{lemma}

\begin{proof}
	By induction on $n$. We prove part i) in detail; the proof of part ii) is analogous.

	\underline{Base case:} $n=0$:

	By definition of the big-step semantics, 
	\[
		\bigsemstep{e : \tau, e' : \tau'}{0} = \delta_{(e : \tau, e' : \tau')}
	\]

	Projecting the first component: $\pi_1(\delta_{(e : \tau, e' : \tau')}) = \delta_{e : \tau} = \bigsemstep{e : \tau}{0}$.

	\underline{Inductive step:} Assume for $n-1$. For $n > 0$:
	\[
		\bigsemstep{e : \tau, e' : \tau'}{n} = \bigsemstep{e : \tau, e' : \tau'}{n-1} \monbind  \lambda (f,f'). \sem{f : \tau, f' : \tau'}
	\]

	By Lemma~\ref{lem:bind-proj},
	\[
		\pi_1(\bigsemstep{e : \tau, e' : \tau'}{n}) = \pi_1(\bigsemstep{e : \tau, e' : \tau'}{n-1}) \monbind  \lambda f. \pi_1(\sem{f : \tau, f' : \tau'})
	\]

	By the IH, $\pi_1(\bigsemstep{e : \tau, e' : \tau'}{n-1}) = \bigsemstep{e : \tau}{n-1}$.

	By Lemma~\ref{lem:small-step-proj}, $\pi_1(\sem{f : \tau, f' : \tau'}) = \sem{f : \tau}$.

	So,
	\[
		\pi_1(\bigsemstep{e : \tau, e' : \tau'}{n}) = \bigsemstep{e : \tau}{n-1} \monbind \lambda f.\sem{f : \tau} = \bigsemstep{e : \tau}{n}.
	\]
\end{proof}

\begin{lemma}[Coupling projection for big-step]\label{lem:big-step-proj}
	Let $\bigsem{e : \tau, e' : \tau'} \in \mathcal{D}(\relation_{\tau} + \bot)$. Then,
	\begin{enumerate}
		\item $\pi_1(\bigsem{e : \tau, e' : \tau'}) = \bigsem{e : \tau}$
		\item $\pi_2(\bigsem{e : \tau, e' : \tau'}) = \bigsem{e' : \tau'}$
	\end{enumerate}
\end{lemma}

\begin{proof}
	\begin{align*}
		&\pi_1(\bigsem{e : \tau, e' : \tau'}) \\
		&= \pi_1(\lim_{n \to \infty} \bigsemstep{e : \tau, e' : \tau'}{n}) \quad \text{(by defintion of big-step pairing semantics)} \\
		&= \lim_{n \to \infty} \pi_1(\bigsemstep{e : \tau, e' : \tau'}{n}) \quad \text{(by communitivity lemma)} \\
		&= \lim_{n \to \infty} \bigsemstep{e : \tau}{n} \quad \text{(by definition of n-step projection lemma)} \\
		&= \bigsem{e : \tau} \quad \text{(by definition of big-step semantics)}
	\end{align*}
\end{proof}
