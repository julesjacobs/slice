\section{Soundness Proof}\label{sec:soundness}

This section establishes the soundness of our discretization transformation. We proceed in three steps: first, we define a nondeterministic small-step semantics for the base language; second, we extend this to a probabilistic semantics that tracks distributions; and third, we prove that discretization preserves the probabilistic semantics.

\subsection{Nondeterministic Small-Step Semantics}

We begin by defining a small-step operational semantics for \Slice{} that treats probabilistic sampling nondeterministically. This semantics captures the possible values that expressions can take without tracking their probabilities.

\subsubsection{Values}

First, we define the set of values that expressions can evaluate to:
\begin{align*}
v ::= &\; c                                  & \text{float constant} \\
    | &\; \text{true} \mid \text{false}      & \text{boolean value} \\
    | &\; \finconst{k}{n}                    & \text{finite type value} \\
    | &\; i                                  & \text{integer value} \\
    | &\; ()                                 & \text{unit value} \\
    | &\; (v_1, v_2)                         & \text{pair value} \\
    | &\; \funkw \; x \; \rightarrow \; e    & \text{function closure} \\
    | &\; \text{nil}                         & \text{empty list} \\
    | &\; v_1 :: v_2                         & \text{list cons value} \\
    | &\; \ell                               & \text{location (for references)}
\end{align*}

\subsubsection{Evaluation Contexts}

We use evaluation contexts to specify the order of evaluation:
\begin{align*}
E ::= &\; [\cdot] \\
    | &\; \letkw \; x = E \; \inkw \; e \\
    | &\; E < e \mid v < E \\
    | &\; E \leq e \mid v \leq E \\
    | &\; E \finlt{n} e \mid v \finlt{n} E \\
    | &\; E \finleq{n} e \mid v \finleq{n} E \\
    | &\; E \logand e \mid v \logand E \\
    | &\; E \logor e \mid v \logor E \\
    | &\; \text{not}\; E \\
    | &\; \ifkw \; E \; \thenkw \; e_1 \; \elsekw \; e_2 \\
    | &\; (E, e) \mid (v, E) \\
    | &\; \fstkw \; E \mid \sndkw \; E \\
    | &\; E \; e \mid v \; E \\
    | &\; \observekw\; E \\
    | &\; E; e \\
    | &\; E :: e \mid v :: E \\
    | &\; \text{match}\; E \; \text{with}\; \ldots \\
    | &\; \text{ref}\; E \\
    | &\; !E \\
    | &\; E := e \mid v := E
\end{align*}

\subsubsection{Small-Step Relation}

We define the small-step relation $e \rightarrow \mathcal{P}(e')$ that maps an expression to a set of possible next expressions. For deterministic constructs, this set is a singleton; for probabilistic sampling, it contains all possible values in the distribution's support.

\paragraph{Basic Rules}

\begin{mathpar}
\inferrule[\textsc{Let}]
{\ }
{\letkw \; x = v \; \inkw \; e \rightarrow \{e[v/x]\}}

\inferrule[\textsc{IfTrue}]
{\ }
{\ifkw \; \text{true} \; \thenkw \; e_1 \; \elsekw \; e_2 \rightarrow \{e_1\}}

\inferrule[\textsc{IfFalse}]
{\ }
{\ifkw \; \text{false} \; \thenkw \; e_1 \; \elsekw \; e_2 \rightarrow \{e_2\}}

\inferrule[\textsc{Fst}]
{\ }
{\fstkw \; (v_1, v_2) \rightarrow \{v_1\}}

\inferrule[\textsc{Snd}]
{\ }
{\sndkw \; (v_1, v_2) \rightarrow \{v_2\}}

\inferrule[\textsc{App}]
{\ }
{(\funkw \; x \; \rightarrow \; e) \; v \rightarrow \{e[v/x]\}}
\end{mathpar}

\paragraph{Comparison Rules}

\begin{mathpar}
\inferrule[\textsc{LessTrue}]
{c_1 < c_2}
{c_1 < c_2 \rightarrow \{\text{true}\}}

\inferrule[\textsc{LessFalse}]
{c_1 \geq c_2}
{c_1 < c_2 \rightarrow \{\text{false}\}}

\inferrule[\textsc{LeqTrue}]
{i_1 \leq i_2}
{i_1 \leq i_2 \rightarrow \{\text{true}\}}

\inferrule[\textsc{LeqFalse}]
{i_1 > i_2}
{i_1 \leq i_2 \rightarrow \{\text{false}\}}
\end{mathpar}

\paragraph{Boolean Operations}

\begin{mathpar}
\inferrule[\textsc{AndTrue}]
{\ }
{\text{true} \logand e \rightarrow \{e\}}

\inferrule[\textsc{AndFalse}]
{\ }
{\text{false} \logand e \rightarrow \{\text{false}\}}

\inferrule[\textsc{OrTrue}]
{\ }
{\text{true} \logor e \rightarrow \{\text{true}\}}

\inferrule[\textsc{OrFalse}]
{\ }
{\text{false} \logor e \rightarrow \{e\}}

\inferrule[\textsc{NotTrue}]
{\ }
{\text{not}\; \text{true} \rightarrow \{\text{false}\}}

\inferrule[\textsc{NotFalse}]
{\ }
{\text{not}\; \text{false} \rightarrow \{\text{true}\}}
\end{mathpar}

\paragraph{List Operations}

\begin{mathpar}
\inferrule[\textsc{MatchNil}]
{\ }
{\text{match}\; \text{nil} \; \text{with}\; \text{nil} \rightarrow e_1 \mid h :: t \rightarrow e_2 \; \text{end} \rightarrow \{e_1\}}

\inferrule[\textsc{MatchCons}]
{\ }
{\text{match}\; v_1 :: v_2 \; \text{with}\; \text{nil} \rightarrow e_1 \mid h :: t \rightarrow e_2 \; \text{end} \rightarrow \{e_2[v_1/h, v_2/t]\}}
\end{mathpar}

\paragraph{Probabilistic Sampling (Nondeterministic)}

For continuous distributions, the small-step relation returns the set of all possible values in the distribution's support:

\begin{mathpar}
\inferrule[\textsc{Uniform}]
{v_1, v_2 \text{ are float values} \\ v_1 < v_2}
{\uniform(v_1, v_2) \rightarrow \{c \mid v_1 \leq c < v_2\}}

\inferrule[\textsc{Gaussian}]
{\mu, \sigma \text{ are float values} \\ \sigma > 0}
{\gaussian(\mu, \sigma) \rightarrow \{c \mid c \in \mathbb{R}\}}

\inferrule[\textsc{Exponential}]
{\lambda \text{ is a float value} \\ \lambda > 0}
{\exponential(\lambda) \rightarrow \{c \mid c \geq 0\}}

\inferrule[\textsc{Beta}]
{\alpha, \beta \text{ are float values} \\ \alpha > 0 \\ \beta > 0}
{\betafn(\alpha, \beta) \rightarrow \{c \mid 0 \leq c \leq 1\}}

\inferrule[\textsc{Discrete}]
{p_0, \ldots, p_n \text{ are probabilities}}
{\discrete(p_0, \ldots, p_n) \rightarrow \{0, 1, \ldots, n\}}
\end{mathpar}

\paragraph{Observations}

\begin{mathpar}
\inferrule[\textsc{ObserveTrue}]
{\ }
{\observekw\; \text{true} \rightarrow \{()\}}

\inferrule[\textsc{ObserveFalse}]
{\ }
{\observekw\; \text{false} \rightarrow \emptyset}
\end{mathpar}

Note that observing a false condition results in an empty set, representing program termination without producing a value.

\paragraph{References and State}

For references, we extend the semantics with a store $\sigma$ that maps locations to values. The relation becomes $\langle e, \sigma \rangle \rightarrow \mathcal{P}(\langle e', \sigma' \rangle)$:

\begin{mathpar}
\inferrule[\textsc{Ref}]
{\ell \text{ is a fresh location}}
{\langle \text{ref}\; v, \sigma \rangle \rightarrow \{\langle \ell, \sigma[\ell \mapsto v] \rangle\}}

\inferrule[\textsc{Deref}]
{\sigma(\ell) = v}
{\langle !\ell, \sigma \rangle \rightarrow \{\langle v, \sigma \rangle\}}

\inferrule[\textsc{Assign}]
{\ }
{\langle \ell := v, \sigma \rangle \rightarrow \{\langle (), \sigma[\ell \mapsto v] \rangle\}}
\end{mathpar}

\paragraph{Context Rule}

\begin{mathpar}
\inferrule[\textsc{Context}]
{e \rightarrow S}
{E[e] \rightarrow \{E[e'] \mid e' \in S\}}
\end{mathpar}

This nondeterministic semantics captures all possible execution paths of a probabilistic program without tracking the probability of each path. In the next subsection, we will extend this to a probabilistic semantics that properly tracks distributions.

\subsection{Probabilistic Small-Step Semantics}

We now define a probabilistic semantics that tracks the distribution over expressions. Define
%
\[
	\extexprs = \exprs \cup  \{ \error \} \cup \{ \obsfail \}~,
\]
%
and call them \emph{extended expressions}.
%
%where $\error$ indicates a type error and $\obsfail$ denotes observation failure.
The semantic function $\sem{\cdot} \colon \exprs \rightarrow \distexprs$ maps expressions to distributions over expressions, where $\dist{X}$ denotes the space of probability distributions over $X$.

We assume the existence of the Giry monad structure on $\mathcal{D}$, which provides:
\begin{itemize}
    \item $\monunit \colon X \rightarrow \dist{X}$ (also written as $\delta_x$ for the Dirac distribution at $x$)
    \item $\monbind \colon \dist{X} \times (X \to \dist{Y}) \rightarrow \dist{Y}$ (monadic bind)
\end{itemize}

\subsubsection{Deterministic Constructs}

For deterministic constructs, the semantics returns a Dirac distribution:

\begin{align*}

\sem{\letkw \; \var = \val \; \inkw \; \expr} &= \delta_{\expr\subst{\val}{\var}} \\

\begin{cases}

\end{cases}

\begin{cases}

	%
	\delta_{\error}  & \text{otherwise}
\end{cases}
%
\\

\sem{ (\funkw \; \var \; \rightarrow \; \expr) \; \val } &= \delta_{\expr\subst{\val}{\var}}\\
\sem{\error} = \delta_\error \\
\sem{\obsfail} = \delta_\obsfail
\end{align*}

\begin{align*}

\subsubsection{Compositional Rules}

For compound expressions, we use monadic bind to compose distributions:

\paragraph{Binary Operations}
\begin{align*}
\sem{ e_1 < e_2 } &= \sem{ e_1 } \gg\!= \lambda v_1. \sem{ v_1 < e_2 } \\
\sem{ v_1 < e_2 } &= \sem{ e_2 } \gg\!= \lambda v_2. \sem{ v_1 < v_2 }
\end{align*}

Similar rules apply for $\leq$, $\logand$, $\logor$, and other binary operations.

\paragraph{Let Binding}
\begin{align*}
\sem{ \letkw \; x = e_1 \; \inkw \; e_2 } &= \sem{ e_1 } \gg\!= \lambda v. \sem{ \letkw \; x = v \; \inkw \; e_2 }
\end{align*}

\paragraph{Conditionals}
\begin{align*}
\sem{ \ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3 } &= \sem{ e_1 } \gg\!= \lambda v. \sem{ \ifkw \; v \; \thenkw \; e_2 \; \elsekw \; e_3 }
\end{align*}

\paragraph{Function Application}
\begin{align*}
\sem{ e_1 \; e_2 } &= \sem{ e_1 } \gg\!= \lambda v_1. \sem{ v_1 \; e_2 } \\
\sem{ v_1 \; e_2 } &= \sem{ e_2 } \gg\!= \lambda v_2. \sem{ v_1 \; v_2 }
\end{align*}

\paragraph{Pairs and Projections}
\begin{align*}
\sem{ (e_1, e_2) } &= \sem{ e_1 } \gg\!= \lambda v_1. \sem{ (v_1, e_2) } \\
\sem{ (v_1, e_2) } &= \sem{ e_2 } \gg\!= \lambda v_2. \delta_{(v_1, v_2)} \\
\sem{ \fstkw \; e } &= \sem{ e } \gg\!= \lambda v. \sem{ \fstkw \; v } \\
\sem{ \sndkw \; e } &= \sem{ e } \gg\!= \lambda v. \sem{ \sndkw \; v }
\end{align*}

\paragraph{Sequencing}
\begin{align*}
\sem{ e_1; e_2 } &= \sem{ e_1 } \gg\!= \lambda v_1. \sem{ e_2 }
\end{align*}

\subsubsection{DistrCase}

For distrcase expressions (which represent explicit discrete distributions in the implementation):

\begin{align*}
\sem{ \text{distrcase}\{(e_0, p_0), \ldots, (e_n, p_n)\} } &= \sum_{i=0}^n p_i \cdot \sem{ e_i }
\end{align*}

\subsubsection{Example}

Consider the expression:
\begin{lstlisting}
let x = uniform(0.0, 1.0) in
if x < 0.5 then 1 else 2
\end{lstlisting}

The probabilistic semantics gives:
\begin{align*}
&\sem{ \text{let } x = \uniform(0.0, 1.0) \text{ in if } x < 0.5 \text{ then } 1 \text{ else } 2 } \\
&= \sem{ \uniform(0.0, 1.0) } \gg\!= \lambda x. \sem{ \text{if } x < 0.5 \text{ then } 1 \text{ else } 2 } \\
&= \text{Uniform}_{[0,1)} \gg\!= \lambda x. \sem{ \text{if } x < 0.5 \text{ then } 1 \text{ else } 2 } \\
&= \text{Uniform}_{[0,1)} \gg\!= \lambda x. \begin{cases}
    \delta_1 & \text{if } x < 0.5 \\
    \delta_2 & \text{if } x \geq 0.5
\end{cases} \\
&= 0.5 \cdot \delta_1 + 0.5 \cdot \delta_2
\end{align*}

\subsection{Measure Space on Expressions}

To make the probabilistic semantics precise, we need to define an appropriate measure space structure on the set of extended expressions. We construct this measure space using the notion of \emph{expression skeletons}.

\subsubsection{Expression Skeletons}

An expression skeleton is an extended expression with ``holes'' where real-valued constants can appear. Formally, we define:

\begin{definition}[Expression Skeleton]
An expression skeleton $s$ is an extended expression where each occurrence of a float constant is replaced by a hole $\square_i$ for some index $i \in \mathbb{N}$. We denote by $\text{Skel}$ the set of all expression skeletons, and by $\text{holes}(s)$ the number of distinct holes in skeleton $s$.
\end{definition}

For example, the expression $\uniform(0.0, 1.0) < 0.5$ has the skeleton $\uniform(\square_1, \square_2) < \square_3$ with three holes.

\subsubsection{Measure Space Construction}

Given a skeleton $s$ with $n = \text{holes}(s)$ holes, we can instantiate it with a vector $\vec{r} = (r_1, \ldots, r_n) \in \mathbb{R}^n$ to produce a concrete expression $s[\vec{r}]$ by replacing each hole $\square_i$ with the real value $r_i$.

The set of all extended expressions can then be represented as:
\[
\extexprs = \bigsqcup_{s \in \text{Skel}} \{s[\vec{r}] \mid \vec{r} \in \mathbb{R}^{\text{holes}(s)}\}
\]

This is a countable disjoint union because:
\begin{enumerate}
    \item The set of skeletons $\text{Skel}$ is countable (expressions have finite syntax trees)
    \item For each skeleton $s$, the set of instantiations forms a copy of $\mathbb{R}^{\text{holes}(s)}$
    \item Different skeletons produce disjoint sets of expressions
\end{enumerate}

\subsubsection{Measurable Sets}

We equip $\extexprs$ with the $\sigma$-algebra generated by sets of the form:
\[
\{s[\vec{r}] \mid \vec{r} \in B\}
\]
where $s$ is a skeleton and $B \subseteq \mathbb{R}^{\text{holes}(s)}$ is a Borel measurable set.

This construction ensures that:
\begin{itemize}
    \item Basic operations like evaluating an expression to a value are measurable
    \item The distributions returned by our probabilistic semantics are well-defined probability measures
    \item We can integrate over expressions when needed for semantic definitions
\end{itemize}

\subsubsection{Distributions Over Expressions}

When we write $\mathcal{D}(\extexprs)$ in our probabilistic semantics, we mean the space of probability measures on $(\extexprs, \mathcal{F})$ where $\mathcal{F}$ is the $\sigma$-algebra described above.

For continuous distributions like $\uniform(a, b)$, the implicit coercion to $\mathcal{D}(\extexprs)$ works as follows:
\begin{enumerate}
    \item The uniform distribution on $[a, b)$ is a measure on $\mathbb{R}$
    \item We lift it to expressions via the skeleton containing a single hole: $s = \square_1$
    \item The resulting distribution on expressions assigns probability to sets of the form $\{r \mid r \in B\}$ for Borel sets $B \subseteq [a, b)$
\end{enumerate}

This measure-theoretic foundation ensures that our probabilistic semantics is mathematically rigorous and that operations like conditioning (via observations) and marginalization are well-defined.

We consider a subset of our language, focusing only on let statements, conditionals, less than comparisons, and uniform and discrete distributions.

\begin{definition}[Sub-Distribution]
	A sub-probability measure (the continuous analogue of a sub-distribution) on a measureable space $(X, \Sigma)$ is a measure $\mu: \Sigma \to [0,1]$ such that $\mu(X) \leq 1$. We write $D(X)$ for the set of all sub-probability measures of $X$. 
\end{definition}

\begin{definition}[Support]
	The support of $\mu \in \mathcal{D}(X)$ is the set of points $x \in X$ such that every open neighborhood of $x$ has positive measure:
	\[
	{support}(\mu) \triangleq \{ x \in X \mid \forall U \ni x, \; U \text{ open}, \; \mu(U) > 0 \}.
	\]
\end{definition}

\begin{definition}[Distribution Monad with Error]
	Define $\text{return} : X \to \mathcal{D}(X + \bot)$ by $\text{return}_X(x) := \delta_x$, where $\delta_x$ denotes the Dirac distribution on $x$: for any $y \in X$, we have $\delta_x(y) = 1$ if $y = x$, otherwise $\delta_x(y) = 0$. Further, define $\text{bind} : \mathcal{D}(X + \bot) \to (X \to \mathcal{D}(Y + \bot)) \to \mathcal{D}(Y + \bot)$ by $\text{bind}(\mu)(f)(y) := \sum_{x \in support(\mu)} \mu(x) \cdot f_{\bot}(x)(y)$, 
	where $f_{\bot}$ extends $f$ by $f_{\bot}(\bot) := \delta_{\bot}$ and $f_{\bot}(x) := f(x) \;\; \text{for every } x \in X.$
\end{definition}

\subsubsection{\dc}
We introduce a new construct, called \dc, that simulates the latter two cases of the discretization of the uniform distribution in the discretization function. We define:

\begin{align*}
	&\discretize{\uniform(e_1 : \floattype{B_1}{V_1}, e_2 : \floattype{B_2}{V_2}) : \floattype{B}{V}} \\
	&= \dc(\discretize{e_1 : \floattype{B_1}{V_1}},\discretize{e_2 : \floattype{B_2}{V_2}})
\end{align*}

where $B \neq \top$, $V_1 = \{v_1, \ldots, v_n\} \neq \top$, and $V_2 = \{w_1, \ldots, w_m\} \neq \top$.

We give a typing rule for this new construct:
\begin{align*}
	\inferrule[\textsc{TDiscreteCase}]
	{e_1 : \fin{|B_1| + 1} \\
	e_2 : \fin{|B_2| + 1}}
	{\dc(e_1, e_2) : \fin{|B|+1}}
\end{align*}

Now the semantics is as follows:

Old:
{
\begin{align*}
	&\sem{ \dc(e_1 : \tau_1, e_2 : \tau_2) : \tau} \\
	&= \begin{cases}
		\sem{ \discretize{\uniform(v_{i+1}, w_{j+1}) : \floattype{B}{\top}} : \tau } \quad &\text{ if } e_1 = i, e_2 = j \\
		\sem{ e_2 : \tau_2 } \gg\!= \lambda f. \sem{ \dc(i : \tau_1, f) } \quad &\text{ if } e_1 = i, e_2 \neq j \\
		\sem{ e_1 : \tau_1 } \gg\!= \lambda f. \sem{ \dc(f, e_2 : \tau_2) } \quad &\text{ otherwise } \\
	\end{cases} \\
	&\text{where } \\
	&\tau = \fin{|B| + 1}, \\
	&\tau_1 = \fin{|B_1| + 1}, \tau_2 = \fin{|B_2| + 1} \\
	&V_1 = \{v_1,\ldots,v_n\}, V_2 = \{w_1,\ldots,w_m\} \\
\end{align*}
}

New:
{
\begin{align*}
	&\sem{ \dc(e_1 : \tau_1, e_2 : \tau_2) : \tau} \\
	&= \begin{cases}
		\sem{ \discretize{\uniform(v_i, w_j) : \floattype{B}{\top}} : \tau } \quad &\text{ if } e_1 = \discretize{v_i : \floattype{B_1}{v_i}} = v', \\
		&e_2 = \discretize{w_j : \floattype{B_2}{w_j}} = w' \\
		\sem{ e_2 : \tau_2 } \gg\!= \lambda f. \sem{ \dc(i : \tau_1, f) } \quad &\text{ if } e_1 = v', e_2 \neq w' \\
		\sem{ e_1 : \tau_1 } \gg\!= \lambda f. \sem{ \dc(f, e_2 : \tau_2) } \quad &\text{ otherwise } \\
	\end{cases} \\
	&\text{where } \\
	&\tau = \fin{|B| + 1}, \\
	&\tau_1 = \fin{|B_1| + 1}, \tau_2 = \fin{|B_2| + 1} \\
	&V_1 = \{v_1,\ldots,v_n\}, V_2 = \{w_1,\ldots,w_m\} \\
\end{align*}
}

\subsubsection{Lemmas for $e$}

\begin{lemma}[Substitution]\label{lem:substitution}
	If $\Gamma, x : \tau_1 \vdash e : \tau_2$ and $\emptyset \vdash v : \tau_1$, then $\Gamma \vdash e[v/x] : \tau_2$. 
\end{lemma}

\begin{proof}
	We proceed by induction on $e$:

	\textbf{Case} $e$ is a variable $y$. There are two subcases. 

	\begin{itemize}
		\item If $y = x$, then
		\[
			e[v/x] = v.
		\]
		We know
		\[
			\Gamma, x : \tau_1 \vdash x : \tau_1,
		\]
		so $\tau_2 = \tau_1$. Since $\emptyset \vdash v : \tau_1$, we also have
		\[
			\emptyset \vdash v : \tau_2.
		\]
		Hence
		\[
			\Gamma \vdash e[v/x] : \tau_2.
		\]

		\item If $y \neq x$, then the substitution does nothing:
		\[
			e[v/x] = y.
		\]
		Since $\Gamma, x : \tau_1 \vdash y : \tau_2$ and $y \neq x$, it follows that $y$ must be bound in $\Gamma$, and therefore
		\[
			\Gamma \vdash y : \tau_2.
		\]
		Thus,
		\[
			\Gamma \vdash e[v/x] : \tau_2.
		\]
	\end{itemize}

	\textbf{Case} $e$ is a constant or a discrete distribution $\discrete(v_0,\ldots,v_n)$. Substitution does not change either:
	\[
		e[v/x] = e.
	\]
	Typing of constants or discrete does not depend on $x$, so
	\[
		\Gamma \vdash e : \tau_2.
	\]
	Hence
	\[
		\Gamma \vdash e[v/x] : \tau_2.
	\]

	\textbf{Case} $e$ is a comparison $e_1 < e_2$. By \textsc{Less}, we have:
	\[
		\inferrule
		{\ \Gamma, x : \tau_1 \vdash e_1 : \tau \quad \Gamma, x : \tau_1 \vdash e_2 : \tau}
		{\Gamma, x : \tau_1 \vdash e_1 < e_2 : \bool}
	\]
	By the induction hypothesis,
	\[
		\Gamma \vdash e_1[v/x] : \tau
		\quad\text{and}\quad
		\Gamma \vdash e_2[v/x] : \tau.
	\]
	Given these, \textsc{Less} lets us derive:
	\[
		\Gamma \vdash e_1[v/x] < e_2[v/x] : \bool,
	\]
	and thus:
	\[
		\Gamma \vdash (e_1 < e_2)[v/x] : \bool
	\]
	since $(e_1 < e_2)[v/x] = e_1[v/x] < e_2[v/x]$ by the definition of substitution.

	\textbf{Case} $e$ is an if-else $\ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3$. By \textsc{If}, we have:
	\[
		\inferrule
		{\ \Gamma, x : \tau_1 \vdash e_1 : \bool \quad \Gamma, x : \tau_1 \vdash e_2 : \tau \quad \Gamma, x : \tau_1 \vdash e_3 : \tau}
		{\Gamma, x : \tau_1 \vdash \ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3 : \tau}
	\]
	By the induction hypothesis,
	\[
		\Gamma \vdash e_1[v/x] : \bool,
		\quad
		\Gamma \vdash e_2[v/x] : \tau,
		\quad
		\Gamma \vdash e_3[v/x] : \tau.
	\]
	Given these, \textsc{If} lets us derive:
	\[
		\Gamma \vdash \ifkw \; e_1[v/x] \; \thenkw \; e_2[v/x] \; \elsekw \; e_3[v/x] : \tau,
	\]
	and thus:
	\[
		\Gamma \vdash (\ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3)[v/x] : \tau
	\]
	since $(\ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3)[v/x] = \ifkw \; e_1[v/x] \; \thenkw \; e_2[v/x] \; \elsekw \; e_3[v/x]$ by the definition of substitution.

	\textbf{Case} $e$ is a let statement $\letkw \; y = e_1 \; \inkw \; e_2$. We distinguish two subcases.

	\begin{itemize}
		\item If $y \neq x$, then by \textsc{Let} we have:
		\[
			\inferrule
			{\ \Gamma, x : \tau_1 \vdash e_1 : \tau_1' \quad \Gamma, x : \tau_1, y : \tau_1' \vdash e_2 : \tau_2'}
			{\Gamma, x : \tau_1 \vdash \letkw \; y = e_1 \; \inkw \; e_2 : \tau_2'}
		\]
		By the induction hypothesis,
		\[
			\Gamma \vdash e_1[v/x] : \tau_1'
			\quad\text{and}\quad
			\Gamma, y : \tau_1' \vdash e_2[v/x] : \tau_2'.
		\]
		Given these, \textsc{Let} lets us derive:
		\[
			\Gamma \vdash \letkw \; y = e_1[v/x] \; \inkw \; e_2[v/x] : \tau_2',
		\]
		and thus:
		\[
			\Gamma \vdash (\letkw \; y = e_1 \; \inkw \; e_2)[v/x] : \tau_2'
		\]
		since $(\letkw \; y = e_1 \; \inkw \; e_2)[v/x] = \letkw \; y = e_1[v/x] \; \inkw \; e_2[v/x]$.

		\item If $y = x$, then by the typing rule:
		\[
			\inferrule
			{\ \Gamma, x : \tau_1 \vdash e_1 : \tau_1' \quad \Gamma, x : \tau_1 \vdash e_2 : \tau_2'}
			{\Gamma, x : \tau_1 \vdash \letkw \; x = e_1 \; \inkw \; e_2 : \tau_2'}
		\]
		By the induction hypothesis,
		\[
			\Gamma \vdash e_1[v/x] : \tau_1'.
		\]
		Given this, \textsc{Let} lets us derive:
		\[
			\Gamma \vdash \letkw \; x = e_1[v/x] \; \inkw \; e_2 : \tau_2',
		\]
		and thus:
		\[
			\Gamma \vdash (\letkw \; x = e_1 \; \inkw \; e_2)[v/x] : \tau_2'
		\]
		since $(\letkw \; x = e_1 \; \inkw \; e_2)[v/x] = \letkw \; x = e_1[v/x] \; \inkw \; e_2$ by capture-avoiding substitution.
	\end{itemize}

	\textbf{Case} $e$ is a uniform distribution $\uniform(e_1,e_2)$. By \textsc{Continuous-2}, we have:
	\[
		\inferrule
		{\ \Gamma, x : \tau_1 \vdash e_1 : \tau \quad \Gamma, x : \tau_1 \vdash e_2 : \tau}
		{\Gamma, x : \tau_1 \vdash \uniform(e_1,e_2) : \tau}
	\]
	By the induction hypothesis,
	\[
		\Gamma \vdash e_1[v/x] : \tau
		\quad\text{and}\quad
		\Gamma \vdash e_2[v/x] : \tau.
	\]
	Given these, \textsc{Continuous-2} lets us derive:
	\[
		\Gamma \vdash \uniform(e_1[v/x], e_2[v/x]) : \tau,
	\]
	and thus:
	\[
		\Gamma \vdash \uniform(e_1,e_2)[v/x] : \tau
	\]
	since $\uniform(e_1,e_2)[v/x] = \uniform(e_1[v/x], e_2[v/x])$ by the definition of substitution.
\end{proof}

\begin{lemma}[Canonical forms]\label{lem:canonical-forms}
	Let $v$ be a closed value. 
	\begin{enumerate}
		\item If $\emptyset \vdash v : \bool$, then $v \in \{\true, \false\}$.
		\item If $\emptyset \vdash v : \floattype{B}{V}$, then $v \in \R$.
		\item If $\emptyset \vdash v : \fin{n}$, then $v \in \Nats$.
	\end{enumerate}
\end{lemma}

\begin{proof}
	By case analysis on the grammar of values and inversion of typing.
\end{proof}

\begin{lemma}[Typed replacement]\label{lem:typed-replacement}
	All under empty context.
	\begin{enumerate}
		\item If $\emptyset \vdash \ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3 : \tau$ and $\emptyset \vdash e_1' : \bool$, then $\emptyset \vdash \ifkw \; e_1' \; \thenkw \; e_2 \; \elsekw \; e_3 : \tau$.
		
		\item If $\emptyset \vdash e_1 < e_2 : \bool$ and $\emptyset \vdash e_1' : \floattype{B}{V}$, then $\emptyset \vdash e_1' < e_2 : \bool$. Likewise, if $\emptyset \vdash e_1 < e_2 : \bool$ and $\emptyset \vdash e_2' : \floattype{B}{V}$, then $\emptyset \vdash e_1 < e_2' : \bool$.
		
		\item If $\emptyset \vdash e_1 \finlt{n} e_2 : \bool$ and $\emptyset \vdash e_1' : \fin{n}$, then $\emptyset \vdash e_1' < e_2 : \bool$. Likewise, if $\emptyset \vdash e_1 \finlt{n} e_2 : \bool$ and $\emptyset \vdash e_2' : \fin{n}$, then $\emptyset \vdash e_1 < e_2' : \bool$.
		
		\item If $\emptyset \vdash \uniform(e_1,e_2) : \floattype{B}{V}$ and $\emptyset \vdash e_1' : \floattype{B}{V}$, then $\emptyset \vdash \uniform(e_1',e_2) : \floattype{B}{V}$. Likewise, if $\emptyset \vdash \uniform(e_1,e_2) : \floattype{B}{V}$ and $\emptyset \vdash e_2' : \floattype{B}{V}$, then $\emptyset \vdash \uniform(e_1,e_2') : \floattype{B}{V}$.
		
		\item If $\emptyset \vdash \letkw \; x = e_1 \; \inkw \; e_2 : \tau_2$ and $\emptyset \vdash e_1' : \tau_1$, then $\emptyset \vdash \letkw \; x = e_1' \; \inkw \; e_2 : \tau_2$.
	\end{enumerate}
\end{lemma}

\begin{proof}
	Each item is a one-step inversion on the given typing and reconstruction with the indicated premise.
\end{proof}

\begin{proposition}[Bind typing principle]\label{prop:bind-typing}
	Let $A,B$ be measurable spaces. If $\mu \in \mathcal{D}(A)$ and $f : A \to \mathcal{D}(B)$, then $\mu \gg\!= f \in \mathcal{D}(B)$. In particular, if $f(a) = \delta_{b_a}$ with $b_a \in B$ for all $a \in A$, then $\mu \gg\!= f \in \mathcal{D}(B)$.
\end{proposition}

	
\begin{lemma}[Well-Defined]\label{lem:well-defined}
	For all well-typed expressions $e : \tau$, $\sem{e : \tau} \in \mathcal{D}(Expr_{\tau} + \bot)$.
\end{lemma}

\begin{proof}
	By induction on the structure of the typing derivation $e : \tau$. We do a case analysis on the last rule used.

		\textbf{Case} constant $v : \tau$, where $\tau = \floattype{B}{V}$. From the semantics, it holds that 
		\[
			\sem{v : \tau} = \delta_{v : \tau} \in \mathcal{D}(Expr_{\tau} + \bot).
		\]

		\textbf{Case} comparisons $e_1 < e_2 : \bool$. 
		From typing we have 
		\[
		\emptyset \vdash e_1 : \float 
		\qquad\text{and}\qquad 
		\emptyset \vdash e_2 : \float.
		\]

		We have the following possibilities:
		\begin{itemize}
		\item If both $e_1 = v_1$ and $e_2 = v_2$ are values, Lemma~\ref{lem:canonical-forms} gives float constants, and either
		\[
			\delta_{\true:\bool} \in \mathcal{D}(Expr_{\bool} + \bot) 
			\quad\text{or}\quad
			\delta_{\false:\bool} \in \mathcal{D}(Expr_{\bool} + \bot)
		\]
		is returned.

		\item At least $e_1$ or $e_2$ is an expression. 
		Suppose $e_1$ is an expression, then the clause
		\[
			\sem{e_1 : \float} \gg\!= \lambda e_1'.\;
			\sem{e_1' < e_2 : \float : \bool}
		\]
		in the semantics is applied. By the IH,
		\[
			\sem{e_1 : \float} \in \mathcal{D}(Expr_{\float} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_1') = \sem{e_1' < e_2},
		\]
		which maps
		\[
			Expr_{\float} \;\to\; \mathcal{D}(Expr_{\bool} + \bot)
		\]
		by Lemma~\ref{lem:typed-replacement}.

		By Proposition~\ref{prop:bind-typing}, we conclude
		\[
			\sem{e_1 : \float} \gg\!= \lambda e_1'.\;
			\sem{e_1' < e_2 : \float : \bool} \in \mathcal{D}(Expr_{\bool} + \bot).
		\]

		Now suppose $e_2$ is an expression, then the clause
		\[
			\sem{e_2 : \float} \gg\!= \lambda e_2'.\;
			\sem{e_1 : \float < e_2' : \bool}
		\]
		in the semantics is applied. By the IH,
		\[
			\sem{e_2 : \float} \in \mathcal{D}(Expr_{\float} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_2') = \sem{e_1 < e_2'},
		\]
		which maps
		\[
			Expr_{\float} \;\to\; \mathcal{D}(Expr_{\bool} + \bot)
		\]
		by Lemma~\ref{lem:typed-replacement}. 

		By Proposition~\ref{prop:bind-typing}, we conclude
		\[
			\sem{e_2 : \float} \gg\!= \lambda e_2'.\;
			\sem{e_1 : \float < e_2' : \bool} \in \mathcal{D}(Expr_{\bool} + \bot).
		\]
		\end{itemize}

		\textbf{Case} if-else $\ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3 : \tau$. From typing we have:
		\[
		\emptyset \vdash e_1 : \bool 
		\qquad\text{and}\qquad 
		\emptyset \vdash e_2 : \tau
		\qquad\text{and}\qquad 
		\emptyset \vdash e_3 : \tau.
		\]
		We have the following possibilities:
		\begin{itemize}
			\item If $e_1 = v$ is a value, Lemma~\ref{lem:canonical-forms} gives true or false, and either
			\[
				\delta_{e_2:\tau} \in \mathcal{D}(Expr_{\tau} + \bot) 
				\quad\text{or}\quad
				\delta_{e_3:\tau} \in \mathcal{D}(Expr_{\tau} + \bot)
			\]
			is returned.

			\item $e_1$ is an expression. Then the clause
			\[
				\sem{e_1 : \bool} \gg\!= \lambda e_1'.\;
				\sem{\ifkw \; e_1' \; \thenkw \; e_2 : \tau \; \elsekw \; e_3 : \tau : \tau}
			\]
			in the semantics is applied. By the IH,
			\[
				\sem{e_1 : \bool} \in \mathcal{D}(Expr_{\bool} + \bot).
			\]
			We use the left bind clause with 
			\[
				f(e_1') = \sem{\ifkw \; e_1' \; \thenkw \; e_2 \; \elsekw \; e_3},
			\]
			which maps
			\[
				Expr_{\bool} \;\to\; \mathcal{D}(Expr_{\tau} + \bot)
			\]
			by Lemma~\ref{lem:typed-replacement}.

			By Proposition~\ref{prop:bind-typing}, we conclude
			\[
				\sem{e_1 : \bool} \gg\!= \lambda e_1'.\;
				\sem{\ifkw \; e_1' \; \thenkw \; e_2 : \tau \; \elsekw \; e_3 : \tau : \tau} \in \mathcal{D}(Expr_{\tau} + \bot).
			\]
		\end{itemize}

		\textbf{Case} let $\letkw \; x = e_1 \; \inkw \; e_2 : \tau_2$. From typing we have:
		\[
			\Gamma \vdash e_1 : \tau_1 
			\qquad\text{and}\qquad 
			x : \tau_1 \vdash e_2 : \tau_2.
		\]
		We have the following possibilities:
		\begin{itemize}
			\item If $e_1 = v$ is a value, then by the substitution lemma:
			\[
				\delta_{e_2[v/x] : \tau_2} \in \mathcal{D}(Expr_{\tau_2} + \bot)
			\]
			\item If $e_1$ is an expression, then the clause:
			\[
				\sem{e_1 : \tau_1} \gg\!= \lambda e_1'.\sem{\letkw \; x = e_1' \; \inkw \; e_2 : \tau_2 : \tau_2}
			\]
			in the semantics is applied. By the IH,
			\[
				\sem{e_1 : \tau_1} \in \mathcal{D}(Expr_{\tau_1} + \bot).
			\]
			We use the left bind clause with 
			\[
				f(e_1') = \sem{\letkw \; x = e_1 \; \inkw \; e_2},
			\]
			which maps
			\[
				Expr_{\tau_1} \;\to\; \mathcal{D}(Expr_{\tau_2} + \bot)
			\]
			by Lemma~\ref{lem:typed-replacement}.

			By Proposition~\ref{prop:bind-typing}, we conclude
			\[
				\sem{e_1 : \tau_1} \gg\!= \lambda e_1'.\;
				\sem{\letkw \; x = e_1' \; \inkw \; e_2 : \tau_2 : \tau_2} \in \mathcal{D}(Expr_{\tau_2} + \bot).
			\]
		\end{itemize}

		\textbf{Case} uniform $\uniform(e_1,e_2) : \float$. 
		From typing we have 
		\[
		\emptyset \vdash e_1 : \float 
		\qquad\text{and}\qquad 
		\emptyset \vdash e_2 : \float.
		\]

		We have the following possibilities:
		\begin{itemize}
		\item If both $e_1 = v_1$ and $e_2 = v_2$ are values, Lemma~\ref{lem:canonical-forms} gives float constants, and either
		\[
			\text{Uniform}(v_1,v_2) \in \mathcal{D}(Expr_{\float} + \bot) 
			\quad\text{or}\quad
			\delta_{\bot} \in \mathcal{D}(Expr_{\float} + \bot)
		\]
		is returned.

		\item At least $e_1$ or $e_2$ is an expression. 
		Suppose $e_1$ is an expression, then the clause
		\[
			\sem{e_1 : \float} \gg\!= \lambda e_1'.\;
			\sem{\uniform(e_1',e_2 : \float) : \float}
		\]
		in the semantics is applied. By the IH,
		\[
			\sem{e_1 : \float} \in \mathcal{D}(Expr_{\float} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_1') = \sem{\uniform(e_1',e_2)},
		\]
		which maps
		\[
			Expr_{\float} \;\to\; \mathcal{D}(Expr_{\float} + \bot)
		\]
		by Lemma~\ref{lem:typed-replacement}.

		By Proposition~\ref{prop:bind-typing}, we conclude
		\[
			\sem{e_1 : \float} \gg\!= \lambda e_1'.\;
			\sem{\uniform(e_1',e_2 : \float) : \float : \float} \in \mathcal{D}(Expr_{\float} + \bot).
		\]

		Now suppose $e_2$ is an expression, then the clause
		\[
			\sem{e_2 : \float} \gg\!= \lambda e_2'.\;
			\sem{\uniform(e_1 : \float, e_2') : \float}
		\]
		in the semantics is applied. By the IH,
		\[
			\sem{e_2 : \float} \in \mathcal{D}(Expr_{\float} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_2') = \sem{\uniform(e_1,e_2')},
		\]
		which maps
		\[
			Expr_{\float} \;\to\; \mathcal{D}(Expr_{\float} + \bot)
		\]
		by Lemma~\ref{lem:typed-replacement}. 

		By Proposition~\ref{prop:bind-typing}, we conclude
		\[
			\sem{e_2 : \float} \gg\!= \lambda e_2'.\;
			\sem{\uniform(e_1 : \float,e_2') : \float} \in \mathcal{D}(Expr_{\float} + \bot).
		\]
		\end{itemize}

		\textbf{Case} discrete $\discrete(v_0, \ldots, v_n) : \tau$, where $\tau = \fin{n}$. It holds that either:
		\[
			\text{Discrete}(v_0,\ldots,v_n) \in \mathcal{D}(Expr_{\fin{n}} + \bot)
			\qquad\text{and}\qquad
			\delta_{\bot} \in \mathcal{D}(Expr_{\fin{n}} + \bot).
		\]
		Hence, for all well-typed expressions $e : \tau$, $\sem{e : \tau} \in \mathcal{D}(Expr_\tau + \bot)$.
\end{proof}

\subsubsection{Relation $\relation_{\tau}$}
We couple the continuous (and type-annotated) program $e$ appropriately with its discretized one $\discretize{e}$. Formally, the coupling relation is:

\[
\relation_{\tau}(e : \tau,e' : \tau') \iff e' : \tau' == \discretize{e : \tau} : \discretize{\tau}.
\]

In particular, we have the following:

\begin{align*}
	\{(\true,\true), (\false,\false)\} \in \relation_{\bool} \\
	\{(v,v) : v \text{ is a value}\} \in \relation_{\float} \\
\end{align*}

% What does this mean precisely? In particular, when the return type of $e$ is $\bool$, then the return type of $\discretize{e}$ is also $\bool$, and they have the same distribution. When the return type of $e$ is $\float$, then the return type of $\discretize{e}$ is also $\float$, and they have the same distribution.

		&\text{and hence, } B = \{\cleq{0.5}\}, V = \top, B_1 = \{\cleq{0}\}, V_1 = \{0\}, B_2 = \{\cleq{1}, \cleq{2}\}, V_2 = \{1,2\} \\[1cm]

		&= \sem{\uniform(0 : \tau_1, (\ifkw \; \uniform(0,1) < 0.5 \; \thenkw \; 1.0 \; \elsekw \; 2.0) : \tau_2) : \tau, \\
		&\letkw \; v_1' = \ifkw \; \discrete(0.5, 0.5) < 1 \; \thenkw \; 0 \; \elsekw \; 1 \; \inkw \\
		&\letkw \; v_2' = 0 \; \inkw \\
		&\ifkw \; v_1' == 0 \; \&\& \; v_2' == 0 \; \thenkw \\
		&\discrete(0.5,0.5) \\
		&\elsekw \\
		&\discrete(0.25,0.75)} \gg\!= \lambda (f,f').\sem{f < 0.5, f' \finlt{n+1} 1} \\[1cm]
		%%%
		&= \sem{\uniform(0 : \tau_1, (\ifkw \; \uniform(0,1) < 0.5 \; \thenkw \; 1.0 \; \elsekw \; 2.0) : \tau_2) : \tau, \\
		&\dc(0, \ifkw \; \discrete(0.5, 0.5) < 1 \; \thenkw \; 0 \; \elsekw \; 1)} \\
		&\gg\!= \lambda (f,f').\sem{f < 0.5, f' \finlt{n+1} 1} \\[1cm]
		%%%
		&= \sem{\ifkw \; \uniform(0,1) < 0.5 \; \thenkw \; 1.0 \; \elsekw \; 2.0 : \tau_2, \\
		&\ifkw \; \discrete(0.5, 0.5) < 1 \; \thenkw \; 0 \; \elsekw \; 1} \\
		&\gg\!= \lambda (f,f').\sem{\uniform(0 : \tau_1, f) : \tau, \dc(0, f')}\\[1cm]
		%%%
		&= (0.5 \cdot \delta_{(1.0,0)} + 0.5 \cdot \delta_{(2.0,1)}) \\
		&\gg\!= \lambda (f,f').\sem{\uniform(0 : \tau_1, f) : \tau, \dc(0, f')}\\[1cm]
		%%%
		&= 0.5 \cdot \sem{\uniform(0,1.0) : \tau, \dc(0,0)} \\
		&+ 0.5 \cdot \sem{\uniform(0,2.0) : \tau, \dc(0,1)} \\
		&\gg\!= \lambda (f,f').\sem{f < 0.5, f' \finlt{n+1} 1} \\[1cm]
		%%%
		&= 0.5 \cdot \sem{\uniform(0,1.0) : \tau, \discretize{\uniform(0,1) : \floattype{B}{\top}}} \\
		&+ 0.5 \cdot \sem{\uniform(0,2.0) : \tau, \discretize{\uniform(0,2) : \floattype{B}{\top}}} \\
		&\gg\!= \lambda (f,f').\sem{f < 0.5, f' \finlt{n+1} 1} \\
		&\text{since } \discretize{0 : \floattype{B_1}{0}} = 0, \discretize{1 : \floattype{B_2}{1}} = 0, \discretize{2 : \floattype{B_2}{2}} = 1 \\[1cm]
		%%%
		&= 0.5 \cdot \sem{\uniform(0,1.0) : \tau, \discrete(0.5, 0.5)} \\
		&+ 0.5 \cdot \sem{\uniform(0,2.0) : \tau, \discrete(0.25,0.75)} \\
		&\gg\!= \lambda (f,f').\sem{f < 0.5, f' \finlt{n+1} 1} \\[1cm]
		%%%
		&= 0.5 \cdot (\text{Uniform}(0,1) \gg\!= \lambda v.\delta_{(v,i)} \text{ s.t. } v \in \intervals{\{<0.5\}}_i) \\
		&+ 0.5 \cdot (\text{Uniform}(0,2) \gg\!= \lambda v.\delta_{(v,i)} \text{ s.t. } v \in \intervals{\{<0.5\}}_i) \\
		&\gg\!= \lambda (f,f').\sem{f < 0.5, f' \finlt{n+1} 1} \\[1cm]
		%%%
		&= 0.5 \cdot (0.5 \cdot \delta_{(\true,\true)} + 0.5 \cdot \delta_{(\false,\false)}) \\
		&+ 0.5 \cdot (0.25 \cdot \delta_{(\true,\true)} + 0.75 \cdot \delta_{(\false,\false)}) \\[1cm]
		&= 0.375 \cdot \delta_{(\true,\true)} + 0.625 \cdot \delta_{(\false,\false)} 
	\end{align*}
	}

\end{example}

\subsubsection{Lemmas for $e,e'$}

\begin{lemma}[Substitution for pairs]\label{lem:substitution-pairs}
	If $\Gamma, (x : \tau_1, x' : \tau_1') \vdash (e : \tau_2, e' : \tau_2')$ and $\Gamma \vdash (v : \tau_1, v' : \tau_1')$, then $\Gamma \vdash (e[v/x] : \tau_2, e'[v'/x'] : \tau_2')$.
\end{lemma}

\begin{proof}
	By induction on $(e,e')$. Apply the substitution lemma from~\ref{lem:substitution} on $e$ and $e'$.
\end{proof}

\begin{lemma}[Discretization Preserves Form]\label{lem:form-preservation}
	Let $e$ be a well-typed expression of type $\tau$. Then:
	\begin{enumerate}
		\item If $e : \tau$ is a value, then $\discretize{e : \tau}$ is a value.
		\item If $e : \tau$ is an expression, then $\discretize{e : \tau}$ is an expression.
	\end{enumerate}
\end{lemma}

\begin{proof}
	By a case analysis on the structure of $e$. 

	\textbf{Case} $e$ is a constant $v : \tau$. $e$ is a value. Via discretization,
	\[
		\discretize{v : \tau} = \begin{cases}
			v &\text{ if } B = \top \\
			\finconst{k}{n+1} \text{ s.t. } c \in \intervals{B}_k &\text{ if } B = \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\}
		\end{cases}
	\]
	both of which are values.
	
	\textbf{Case} $e$ is a variable $x : \tau$. Via discretization, \[
		\discretize{x : \tau} = x.
	\]
	$e$ can either be a value or an expression, but since the discretization leaves $x$ unchanged, the value or expression is preserved.

	\textbf{Case} $e$ is a let statement $\letkw \; x = e_1 : \tau_1 \; \inkw \; e_2 : \tau_2 : \tau_2$. $e$ is an expression. Via discretization,
	\[
	\discretize{\letkw \; x = e_1 : \tau_1 \; \inkw \; e_2 : \tau_2 : \tau_2} = 
	\letkw \; x = \discretize{e_1 : \tau_1} \; \inkw \; \discretize{e_2 : \tau_2},
	\]
	which is also an expression.

	\textbf{Case} $e$ is a comparison $e_1 : \tau_1 < e_2 : \tau_2 : \bool$, where $\tau_1 = \floattype{B_1}{V_1}$ and $\tau_2 = \floattype{B_2}{V_2}$. $e$ is an expression. Via discretization,
	\[
	\discretize{e_1 : \tau_1 < e_2 : \tau_2 : \bool} = \begin{cases}
		\discretize{e_1 : \tau_1} < \discretize{e_2 : \tau_2} &\text{ if } B = \top \\
		\discretize{e_1 : \tau_1} \finlt{n+1} \discretize{e_2 : \tau_2} &\text{ if } B = \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\} \\
	\end{cases}
	\]
	both of which are expressions.

	\textbf{Case} $e$ is a conditional $\ifkw \; e_1 : \bool \; \thenkw \; e_2 : \tau \; \elsekw \; e_3 : \tau : \tau$. $e$ is an expression. Via discretization,
	\[
		\discretize{\ifkw \; e_1 : \bool \; \thenkw \; e_2 : \tau \; \elsekw \; e_3 : \tau : \tau} = \ifkw \; \discretize{e_1 : \bool} \; \thenkw \; \discretize{e_2 : \tau} \; \elsekw \; \discretize{e_3 : \tau}
	\]
	which is also an expression.

	\textbf{Case} $e$ is a uniform distribution $\uniform(e_1 : \tau_1, e_2 : \tau_2) : \tau$, where $\tau_1 = \floattype{B}{V}$. $e$ is an expression. Via discretization,
	\begin{align*}
		&\discretize{\uniform(e_1 : \tau_1, e_2 : \tau_2) : \tau} \\
		&= \begin{cases}
			\uniform(\discretize{e_1 : \tau_1}, \discretize{e_2 : \tau_2}) &\text{ if } B = \top \\
			\discrete(p_0,\ldots,p_n) &\text{ if } B = \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\}, e_1,e_2 \in \R \\
			\dc(e_1 : \tau_1,e_2 : \tau_2) &\text{ if } B = \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\}, e_1,e_2 \notin \R
		\end{cases}
	\end{align*}
	all of which are expressions. In particular, \dc is a let statement.
\end{proof}


\begin{lemma}[Well-defined]
	For all well-typed expressions $(e : \tau, e' : \tau')$ such that $\relation_{\tau}(e,e')$, $\sem{(e : \tau,e' : \tau')} \in \mathcal{D}(\relation_{\tau} + \bot)$.
\end{lemma}

\begin{proof}
	By induction on the structure of the typing derivation $e : \tau$. We do a case analysis on the last rule used.

	\textbf{Case} constant $(v : \tau, \discretize{v : \tau} : \discretize{\tau}) \in \relation_{\tau}$, where $\tau = \floattype{B}{V}$. It holds that 
	\[
		\sem{v : \tau, \discretize{v : \tau} : \discretize{\tau}} = \delta_{(v : \tau, \discretize{v : \tau} : \discretize{\tau})} \in \mathcal{D}(\relation_{\tau} + \bot).
	\]

	\textbf{Case} comparison $(e_1 < e_2 : \bool, \discretize{e_1} < \discretize{e_2} : \bool) \in \relation_{\tau}$. We have the following possibilities:
	\begin{itemize}
		\item If $e_1 = v_1$ and $e_2 = v_2$ are values, then by Lemma~\ref{lem:form-preservation}, $\discretize{e_1} = v_1'$ and $\discretize{e_2} = v_2'$ are also values. Either:
		\[
			\delta_{(\true:\bool, \true:\bool)} \in \mathcal{D}(\relation_{\bool} + \bot) 
			\quad\text{or}\quad
			\delta_{(\false:\bool, \false:\bool)} \in \mathcal{D}(\relation_{\bool} + \bot) 
		\]
		\item At least $e_1$ or $e_2$ is an expression. Suppose $e_1$ is an expression; by Lemma~\ref{lem:form-preservation}, $\discretize{e_1}$ is also an expression. The clause
		\[
			\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} } \gg\!= \lambda (e_1',e_1''). \sem{ e_1' < e_2 : \tau_2, e_1'' < \discretize{e_2 : \tau_2} }
		\]
		in the semantics is applied. By the IH,
		\[
			\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} } \in \mathcal{D}(\relation_{\tau_1} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_1',e_1'') = \sem{ e_1' < e_2 : \tau_2, e_1'' < \discretize{e_2 : \tau_2} }
		\]
		which maps
		\[
			\relation_{\tau_1} \to \mathcal{D}(\relation_{\tau_2} + \bot).
		\]

		By Proposition~\ref{prop:bind-typing}, we conclude
		\[
			\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} } \gg\!= \lambda (f,f'). \sem{ f < e_2 : \tau_2, f' < \discretize{e_2 : \tau_2} } \in \mathcal{D}(\relation_{\tau_2} + \bot).
		\] 
	\end{itemize}
\end{proof}

\begin{definition}[Projection]
	The first and second projections $\pi_1(\mu)$ and $\pi_2(\mu)$ of a distribution $\mu$ over $A \times B$ are defined by $\pi_1(\mu)(a) = \sum_{b \in B} \mu(a,b)$ and $\pi_2(\mu)(n) = \sum_{b \in A} \mu(a,b)$.
\end{definition}
