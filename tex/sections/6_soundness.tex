\subsubsection{Brief recap of measure theory} 
We first give a brief overview of basic measure theory terms.

\paragraph{Topological space} A topological space is a pair $(X, \mathcal{T})$ where $\mathcal{T}$ is a collection of subsets of $X$, called open sets, such that $\emptyset, X \in \mathcal{T}$, and $\mathcal{T}$ is closed under arbitrary unions and finite intersections.

\paragraph{$\sigma$-algebras} A $\sigma$-algebra on a set $X$ is a collection $\Sigma$ of subsets of $X$ that includes $X$ itself and is closed under complement and under countable unions. By implication, a $\sigma$-algebra includes the empty set and is closed under countable intersections. 
Given a topological space $(X, \mathcal{T})$, the Borel $\sigma$-algebra $\mathcal{B}(X)$ is the smallest $\sigma$-algebra containing all open sets in $\mathcal{T}$.

\paragraph{Measurable spaces} A measurable space $(X, \Sigma)$ consists of an underlying set and its $\sigma$-algebra.  
When $\Sigma$ is the Borel $\sigma$-algebra on a topological space $X$, we call $(X, \Sigma)$ a Borel measurable space.

\paragraph{Measures} Let $(X,\Sigma)$ be a measurable space. A function $\mu : \Sigma \to \R^+$ is called a measure if it satisfies nonnegativity, null empty set, and countable additivity. 
When $\Sigma$ is a Borel $\sigma$-algebra, $\mu$ is called a Borel measure.

\paragraph{Measure spaces} The triple $(X, \Sigma, \mu)$ is called a measure space.
 When $\Sigma$ is the Borel $\sigma$-algebra and $\mu$ is the Borel measure, $(X, \Sigma, \mu)$ is called a Borel measure space.

\paragraph{(Sub)probability measures} When $\mu(X) \leq 1$, it is called a (Borel) subprobability measure. When $\mu(X) = 1$, it is called a (Borel) probability measure and $(X, \Sigma, \mu)$ a (Borel) probability space.

\paragraph{Product measure} The product of two measurable spaces $(X_1, \Sigma_1)$ and $(X_2, \Sigma_2)$ is also a measurable space, namely $(X_1 \times X_2, \Sigma_1 \otimes \Sigma_2)$.

\begin{definition}[Support]
	We define the support of a measure separately for discrete and continuous cases:
	\begin{enumerate}
		\item Let $(X, \Sigma, \mu)$ be a measure space where $\mu$ is a discrete measure. The support of $\mu$ is the set of all points with positive measure:
		\[
			{support}(\mu) \triangleq \{ x \in X \mid \mu(x) > 0 \}.
		\]

		\item Let $(X, \Sigma, \mu)$ be a Borel measure space where $X$ is a topological space and $\mu$ is a continuous measure. The support of $\mu$ is:
		\[
			{support}(\mu) \triangleq \{ x \in X \mid \forall U \in \Sigma, \; (U \text{ open and } x \in U) \implies \mu(U) > 0 \}.
		\]
	\end{enumerate}
\end{definition}

\begin{definition}[Projection with error]
	Let $(X + \bot, \Sigma_{X + \bot})$ and $(Y + \bot, \Sigma_{Y + \bot})$ be measurable spaces, with $(X, \Sigma_X)$ and $(Y, \Sigma_Y)$ measurable spaces as well. Let $\mu \in \mathcal{D}((X \times Y) + \bot)$. We define the first and second projection measures as:
	\begin{enumerate}
	\item $\pi_1(\mu)(A) := \mu(A \times Y)$ for every measurable $A \in \Sigma_X$, and $\pi_1(\mu)({\bot}) := \mu({\bot})$
	\item $\pi_2(\mu)(B) := \mu(X \times B)$ for every measurable $B \in \Sigma_Y$, and $\pi_2(\mu)({\bot}) := \mu({\bot})$
	\end{enumerate}
	Then, $\pi_1(\mu)$ is a measure on $(X + \bot, \Sigma_{X + \bot})$ and $\pi_2(\mu)$ is a measure on $(Y + \bot, \Sigma_{Y + \bot})$.
\end{definition}

\begin{definition}[Distribution Monad with Error]
	Define $\text{return} : X \to \mathcal{D}(X + \bot)$ by $\text{return}_X(x) := \delta_x$, where $\delta_x$ denotes the Dirac distribution on $x$: for any $y \in X$, we have $\delta_x(y) = 1$ if $y = x$, otherwise $\delta_x(y) = 0$. Further, define $\text{bind} : \mathcal{D}(X + \bot) \to (X \to \mathcal{D}(Y + \bot)) \to \mathcal{D}(Y + \bot)$ by:
	\begin{enumerate}
		\item $\text{bind}(\mu)(f)(y) := \sum_{x \in support(\mu)} \mu(x) \cdot f_{\bot}(x)(y)$ if $\mu$ is discrete
		\item $\text{bind}(\mu)(f)(A) := \int_{x \in support(\mu)} f_{\bot}(x)(A) \cdot d\mu(x)$ if $\mu$ is continuous
	\end{enumerate}
	where $f_{\bot}$ extends $f$ by $f_{\bot}(\bot) := \delta_{\bot}$ and $f_{\bot}(x) := f(x) \;\; \text{for every } x \in X.$
\end{definition}

\begin{lemma}[Bind-projection identity with error]\label{lem:bind-proj}
	Let $(X + \bot, \Sigma_{X + \bot})$ and $(Y + \bot, \Sigma_{Y + \bot})$ be measurable spaces, with $(X, \Sigma_X)$ and $(Y, \Sigma_Y)$ measurable spaces as well, and let $((X \times Y) + \bot, \Sigma_{(X \times Y) + \bot})$ be a product measurable space with error. Let
	\[
		\mu \in \mathcal{D}((X \times Y) + \bot)
	\]
	be a probability measure, and let 
	\[
		f_{\bot} : X \times Y \to \mathcal{D}((X \times Y) + \bot)
	\]
	be a kernel extended to handle $\bot$ via $f_\bot(\bot) := \delta_\bot$. Then,
	\[
		\pi_1(\mu \gg\!= f_{\bot}) = \pi_1(\mu) \gg\!= (x \mapsto \pi_1(f_{\bot}(x, \cdot))).
	\]
\end{lemma}

Informally, this lemma states that first marginal of a bind is the bind of the first marginal against the first marginal of the continuation.

\begin{proof}
	For any measurable set $A \subseteq X$,
	\begin{align*}
		&\pi_1(\mu \gg\!= f_{\bot})(A + \bot) \\
		&= (\mu \gg\!= f_{\bot})((A \times Y) + \bot) && \text{(by definition of $\pi_1$)} \\
		&= \sum \mu(x,y) f_{\bot}(x,y)((A \times Y) + \bot) \text{ or } \int f_{\bot}(x,y)((A \times Y) + \bot) \mu(d(x,y)) && \text{(by definition of bind)} \\
		&= \sum \mu(x,y) \pi_1(f_{\bot}(x,y))(A + \bot) \text{ or } \int \pi_1(f_{\bot}(x,y))(A + \bot) \mu(d(x,y)) && \text{(by definition of $\pi_1$)} \\
	\end{align*}
	
	For the discrete case:
	\begin{align*}
		&\sum_{(x,y) \in \text{supp}(\mu)} \mu(x,y) \pi_1(f_{\bot}(x,y))(A + \bot) \\
		&= \sum_{x \in \pi_1(\text{supp}(\mu))} \left[ \sum_{y \in \pi_2(\text{supp}(\mu))} \mu(x,y) \pi_1(f_{\bot}(x,y))(A + \bot) \right] && \text{(rearrange sum)} \\
		&= \sum_{x \in \pi_1(\text{supp}(\mu))} \pi_1(f_{\bot}(x,y))(A + \bot) \left[ \sum_{y \in \pi_2(\text{supp}(\mu))} \mu(x,y) \right] && \text{(factor constant in $y$)} \\
		&= \sum_{x \in \pi_1(\text{supp}(\mu))} \pi_1(f_{\bot}(x,y))(A + \bot) (\pi_1 \mu)(x,y) && \text{(definition of $\pi_1$)} \\
		&= (\pi_1 \mu) \gg\!= (x \mapsto \pi_1(f_{\bot}(x,y))) && \text{(definition of bind)}
	\end{align*}

	For the continuous case: 
	\begin{align*}
		&\int_{(x,y) \in \text{supp}(\mu)} \pi_1(f_{\bot}(x,y))(A + \bot) \mu(d(x,y)) \\
		&= \int_{x \in \pi_1(\text{supp}(\mu))} \left[ \int_{y \in \pi_2(\text{supp}(\mu))} \pi_1(f_{\bot}(x,y))(A + \bot) \mu(d(x,y)) \right] && \text{(by Fubini's theorem)} \\
		&= \int_{x \in \pi_1(\text{supp}(\mu))} \pi_1(f_{\bot}(x,y))(A + \bot) \left[ \int_{y \in \pi_2(\text{supp}(\mu))} \mu(d(x,y)) \right] && \text{(factor constant in $y$)} \\
		&= \int_{x \in \pi_1(\text{supp}(\mu))} \pi_1(f_{\bot}(x,y))(A + \bot) (\pi_1 \mu)(d(x,y)) && \text{(definition of $\pi_1$)} \\
		&= (\pi_1 \mu) \gg\!= (x \mapsto \pi_1(f_{\bot}(x,y))) && \text{(definition of bind)}
	\end{align*}
\end{proof}

\subsection{Measure space on expressions}
To make the probabilistic semantics precise, we define an appropriate measure space structure on the set of expressions. We construct a Borel measurable space on $Expr_{\tau}$ by factoring out real numbers using the notion of what we call \emph{expression skeletons}.

\begin{definition}[Expression Skeleton]
	An expression skeleton $s : \tau$ is a well-typed expression $e : \tau$ where each occurrence of a float constant is replaced by a hole $\square$, generated as follows:
	\begin{align*}
		s ::= \; 
		&\square \mid x \mid \true \mid \false \mid \finconst{k}{n} \mid \letkw \; x = s_1 \; \inkw \; s_2 \mid \ifkw \; s_1 \; \thenkw \; s_2 \; \elsekw \; s_3 \mid s_1 < s_2 \mid \uniform(s_1, s_2)
	\end{align*}
	with the understanding that each hole $\square$ has type $\floattype{B}{V}$, and that all other types appearing in the original expression are preserved in its corresponding skeleton.
	
	We denote by $\skel_{\tau}$ the set of all expression skeletons, and by $\holes(s)$ the number of distinct holes in skeleton $s$.
\end{definition}

\begin{example}
	The expression $\uniform(0.0, 1.0) < 0.5$ has the skeleton $\uniform(\square, \square) < \square$ with three holes. 
\end{example}

\begin{lemma}[Skel is countable]\label{lem:skel-countable}
	The set $\skel_{\tau}$ is countable.
\end{lemma}

\begin{proof}
	Each skeleton is a finite syntactic tree where:
	\begin{itemize}
	\item Internal nodes are labeled by constructors from a finite set of ${\letkw, \ifkw, <, \uniform}$
	\item Leaves are labeled by literals or variables (countable) or holes $\square$ (countable)
	\end{itemize}
	The set of all finite trees over countable labels is countable.
\end{proof}

This bijection holds because every well-typed expression can be uniquely decomposed into its skeleton (obtained by replacing constants with holes) and the corresponding tuple of real values filling those holes.

\begin{definition}[Measurable space on well-typed expressions]
	For a given $s : \tau \in \skel_{\tau}$, define the subset:
	\[
		A_{s : \tau} := \{f(e : \tau)_2 \mid f(e : \tau)_1 = s : \tau \text{ and } e : \tau \in Expr_{\tau}\}.
	\]
	Let $s[\vec{r}] : \tau$ denote the (unique) well-typed expression of type $\tau$ obtained by filling the holes of $s$ with the coordinates of $\vec{r}$. We then define the measurable space of well-typed expressions as the union:
	\[
		Expr_{\tau} := \bigcup_{s : \tau \in \skel_{\tau}} \{s[\vec{r}] : \tau \mid \vec{r} \in A_{s : \tau}\}.
	\]
\end{definition}

\begin{example}
	Given $s : \tau = (\square : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\top} < \square : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\{3.5,4.5\}}) : \bool$, we compute $A_{s : \tau}$.

	\begin{align*}
		f(e : \bool)_1 
			&= (\square : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\top} 
				< \square : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\{3.5,4.5\}}) : \bool \\[6pt]
		\iff\;& 
			e = e_1 < e_2 \\[3pt]
		&\quad 
			f(e_1 : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\top})_1 
			= \square : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\top}, \\[3pt]
		&\quad 
			f(e_2 : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\{3.5,4.5\}})_1 
				= \square : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\{3.5,4.5\}} \\[6pt]
		\iff\;& 
			e_1 = c \in \R, \quad e_2 \in \{3.5, 4.5\}.
	\end{align*}
	
	Thus,
	\begin{align*}
		f(c : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\top})_2 &= \R, \\[3pt]
		f(3.5 : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\{3.5,4.5\}})_2 &= \{3.5\}, \\[3pt]
		f(4.5 : \floattype{\{\clt{3.5}, \clt{4.5}\}}{\{3.5,4.5\}})_2 &= \{4.5\}.
	\end{align*}
	
	Hence,
	\begin{align*}
		A_{s : \tau} 
			&= \{(x, 3.5) \mid x \in \R\} \cup \{(x, 4.5) \mid x \in \R\} \\[3pt]
			&= \R \times \{3.5, 4.5\} \subseteq \R^2.
	\end{align*}	
\end{example}

\begin{lemma}
	$(Expr_{\tau}, \mathcal{F}_{Expr_{\tau}})$ is a Borel measurable space.
\end{lemma}

\begin{proof}
	Follows by structural recursion on expressions. 
	
	Fix an $s \in \skel$. At the base cases, we either get a finite value set $V$ (a finite union of Borel singletons, which is Borel), $V = \top$ (the entire $\R$, which is Borel), or $\emptyset$ (Borel). Each construct combines its subexpressions through Cartesian products (Borel sets are closed under finite products).

	Since $\skel$ is countable by Lemma~\ref{lem:skel-countable}, and since Borel sets are closed under countable unions, $Expr_{\tau}$ forms a Borel measurable space.
\end{proof}

\end{lemma}

\begin{proof}
	Given an expression skeleton $s \in \skel$ with $n = \holes(s)$ where $s[\vec{r}] : \tau$, $A_{s,\tau}$ is a Borel subset of $\R^n$. Since $\skel$ is countable, the countable union of Borel measurable spaces remains a Borel measurable space.
\end{proof}


\begin{definition}[Extension with error element]\label{def:meas-space-with-error}
	We extend the Borel measurable space of expressions to include a distinguished error element $\bot$, defining the measurable space of extended expressions as
	\[
		Expr_{\tau} + \bot := Expr_{\tau} \cup \{\bot\},
	\]
	with the topology $\mathcal{T}_{Expr_{\tau} + \bot} = \{U, U \cup \{\bot\} \mid U \in \mathcal{T}_{Expr_{\tau}}\}$, equipped with the Borel $\sigma$-algebra $\mathcal{F}_{Expr_{\tau} + \bot} := \sigma(\mathcal{T}_{Expr_{\tau} + \bot})$.
\end{definition}

\begin{lemma}
	$(Expr_{\tau} + \bot, \mathcal{F}_{Expr_{\tau} + \bot})$ is a Borel measurable space.
\end{lemma}

\begin{proof}
	$Expr_{\tau}$ is itself a Borel set, and $\{\bot\}$ is an open singleton in the extended topology. The union of a Borel set with an open set is Borel. 
\end{proof}

We can equip this measurable space with a (sub)probability measure such that $\mu(Expr_{\tau} + \bot) \leq 1$. And in particular, any (sub)probability measure $\mu$ on $\R^{\holes(s)}$ for a skeleton $s$ can be lifted to a (sub)probability measure on $Expr_{\tau} + \bot$, as follows:

	\mu_{Expr_{\tau} + \bot}(A) = \mu(\{\vec{r} \in \R^{\holes(s)} \mid s[\vec{r}] \in A\}).

\begin{lemma}
	Let $(Expr_{\tau} + \bot, \mathcal{F}_{Expr_{\tau} + \bot})$ be the measurable space of expressions with error from Definition~\ref{def:meas-space-with-error}. Let $V \subset Expr_{\tau}$ be the set of all value expressions. Let $N \subset Expr_{\tau}$, defined as $N = Expr_{\tau} \setminus V$, be the set of non-value expressions. Let $\{\bot\}$ be the singleton set containing the error element $\bot$. The following are true:
	\begin{enumerate}
		\item The set $\vals$ is measurable.
		\item The set $\{\bot\}$ is measurable.
		\item The set $\nonvals$ is measurable.
	\end{enumerate}
\end{lemma}

\begin{proof}
	We prove the measurability of $\vals$, $\{\bot\}$, and $N$ in separate parts.
	\begin{enumerate}
		\item \textbf{(Measurability of $\vals$):} By the construction of $\mathcal{F}_{Expr_{\tau}}$, for each skeleton $s$ that represents a value, the set $\{\vec{r} \in \R^{\holes(s)} \mid \vdash s[\vec{r}] : \tau\}$ is a measurable set in the union. The set $V$ is the countable union of all such sets corresponding to value skeletons. Since $\skel$ is countable, $V$ is a countable union of measurable sets and is therefore itself measurable, e.g. $V \in \mathcal{F}_{Expr_{\tau}}$. As $\mathcal{F}_{Expr_{\tau}} \subset \mathcal{F}_{Expr_{\tau} + \bot}$, it follows that $V \in \mathcal{F}_{Expr_{\tau} + \bot}$.
		
		\item \textbf{(Measurability of $\{\bot\}$):} By Definition~\ref{def:meas-space-with-error}, the topology $\mathcal{T}_{Expr_{\tau} + \bot}$ contains all sets of the form $U \cup \{\bot\}$, where $U \in \mathcal{T}_{Expr_{\tau}}$. In particular, taking $U = \emptyset$ gives $\emptyset \cup \{\bot\} = \{\bot\} \in \mathcal{T}_{Expr_{\tau} + \bot}$. Since the Borel $\sigma$-algebra $\mathcal{F}_{Expr_{\tau} + \bot}$ is generated by $\mathcal{T}_{Expr_{\tau} + \bot}$, it follows that $\{\bot\} \in \mathcal{F}_{Expr_{\tau} + \bot}$.

		\item \textbf{(Measurablity of $N$):} By definition, $N$ is the set of non-value expressions, e.g. $N = Expr_{\tau} \setminus V$. From part i), we have established that $V \in \mathcal{F}_{Expr_{\tau}}$. Since $\mathcal{F}_{Expr_{\tau}}$ is a $\sigma$-algebra and is closed under complement, the complement of $V$ within $Expr_{\tau}$, which is precisely $N$, is also measurable, e.g. $N \in \mathcal{F}_{Expr_{\tau}}$. As $\mathcal{F}_{Expr_{\tau}} \subset \mathcal{F}_{Expr_{\tau} + \bot}$, it follows that $N \in \mathcal{F}_{Expr_{\tau} + \bot}$.
	\end{enumerate}
\end{proof}

Note that in general, given $(X, \Sigma_X)$ and $(Y, \Sigma_Y)$ to be measurable spaces, $((X \times Y) + \bot)$ forms a measurable space, since the product $(X \times Y)$ is measurable and for any measurable space $(Z, \Sigma_Z)$, the space $(Z + \bot, \Sigma_{Z + \bot})$ with $\Sigma_{Z + \bot}$ generated by $\Sigma_Z \cup {{\bot}}$ is measurable.

\subsubsection{\dc}
To model the discretization behavior when $B \neq \top$ for the uniform distribution in the discretization function, we introduce a new construct called \dc. We give the following definition:

\begin{definition}[Discretization of uniform when $B \neq \top$]
	Let $B \neq \top$, $B_1 \neq \top$, and $B_2 \neq \top$ be finite cut sets. Let $V_1 = \{v_1, \ldots, v_n\} \neq \top$, and $V_2 = \{w_1, \ldots, w_m\} \neq \top$ be finite value sets. Then, the discretization of the uniform distribution can be given as:

	\begin{align*}
		&\discretize{\uniform(e_1 : \floattype{B_1}{V_1}, e_2 : \floattype{B_2}{V_2}) : \floattype{B}{V}} \\
		&= \dc(\discretize{e_1 : \floattype{B_1}{V_1}},\discretize{e_2 : \floattype{B_2}{V_2}}).
	\end{align*}

	where $\dc$ implements the behavior of $\discretize{\uniform(e_1 : \floattype{B_1}{V_1}, e_2 : \floattype{B_2}{V_2}) : \floattype{B}{V}}$ defined in the discretization function.
\end{definition}

We give a typing rule for \dc:
\begin{align*}
	\inferrule[\textsc{TDiscreteCase}]
	{e_1 : \fin{|B_1| + 1} \\
	e_2 : \fin{|B_2| + 1}}
	{\dc(e_1, e_2) : \fin{|B|+1}}
\end{align*}

And the semantics is as follows:
{
\begin{align*}
	&\sem{ \dc(e_1 : \tau_1, e_2 : \tau_2) : \tau} \\
	&= \begin{cases}
		\sem{ \discretize{\uniform(v_i, w_j) : \floattype{B}{\top}} : \tau } \quad &\text{ if } e_1 = \discretize{v_i : \floattype{B_1}{\{v_i\}}} = v', \\
		&e_2 = \discretize{w_j : \floattype{B_2}{\{w_j\}}} = w' \\
		\sem{ e_2 : \tau_2 } \gg\!= \lambda f. \sem{ \dc(v' : \tau_1, f) } \quad &\text{ if } e_1 = v', e_2 \neq w' \\
		\sem{ e_1 : \tau_1 } \gg\!= \lambda f. \sem{ \dc(f, e_2 : \tau_2) } \quad &\text{ otherwise } \\
	\end{cases} \\
	&\text{where } \\
	&\tau = \fin{|B| + 1}, \\
	&\tau_1 = \fin{|B_1| + 1}, \tau_2 = \fin{|B_2| + 1} \\
	&V_1 = \{v_1,\ldots,v_n\}, V_2 = \{w_1,\ldots,w_m\} \\
\end{align*}
}

\subsubsection{Lemmas for $e$}

\begin{lemma}[Substitution]\label{lem:substitution}
	If $\Gamma, x : \tau_1 \vdash e : \tau_2$ and $\emptyset \vdash v : \tau_1$, then $\Gamma \vdash e[v/x] : \tau_2$. 
\end{lemma}

\begin{proof}
	We proceed by induction on $e$:

	\textbf{Case} $e$ is a variable $y$. There are two subcases. 

	\begin{itemize}
		\item If $y = x$, then
		\[
			e[v/x] = v.
		\]
		We know
		\[
			\Gamma, x : \tau_1 \vdash x : \tau_1,
		\]
		so $\tau_2 = \tau_1$. Since $\emptyset \vdash v : \tau_1$, we also have
		\[
			\emptyset \vdash v : \tau_2.
		\]
		Hence
		\[
			\Gamma \vdash e[v/x] : \tau_2.
		\]

		\item If $y \neq x$, then the substitution does nothing:
		\[
			e[v/x] = y.
		\]
		Since $\Gamma, x : \tau_1 \vdash y : \tau_2$ and $y \neq x$, it follows that $y$ must be bound in $\Gamma$, and therefore
		\[
			\Gamma \vdash y : \tau_2.
		\]
		Thus,
		\[
			\Gamma \vdash e[v/x] : \tau_2.
		\]
	\end{itemize}

	\textbf{Case} $e$ is a constant or a discrete distribution $\discrete(v_0,\ldots,v_n)$. Substitution does not change either:
	\[
		e[v/x] = e.
	\]
	Typing of constants or discrete does not depend on $x$, so
	\[
		\Gamma \vdash e : \tau_2.
	\]
	Hence
	\[
		\Gamma \vdash e[v/x] : \tau_2.
	\]

	\textbf{Case} $e$ is a comparison $e_1 < e_2$. By \textsc{Less}, we have:
	\[
		\inferrule
		{\ \Gamma, x : \tau_1 \vdash e_1 : \tau \quad \Gamma, x : \tau_1 \vdash e_2 : \tau}
		{\Gamma, x : \tau_1 \vdash e_1 < e_2 : \bool}
	\]
	By the induction hypothesis,
	\[
		\Gamma \vdash e_1[v/x] : \tau
		\quad\text{and}\quad
		\Gamma \vdash e_2[v/x] : \tau.
	\]
	Given these, \textsc{Less} lets us derive:
	\[
		\Gamma \vdash e_1[v/x] < e_2[v/x] : \bool,
	\]
	and thus:
	\[
		\Gamma \vdash (e_1 < e_2)[v/x] : \bool
	\]
	since $(e_1 < e_2)[v/x] = e_1[v/x] < e_2[v/x]$ by the definition of substitution.

	\textbf{Case} $e$ is an if-else $\ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3$. By \textsc{If}, we have:
	\[
		\inferrule
		{\ \Gamma, x : \tau_1 \vdash e_1 : \bool \quad \Gamma, x : \tau_1 \vdash e_2 : \tau \quad \Gamma, x : \tau_1 \vdash e_3 : \tau}
		{\Gamma, x : \tau_1 \vdash \ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3 : \tau}
	\]
	By the induction hypothesis,
	\[
		\Gamma \vdash e_1[v/x] : \bool,
		\quad
		\Gamma \vdash e_2[v/x] : \tau,
		\quad
		\Gamma \vdash e_3[v/x] : \tau.
	\]
	Given these, \textsc{If} lets us derive:
	\[
		\Gamma \vdash \ifkw \; e_1[v/x] \; \thenkw \; e_2[v/x] \; \elsekw \; e_3[v/x] : \tau,
	\]
	and thus:
	\[
		\Gamma \vdash (\ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3)[v/x] : \tau
	\]
	since $(\ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3)[v/x] = \ifkw \; e_1[v/x] \; \thenkw \; e_2[v/x] \; \elsekw \; e_3[v/x]$ by the definition of substitution.

	\textbf{Case} $e$ is a let statement $\letkw \; y = e_1 \; \inkw \; e_2$. We distinguish two subcases.

	\begin{itemize}
		\item If $y \neq x$, then by \textsc{Let} we have:
		\[
			\inferrule
			{\ \Gamma, x : \tau_1 \vdash e_1 : \tau_1' \quad \Gamma, x : \tau_1, y : \tau_1' \vdash e_2 : \tau_2'}
			{\Gamma, x : \tau_1 \vdash \letkw \; y = e_1 \; \inkw \; e_2 : \tau_2'}
		\]
		By the induction hypothesis,
		\[
			\Gamma \vdash e_1[v/x] : \tau_1'
			\quad\text{and}\quad
			\Gamma, y : \tau_1' \vdash e_2[v/x] : \tau_2'.
		\]
		Given these, \textsc{Let} lets us derive:
		\[
			\Gamma \vdash \letkw \; y = e_1[v/x] \; \inkw \; e_2[v/x] : \tau_2',
		\]
		and thus:
		\[
			\Gamma \vdash (\letkw \; y = e_1 \; \inkw \; e_2)[v/x] : \tau_2'
		\]
		since $(\letkw \; y = e_1 \; \inkw \; e_2)[v/x] = \letkw \; y = e_1[v/x] \; \inkw \; e_2[v/x]$.

		\item If $y = x$, then by the typing rule:
		\[
			\inferrule
			{\ \Gamma, x : \tau_1 \vdash e_1 : \tau_1' \quad \Gamma, x : \tau_1 \vdash e_2 : \tau_2'}
			{\Gamma, x : \tau_1 \vdash \letkw \; x = e_1 \; \inkw \; e_2 : \tau_2'}
		\]
		By the induction hypothesis,
		\[
			\Gamma \vdash e_1[v/x] : \tau_1'.
		\]
		Given this, \textsc{Let} lets us derive:
		\[
			\Gamma \vdash \letkw \; x = e_1[v/x] \; \inkw \; e_2 : \tau_2',
		\]
		and thus:
		\[
			\Gamma \vdash (\letkw \; x = e_1 \; \inkw \; e_2)[v/x] : \tau_2'
		\]
		since $(\letkw \; x = e_1 \; \inkw \; e_2)[v/x] = \letkw \; x = e_1[v/x] \; \inkw \; e_2$ by capture-avoiding substitution.
	\end{itemize}

	\textbf{Case} $e$ is a uniform distribution $\uniform(e_1,e_2)$. By \textsc{Continuous-2}, we have:
	\[
		\inferrule
		{\ \Gamma, x : \tau_1 \vdash e_1 : \tau \quad \Gamma, x : \tau_1 \vdash e_2 : \tau}
		{\Gamma, x : \tau_1 \vdash \uniform(e_1,e_2) : \tau}
	\]
	By the induction hypothesis,
	\[
		\Gamma \vdash e_1[v/x] : \tau
		\quad\text{and}\quad
		\Gamma \vdash e_2[v/x] : \tau.
	\]
	Given these, \textsc{Continuous-2} lets us derive:
	\[
		\Gamma \vdash \uniform(e_1[v/x], e_2[v/x]) : \tau,
	\]
	and thus:
	\[
		\Gamma \vdash \uniform(e_1,e_2)[v/x] : \tau
	\]
	since $\uniform(e_1,e_2)[v/x] = \uniform(e_1[v/x], e_2[v/x])$ by the definition of substitution.
\end{proof}

\begin{lemma}[Canonical forms]\label{lem:canonical-forms}

\begin{lemma}[Typed replacement]\label{lem:typed-replacement}
	All under empty context.
	\begin{enumerate}
		\item If $\emptyset \vdash \ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3 : \tau$ and $\emptyset \vdash e_1' : \bool$, then $\emptyset \vdash \ifkw \; e_1' \; \thenkw \; e_2 \; \elsekw \; e_3 : \tau$.
		
		\item If $\emptyset \vdash e_1 < e_2 : \bool$ and $\emptyset \vdash e_1' : \floattype{B}{V}$, then $\emptyset \vdash e_1' < e_2 : \bool$. Likewise, if $\emptyset \vdash e_1 < e_2 : \bool$ and $\emptyset \vdash e_2' : \floattype{B}{V}$, then $\emptyset \vdash e_1 < e_2' : \bool$.
		
		\item If $\emptyset \vdash e_1 \finlt{n} e_2 : \bool$ and $\emptyset \vdash e_1' : \fin{n}$, then $\emptyset \vdash e_1' < e_2 : \bool$. Likewise, if $\emptyset \vdash e_1 \finlt{n} e_2 : \bool$ and $\emptyset \vdash e_2' : \fin{n}$, then $\emptyset \vdash e_1 < e_2' : \bool$.
		
		\item If $\emptyset \vdash \uniform(e_1,e_2) : \floattype{B}{V}$ and $\emptyset \vdash e_1' : \floattype{B}{V}$, then $\emptyset \vdash \uniform(e_1',e_2) : \floattype{B}{V}$. Likewise, if $\emptyset \vdash \uniform(e_1,e_2) : \floattype{B}{V}$ and $\emptyset \vdash e_2' : \floattype{B}{V}$, then $\emptyset \vdash \uniform(e_1,e_2') : \floattype{B}{V}$.
		
		\item If $\emptyset \vdash \letkw \; x = e_1 \; \inkw \; e_2 : \tau_2$ and $\emptyset \vdash e_1' : \tau_1$, then $\emptyset \vdash \letkw \; x = e_1' \; \inkw \; e_2 : \tau_2$.
		
		\item If $\emptyset \vdash \dc(e_1,e_2) : \fin{|B|+1}$ and $\emptyset \vdash e_1' : \fin{|B_1|+1}$, then $\emptyset \vdash \dc(e_1',e_2) : \fin{|B|+1}$. Likewise, if $\emptyset \vdash \dc(e_1,e_2) : \fin{|B|+1}$ and $\emptyset \vdash e_2' : \fin{|B_2|+1}$, then $\emptyset \vdash \dc(e_1,e_2') : \fin{|B|+1}$.

\begin{proposition}[Bind typing principle]\label{prop:bind-typing}
	Let $A,B$ be measurable spaces. If $\mu \in \mathcal{D}(A)$ and $f : A \to \mathcal{D}(B)$, then $\mu \gg\!= f \in \mathcal{D}(B)$. In particular, if $f(a) = \delta_{b_a}$ with $b_a \in B$ for all $a \in A$, then $\mu \gg\!= f \in \mathcal{D}(B)$.
\end{proposition}

	
\begin{lemma}[Well-Defined]\label{lem:well-defined}
	For all well-typed expressions $e : \tau$, $\sem{e : \tau} \in \mathcal{D}(Expr_{\tau} + \bot)$.
\end{lemma}

\begin{proof}
	By induction on the structure of the typing derivation $e : \tau$. We do a case analysis on the last rule used.

		\textbf{Case} float constant $v : \floattype{B}{V}$. From the semantics, it holds trivially
		\[
			\sem{v : \floattype{B}{V}} = \delta_{v : \floattype{B}{V}} \in \mathcal{D}(Expr_{\floattype{B}{V}} + \bot).
		\]

		\textbf{Case} finite constant $\finconst{k}{n} : \fin{n}$. Analogous to float constant.

		\textbf{Case} float less than $e_1 < e_2 : \bool$. 
		From typing we have 
		\[
		\emptyset \vdash e_1 : \floattype{B}{V_1} 
		\qquad\text{and}\qquad 
		\emptyset \vdash e_2 : \floattype{B}{V_2}.
		\]

		We have the following possibilities:
		\begin{itemize}
		\item If both $e_1 = v_1$ and $e_2 = v_2$ are values, Lemma~\ref{lem:canonical-forms} gives float constants, and either
		\[
			\delta_{\true:\bool} \in \mathcal{D}(Expr_{\bool} + \bot) 
			\quad\text{or}\quad
			\delta_{\false:\bool} \in \mathcal{D}(Expr_{\bool} + \bot)
		\]
		is returned.

		\item At least $e_1$ or $e_2$ is an expression. 
		Suppose $e_1$ is an expression, then the clause
		\[
			\sem{e_1 : \floattype{B}{V_1}} \gg\!= \lambda e_1'.\;
			\sem{e_1' < e_2 : \floattype{B}{V_2} : \bool}
		\]
		in the semantics is applied. By the IH,
		\[
			\sem{e_1 : \floattype{B}{V_1}} \in \mathcal{D}(Expr_{\floattype{B}{V_1}} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_1') = \sem{e_1' < e_2},
		\]
		which maps
		\[
			Expr_{\floattype{B}{V_1}} \;\to\; \mathcal{D}(Expr_{\bool} + \bot)
		\]
		by Lemma~\ref{lem:typed-replacement}.

		By Proposition~\ref{prop:bind-typing}, we conclude
		\[
			\sem{e_1 : \floattype{B}{V_1}} \gg\!= \lambda e_1'.\;
			\sem{e_1' < e_2 : \floattype{B}{V_2} : \bool} \in \mathcal{D}(Expr_{\bool} + \bot).
		\]

		Now suppose $e_2$ is an expression, then the clause
		\[
			\sem{e_2 : \floattype{B}{V_2}} \gg\!= \lambda e_2'.\;
			\sem{e_1 : \floattype{B}{V_1} < e_2' : \bool}
		\]
		in the semantics is applied. By the IH,
		\[
			\sem{e_2 : \floattype{B}{V_2}} \in \mathcal{D}(Expr_{\floattype{B}{V_2}} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_2') = \sem{e_1 < e_2'},
		\]
		which maps
		\[
			Expr_{\floattype{B}{V_2}} \;\to\; \mathcal{D}(Expr_{\bool} + \bot)
		\]
		by Lemma~\ref{lem:typed-replacement}. 

		By Proposition~\ref{prop:bind-typing}, we conclude
		\[
			\sem{e_2 : \floattype{B}{V_2}} \gg\!= \lambda e_2'.\;
			\sem{e_1 : \floattype{B}{V_1} < e_2' : \bool} \in \mathcal{D}(Expr_{\bool} + \bot).
		\]
		\end{itemize}

		\textbf{Case} finite less than $e_1 \finlt{n} e_2 : \bool$. Analogous to float less than.

		\textbf{Case} if-else $\ifkw \; e_1 \; \thenkw \; e_2 \; \elsekw \; e_3 : \tau$. From typing we have:
		\[
		\emptyset \vdash e_1 : \bool 
		\qquad\text{and}\qquad 
		\emptyset \vdash e_2 : \tau
		\qquad\text{and}\qquad 
		\emptyset \vdash e_3 : \tau.
		\]
		We have the following possibilities:
		\begin{itemize}
			\item If $e_1 = v$ is a value, Lemma~\ref{lem:canonical-forms} gives true or false, and either
			\[
				\delta_{e_2:\tau} \in \mathcal{D}(Expr_{\tau} + \bot) 
				\quad\text{or}\quad
				\delta_{e_3:\tau} \in \mathcal{D}(Expr_{\tau} + \bot)
			\]
			is returned.

			\item $e_1$ is an expression. Then the clause
			\[
				\sem{e_1 : \bool} \gg\!= \lambda e_1'.\;
				\sem{\ifkw \; e_1' \; \thenkw \; e_2 : \tau \; \elsekw \; e_3 : \tau : \tau}
			\]
			in the semantics is applied. By the IH,
			\[
				\sem{e_1 : \bool} \in \mathcal{D}(Expr_{\bool} + \bot).
			\]
			We use the left bind clause with 
			\[
				f(e_1') = \sem{\ifkw \; e_1' \; \thenkw \; e_2 \; \elsekw \; e_3},
			\]
			which maps
			\[
				Expr_{\bool} \;\to\; \mathcal{D}(Expr_{\tau} + \bot)
			\]
			by Lemma~\ref{lem:typed-replacement}.

			By Proposition~\ref{prop:bind-typing}, we conclude
			\[
				\sem{e_1 : \bool} \gg\!= \lambda e_1'.\;
				\sem{\ifkw \; e_1' \; \thenkw \; e_2 : \tau \; \elsekw \; e_3 : \tau : \tau} \in \mathcal{D}(Expr_{\tau} + \bot).
			\]
		\end{itemize}

		\textbf{Case} let $\letkw \; x = e_1 \; \inkw \; e_2 : \tau_2$. From typing we have:
		\[
			\Gamma \vdash e_1 : \tau_1 
			\qquad\text{and}\qquad 
			x : \tau_1 \vdash e_2 : \tau_2.
		\]
		We have the following possibilities:
		\begin{itemize}
			\item If $e_1 = v$ is a value, then by the substitution lemma:
			\[
				\delta_{e_2[v/x] : \tau_2} \in \mathcal{D}(Expr_{\tau_2} + \bot)
			\]
			\item If $e_1$ is an expression, then the clause:
			\[
				\sem{e_1 : \tau_1} \gg\!= \lambda e_1'.\sem{\letkw \; x = e_1' \; \inkw \; e_2 : \tau_2 : \tau_2}
			\]
			in the semantics is applied. By the IH,
			\[
				\sem{e_1 : \tau_1} \in \mathcal{D}(Expr_{\tau_1} + \bot).
			\]
			We use the left bind clause with 
			\[
				f(e_1') = \sem{\letkw \; x = e_1 \; \inkw \; e_2},
			\]
			which maps
			\[
				Expr_{\tau_1} \;\to\; \mathcal{D}(Expr_{\tau_2} + \bot)
			\]
			by Lemma~\ref{lem:typed-replacement}.

			By Proposition~\ref{prop:bind-typing}, we conclude
			\[
				\sem{e_1 : \tau_1} \gg\!= \lambda e_1'.\;
				\sem{\letkw \; x = e_1' \; \inkw \; e_2 : \tau_2 : \tau_2} \in \mathcal{D}(Expr_{\tau_2} + \bot).
			\]
		\end{itemize}

		\textbf{Case} uniform $\uniform(e_1,e_2) : \floattype{B}{V}$. 
		From typing we have 
		\[
		\emptyset \vdash e_1 : \floattype{B_1}{V_1} 
		\qquad\text{and}\qquad 
		\emptyset \vdash e_2 : \floattype{B_2}{V_2}.
		\]

		We have the following possibilities:
		\begin{itemize}
		\item If both $e_1 = v_1$ and $e_2 = v_2$ are values, Lemma~\ref{lem:canonical-forms} gives float constants, and either
		\[
			\text{Uniform}(v_1,v_2) \in \mathcal{D}(Expr_{\floattype{B}{V}} + \bot) 
			\quad\text{or}\quad
			\delta_{\bot} \in \mathcal{D}(Expr_{\floattype{B}{V}} + \bot)
		\]
		is returned.

		\item At least $e_1$ or $e_2$ is an expression. 
		Suppose $e_1$ is an expression, then the clause
		\[
			\sem{e_1 : \floattype{B_1}{V_1}} \gg\!= \lambda e_1'.\;
			\sem{\uniform(e_1',e_2 : \floattype{B_2}{V_2}) : \floattype{B}{V}}
		\]
		in the semantics is applied. By the IH,
		\[
			\sem{e_1 : \floattype{B_1}{V_1}} \in \mathcal{D}(Expr_{\floattype{B_1}{V_1}} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_1') = \sem{\uniform(e_1',e_2)},
		\]
		which maps
		\[
			Expr_{\floattype{B_1}{V_1}} \;\to\; \mathcal{D}(Expr_{\floattype{B}{V}} + \bot)
		\]
		by Lemma~\ref{lem:typed-replacement}.

		By Proposition~\ref{prop:bind-typing}, we conclude
		\begin{align*}
			\sem{e_1 : \floattype{B_1}{V_1}} \gg\!= \lambda e_1'.\;
			\sem{\uniform(e_1',e_2 : \floattype{B_2}{V_2}) : \float} \\
			\in \mathcal{D}(Expr_{\floattype{B}{V}} + \bot).
		\end{align*}

		Now suppose $e_2$ is an expression, then the clause
		\[
			\sem{e_2 : \floattype{B_2}{V_2}} \gg\!= \lambda e_2'.\;
			\sem{\uniform(e_1 : \floattype{B_1}{V_1}, e_2') : \floattype{B}{V}}
		\]
		in the semantics is applied. By the IH,
		\[
			\sem{e_2 : \floattype{B_2}{V_2}} \in \mathcal{D}(Expr_{\floattype{B_2}{V_2}} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_2') = \sem{\uniform(e_1,e_2')},
		\]
		which maps
		\[
			Expr_{\floattype{B_2}{V_2}} \;\to\; \mathcal{D}(Expr_{\floattype{B}{V}} + \bot)
		\]
		by Lemma~\ref{lem:typed-replacement}. 

		By Proposition~\ref{prop:bind-typing}, we conclude
		\begin{align*}
			\sem{e_2 : \floattype{B_2}{V_2}} \gg\!= \lambda e_2'.\;
			\sem{\uniform(e_1 : \floattype{B_1}{V_1}, e_2') : \floattype{B}{V}} \\
			\in \mathcal{D}(Expr_{\floattype{B}{V}} + \bot).
		\end{align*}
		\end{itemize}

		\textbf{Case} discrete $\discrete(v_0, \ldots, v_n) : \tau$, where $\tau = \fin{n}$. It holds that either:
		\[
			\text{Discrete}(v_0,\ldots,v_n) \in \mathcal{D}(Expr_{\fin{n}} + \bot)
			\qquad\text{and}\qquad
			\delta_{\bot} \in \mathcal{D}(Expr_{\fin{n}} + \bot).
		\]

		\textbf{Case} $\dc : \fin{|B| + 1}$. From typing we have
		\[
			\emptyset \vdash e_1 : \fin{|B_1| + 1} 
			\qquad\text{and}\qquad 
			\emptyset \vdash e_2 : \fin{|B_2| + 1}.
		\]  
		We have the following possibilities:
		\begin{itemize}
			\item If both $e_1 = v'$ and $e_2 = w'$ are values, then by Lemma~\ref{lem:canonical-forms} they are finite constants, representing discretized values. Since $B_1 \neq \top$ and $B_2 \neq \top$, $v'$ and $w'$ are indices of $\intervals{B_1}$ and $\intervals{B_2}$ respectively, by definition of the discretization function (float constant case).

			By Definition~\ref{def:distinguishes} and the definition of the discretization function (float constant case), the existence of unique values $v_i \in V_1$ and $w_j \in V_2$ such that $\discretize{v_i : \floattype{B_1}{V_1}} = v'$ and $\discretize{w_j : \floattype{B_2}{V_2}} = w'$ is guaranteed.

			Since $v_i$ and $w_j$ are concrete float values and $B \neq \top$, by definition of the discretization function (uniform case where $B \neq \top$ and $e_1,e_2 \in \R$):
			\[
				\discretize{\uniform(v_i,w_j) : \floattype{B}{\top}} = \discrete(p_0, \ldots, p_{|B|})
			\]
			where $p_k$ denotes the difference in the CDF of $\uniform(v_i,w_j)$ evaluated at consecutive cuts of $B$.

			As proved in the case above,
			\[
				\discrete(p_0, \ldots, p_{|B|}) \in \mathcal{D}(Expr_{\fin{|B|+1}}),
			\]
			and thus,
			\[
				\discretize{\uniform(v_i,w_j) : \floattype{B}{\top}} \in \mathcal{D}(Expr_{\fin{|B|+1}})
			\]
			as desired.

			\item At least $e_1$ or $e_2$ is an expression. 
			Suppose $e_1$ is an expression, then the clause
			\[
				\sem{e_1 : \fin{|B_1| + 1}} \gg\!= \lambda e_1'.\;
				\sem{\dc(e_1',e_2 : \fin{|B_2| + 1}) : \fin{|B| + 1}}
			\]
			in the semantics is applied. By the IH,
			\[
				\sem{e_1 : \fin{|B_1| + 1}} \in \mathcal{D}(Expr_{\fin{|B_1| + 1}} + \bot).
			\]
			We use the left bind clause with 
			\[
				f(e_1') = \sem{\dc(e_1',e_2)},
			\]
			which maps
			\[
				Expr_{\fin{|B_1| + 1}} \;\to\; \mathcal{D}(Expr_{\fin{|B| + 1}} + \bot)
			\]
			by Lemma~\ref{lem:typed-replacement}.

			By Proposition~\ref{prop:bind-typing}, we conclude
			\begin{align*}
				\sem{e_1 : \fin{|B_1| + 1}} \gg\!= \lambda e_1'.\;
				\sem{\dc(e_1',e_2 : \fin{|B_2| + 1}) : \fin{|B| + 1}} \\
				\in \mathcal{D}(Expr_{\fin{|B| + 1}} + \bot).
			\end{align*}

			Now suppose $e_2$ is an expression, then the clause
			\[
				\sem{e_2 : \fin{|B_2| + 1}} \gg\!= \lambda e_2'.\;
				\sem{\dc(e_1 : \fin{|B_1| + 1}, e_2') : \fin{|B| + 1}}
			\]
			in the semantics is applied. By the IH,
			\[
				\sem{e_2 : \fin{|B_2| + 1}} \in \mathcal{D}(Expr_{\fin{|B_2| + 1}} + \bot).
			\]
			We use the left bind clause with 
			\[
				f(e_2') = \sem{\dc(e_1,e_2')},
			\]
			which maps
			\[
				Expr_{\fin{|B_2| + 1}} \;\to\; \mathcal{D}(Expr_{\fin{|B| + 1}} + \bot)
			\]
			by Lemma~\ref{lem:typed-replacement}. 

			By Proposition~\ref{prop:bind-typing}, we conclude
			\begin{align*}
				\sem{e_2 : \fin{|B_2| + 1}} \gg\!= \lambda e_2'.\;
				\sem{\dc(e_1 : \fin{|B_1| + 1},e_2') : \float} \\
				\in \mathcal{D}(Expr_{\fin{|B| + 1}} + \bot).
			\end{align*}
		\end{itemize}
		Hence, for all well-typed expressions $e : \tau$, $\sem{e : \tau} \in \mathcal{D}(Expr_\tau + \bot)$.
\end{proof}

\subsubsection{Relation $\relation_{\tau}$}
We couple the continuous (and type-annotated) program $e$ appropriately with its discretized one $\discretize{e}$. Formally, the coupling relation is:

\[
\relation_{\tau}(e : \tau,e' : \tau') \iff e' : \tau' == \discretize{e : \tau} : \discretize{\tau}.
\]

In particular, we have the following:

\begin{align*}
	\{(\true,\true), (\false,\false)\} \in \relation_{\bool} \\
	\{(v,v) : v \text{ is a value}\} \in \relation_{\float} \\
\end{align*}

% What does this mean precisely? In particular, when the return type of $e$ is $\bool$, then the return type of $\discretize{e}$ is also $\bool$, and they have the same distribution. When the return type of $e$ is $\float$, then the return type of $\discretize{e}$ is also $\float$, and they have the same distribution. 

\subsubsection{Small-step semantics for $e : \tau, e' : \tau'$.  $\relation_{\tau} \to \mathcal{D}(\relation_{\tau} + \bot)$.}

		\text{s.t. } \discretize{v_i : \floattype{B_1}{\{v_i\}}} = \discretize{v : \floattype{B_1}{\{v\}}} \\
		\text{ and } \discretize{w_j : \floattype{B_2}{\{w_j\}}} = \discretize{w : \floattype{B_2}{\{w\}}} \quad &\text{ if } e_1 = v, e_2 = w \\
		\sem{e_2 : \tau_2, \discretize{e_2 : \tau_2} : \discretize{\tau_2}} \gg\!= \lambda (f,f').\sem{ \uniform(v : \tau_1, f) : \tau, \\
		\dc(\discretize{v : \tau_1} : \discretize{\tau_1}, f')} \quad &\text{ if } e_1 = v, e_2 \neq w \\
		\sem{e_1 : \tau_1, \discretize{e_1 : \tau_1} : \discretize{\tau_1}} \gg\!= \lambda (f,f').\sem{ \uniform(f, e_2 : \tau_2) : \tau, \\
		\dc(f', \discretize{e_2 : \tau_2} : \discretize{\tau_2})} \quad &\text{ otherwise } \\
	\end{cases} \\
	&\text{where } \tau = \floattype{B}{V}, B \neq \top, \\
	&\tau_1 = \floattype{B_1}{V_1}, V_1 = \{v_1, \ldots, v_n\}, \\
	&\tau_2 = \floattype{B_2}{V_2}, V_2 = \{w_1, \ldots, w_m\} \\[1cm]
	%
	%% value
	%
	&\sem{ v : \tau, \\
	&\discretize{v : \tau} : \discretize{\tau} } \\
	&= \delta_{(v : \tau, \discretize{v : \tau} : \discretize{\tau})} \\
	&\text{where } \tau = \floattype{B}{V}, B = \top \text{ or } B \neq \top \\
\end{align*}
}

\subsubsection{Lemmas for $e,e'$}

\begin{lemma}[Discretization Preserves Form]\label{lem:form-preservation}
	Let $e$ be a well-typed expression of type $\tau$. Then:
	\begin{enumerate}
		\item If $e : \tau$ is a value, then $\discretize{e : \tau}$ is a value.
		\item If $e : \tau$ is an expression, then $\discretize{e : \tau}$ is an expression.
	\end{enumerate}
\end{lemma}

\begin{proof}
	By a case analysis on the structure of $e$. 

	\textbf{Case} $e$ is a constant $v : \tau$. $e$ is a value. Via discretization,
	\[
		\discretize{v : \tau} = \begin{cases}
			v &\text{ if } B = \top \\
			\finconst{k}{n+1} \text{ s.t. } c \in \intervals{B}_k &\text{ if } B = \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\}
		\end{cases}
	\]
	both of which are values.
	
	\textbf{Case} $e$ is a variable $x : \tau$. Via discretization, \[
		\discretize{x : \tau} = x.
	\]
	$e$ can either be a value or an expression, but since the discretization leaves $x$ unchanged, the value or expression is preserved.

	\textbf{Case} $e$ is a let statement $\letkw \; x = e_1 : \tau_1 \; \inkw \; e_2 : \tau_2 : \tau_2$. $e$ is an expression. Via discretization,
	\[
	\discretize{\letkw \; x = e_1 : \tau_1 \; \inkw \; e_2 : \tau_2 : \tau_2} = 
	\letkw \; x = \discretize{e_1 : \tau_1} \; \inkw \; \discretize{e_2 : \tau_2},
	\]
	which is also an expression.

	\textbf{Case} $e$ is a comparison $e_1 : \tau_1 < e_2 : \tau_2 : \bool$, where $\tau_1 = \floattype{B_1}{V_1}$ and $\tau_2 = \floattype{B_2}{V_2}$. $e$ is an expression. Via discretization,
	\[
	\discretize{e_1 : \tau_1 < e_2 : \tau_2 : \bool} = \begin{cases}
		\discretize{e_1 : \tau_1} < \discretize{e_2 : \tau_2} &\text{ if } B = \top \\
		\discretize{e_1 : \tau_1} \finlt{n+1} \discretize{e_2 : \tau_2} &\text{ if } B = \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\} \\
	\end{cases}
	\]
	both of which are expressions.

	\textbf{Case} $e$ is a conditional $\ifkw \; e_1 : \bool \; \thenkw \; e_2 : \tau \; \elsekw \; e_3 : \tau : \tau$. $e$ is an expression. Via discretization,
	\[
		\discretize{\ifkw \; e_1 : \bool \; \thenkw \; e_2 : \tau \; \elsekw \; e_3 : \tau : \tau} = \ifkw \; \discretize{e_1 : \bool} \; \thenkw \; \discretize{e_2 : \tau} \; \elsekw \; \discretize{e_3 : \tau}
	\]
	which is also an expression.

	\textbf{Case} $e$ is a uniform distribution $\uniform(e_1 : \tau_1, e_2 : \tau_2) : \tau$, where $\tau_1 = \floattype{B}{V}$. $e$ is an expression. Via discretization,
	\begin{align*}
		&\discretize{\uniform(e_1 : \tau_1, e_2 : \tau_2) : \tau} \\
		&= \begin{cases}
			\uniform(\discretize{e_1 : \tau_1}, \discretize{e_2 : \tau_2}) &\text{ if } B = \top \\
			\discrete(p_0,\ldots,p_n) &\text{ if } B = \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\}, e_1,e_2 \in \R \\
			\dc(e_1 : \tau_1,e_2 : \tau_2) &\text{ if } B = \{\sim_1\!\!b_1, \ldots, \sim_n\!\!b_n\}, e_1,e_2 \notin \R
		\end{cases}
	\end{align*}
	all of which are expressions. In particular, \dc is a let statement.
\end{proof}


\begin{lemma}[Well-defined]
	For all well-typed expressions $(e : \tau, e' : \tau')$ such that $\relation_{\tau}(e,e')$, $\sem{(e : \tau,e' : \tau')} \in \mathcal{D}(\relation_{\tau} + \bot)$.
\end{lemma}

\begin{proof}
	By induction on the structure of the typing derivation $e : \tau$. We do a case analysis on the last rule used.

	\textbf{Case} constant $(v : \tau, \discretize{v : \tau} : \discretize{\tau}) \in \relation_{\tau}$, where $\tau = \floattype{B}{V}$. It holds that 
	\[
		\sem{v : \tau, \discretize{v : \tau} : \discretize{\tau}} = \delta_{(v : \tau, \discretize{v : \tau} : \discretize{\tau})} \in \mathcal{D}(\relation_{\tau} + \bot).
	\]

	\textbf{Case} comparison $(e_1 < e_2 : \bool, \discretize{e_1} < \discretize{e_2} : \discretize{\bool}) \in \relation_{\bool}$. We have the following possibilities:
	\begin{itemize}
		\item If $e_1 = v_1$ and $e_2 = v_2$ are values, then by Lemma~\ref{lem:form-preservation}, $\discretize{e_1} = v_1'$ and $\discretize{e_2} = v_2'$ are also values. Either:
		\[
			\delta_{(\true:\bool, \true:\bool)} \in \mathcal{D}(\relation_{\bool} + \bot) 
			\quad\text{or}\quad
			\delta_{(\false:\bool, \false:\bool)} \in \mathcal{D}(\relation_{\bool} + \bot) 
		\]
		\item At least $e_1$ or $e_2$ is an expression. Suppose $e_1$ is an expression; by Lemma~\ref{lem:form-preservation}, $\discretize{e_1}$ is also an expression. Let $\tau_1 = \floattype{B_1}{V_1}$ and $\tau_2 = \floattype{B_2}{V_2}$. The clause
		\begin{align*}
			&\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} : \discretize{\tau_1}} \\
			&\gg\!= \lambda (e_1',e_1''). \sem{ e_1' < e_2 : \tau_2 : \bool, e_1'' < \discretize{e_2 : \tau_2} : \discretize{\tau_2} : \discretize{\bool} }
		\end{align*}
		in the semantics is applied. By the IH,
		\[
			\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} : \discretize{\tau_1}} \in \mathcal{D}(\relation_{\tau_1} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_1',e_1'') = \sem{ e_1' < e_2, e_1'' < \discretize{e_2 : \tau_2}}
		\]
		which maps
		\[
			\relation_{\tau_1} \to \mathcal{D}(\relation_{\bool} + \bot).
		\]
		by Lemma~\ref{lem:typed-replacement} and Lemma~\ref{lem:type-transformations}.

		By Proposition~\ref{prop:bind-typing}, we conclude
		\begin{align*}
			&\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} : \discretize{\tau_1}} \\
			&\gg\!= \lambda (e_1',e_1''). \sem{ e_1' < e_2 : \tau_2 : \bool, e_1'' < \discretize{e_2 : \tau_2} : \discretize{\tau_2} : \discretize{\bool} } \in \mathcal{D}(\relation_{\bool} + \bot).
		\end{align*}

		Suppose $e_2$ is an expression; by Lemma~\ref{lem:form-preservation}, $\discretize{e_2}$ is also an expression. The clause
		\begin{align*}
			&\sem{ e_2 : \tau_2, \discretize{e_2 : \tau_2} : \discretize{\tau_2}} \\
			&\gg\!= \lambda (e_2',e_2''). \sem{ e_1 : \tau_1 < e_2' : \bool, \discretize{e_1 : \tau_1 } : \discretize{\tau_1} < e_2'' : \discretize{\bool} }
		\end{align*}
		in the semantics is applied. By the IH,
		\[
			\sem{ e_2 : \tau_2, \discretize{e_2 : \tau_2} : \discretize{\tau_2}} \in \mathcal{D}(\relation_{\tau_2} + \bot).
		\]
		We use the left bind clause with 
		\[
			f(e_2',e_2'') = \sem{ e_1 < e_2', \discretize{e_1 : \tau_1} < e_2''}
		\]
		which maps
		\[
			\relation_{\tau_2} \to \mathcal{D}(\relation_{\bool} + \bot).
		\]
		by Lemma~\ref{lem:typed-replacement} and Lemma~\ref{lem:type-transformations}.

		By Proposition~\ref{prop:bind-typing}, we conclude
		\begin{align*}
			&\sem{ e_2 : \tau_2, \discretize{e_2 : \tau_2} : \discretize{\tau_2}} \\
			&\gg\!= \lambda (e_2',e_2''). \sem{ e_1 : \tau_1 < e_2' : \bool, \discretize{e_1 : \tau_1 } : \discretize{\tau_1} < e_2'' : \discretize{\bool} } \in \mathcal{D}(\relation_{\bool} + \bot).
		\end{align*}
	\end{itemize}

	\textbf{Case} let $(\letkw \; x = e_1 \; \inkw \; e_2 : \tau_2, \letkw \; x = \discretize{e_1} \; \inkw \; \discretize{e_2} : \discretize{\tau_2}) \in \relation_{\tau_2}$. We have the following possibilities:
	\begin{itemize}
		\item If $e_1 = v$ is a value, then $\discretize{e_1}$ is also a value by Lemma~\ref{lem:form-preservation}. By the substitution lemma, 
		\[
			\delta_{(e_2[v/x] : \tau_2, \discretize{e_2 : \tau_2}[\discretize{v : \tau_1}/x] : \discretize{\tau_2})} \in \mathcal{D}(\relation_{\tau_2} + \bot).
		\]

		\item If $e_1$ is an expression, then the clause:
		\begin{align*}
			\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} : \discretize{\tau_1}} \gg\!= \lambda (e_1', e_1''). \sem{ \letkw \; x = e_1' \; \inkw \; e_2 : \tau_2 : \tau_2, \\
			\letkw \; x = e_1'' \; \inkw \; \discretize{e_2 : \tau_2} : \discretize{\tau_2} : \discretize{\tau_2}}
		\end{align*}
		in the semantics is applied. By the IH,
		\[
			\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} : \discretize{\tau_1}} \in \mathcal{D}(\relation_{\tau_1} + \bot).
		\]
		We use the left bind clause with
		\[
			f(e_1',e_1'') = \sem{\letkw \; x = e_1' \; \inkw \; e_2, \\
			\letkw \; x = e_1'' \; \inkw \; \discretize{e_2 : \tau_2}}
		\]
		which maps
		\[
			\relation_{\tau_1} \to \mathcal{D}(\relation_{\tau_2} + \bot)
		\]
		by Lemma~\ref{lem:typed-replacement} and Lemma~\ref{lem:type-transformations}.

		By Proposition~\ref{prop:bind-typing}, we conclude
		\begin{align*}
			\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} : \discretize{\tau_1}} \gg\!= \lambda (e_1', e_1''). \sem{ \letkw \; x = e_1' \; \inkw \; e_2 : \tau_2 : \tau_2, \\
			\letkw \; x = e_1'' \; \inkw \; \discretize{e_2 : \tau_2} : \discretize{\tau_2} : \discretize{\tau_2}} \in \mathcal{D}(\relation_{\tau_2} + \bot).
		\end{align*}
	\end{itemize}

	\textbf{Case} uniform $(\uniform(e_1,e_2) : \tau, \discrete(v_0',\ldots,v_n') : \discretize{\tau}) \in \relation_{\tau}$, where $\tau = \floattype{B}{V}$ and $B \neq \top$. We have the following:
	\begin{itemize}
		\item If $e_1 = v_1$ and $e_2 = v_2$ are values, then either:
		\[
			\text{Uniform}(v_1,v_2) \gg\!= \lambda.\delta_{(v,i)} \in \mathcal{D}(\relation_{\float} + \bot)
			\quad\text{or}\quad
			\delta_{\bot} \in \mathcal{D}(\relation_{\float} + \bot)
		\]
	\end{itemize}

	\textbf{Case} uniform $(\uniform(e_1,e_2) : \tau, \dc(\discretize{e_1}, \discretize{e_2}) : \discretize{\tau}) \in \relation_{\tau}$, where $\tau = \floattype{B}{V}$ and $B \neq \top$. We have the following possibilities:
	\begin{itemize}
		\item If $e_1 = v$ and $e_2 = w$, then by Lemma~\ref{lem:form-preservation}, $\discretize{v} = v'$ and $\discretize{w} = w'$ are also values, namely discretized values where $v'$ and $w'$ are indices of $\intervals{B_1}$ and $\intervals{B_2}$, by definition of the discretization function (float constant case). 
		
		By Definition~\ref{def:distinguishes} and the definition of the discretization function (float constant case), the existence of unique values $v_i \in V_1$ and $w_j \in V_2$ such that $\discretize{v_i : \floattype{B_1}{V_1}} = v'$ and $\discretize{w_j : \floattype{B_2}{V_2}} = w'$ is guaranteed. In particular, $v_i = v$ and $w_j = w$ by construction.

		Since $v_i$ and $w_j$ are values and $B \neq \top$, by definition of the discretization function (uniform case where $B \neq \top$ and $e_1,e_2 \in \R$):
		\[
			\discretize{\uniform(v_i,w_j) : \floattype{B}{\top}} = \discrete(p_0, \ldots, p_{|B|})
		\]
		where $p_k$ denotes the difference in the CDF of $\uniform(v_i,w_j)$ evaluated at consecutive cuts of $B$. 
		
		Thus,
		\[
			\sem{\uniform(v,w) : \tau, \discrete(p_0',\ldots,p_{|B|}) : \discretize{\tau}} \in \mathcal{D}(\relation_{\float} + \bot),
		\]
		for which we proved in the above case.
		
		\item At least $e_1$ or $e_2$ is an expression. Same as the comparison case.
	\end{itemize}
\end{proof}

\begin{definition}[Projection functions for small-step]
	% The first and second projections $\pi_1(\mu)$ and $\pi_2(\mu)$ of a distribution $\mu$ over $A \times B$ are defined by $\pi_1(\mu)(a) = \sum_{b \in B} \mu(a,b)$ and $\pi_2(\mu)(n) = \sum_{b \in A} \mu(a,b)$.
	Let $(E_1, \Sigma_1)$ and $(E_2, \Sigma_2)$ be measurable spaces, and let $\bigsem{e,e'} \in \mathcal{D}(E_1 \times E_2)$. We define the first and second projection measures as:
	\begin{enumerate}
		\item $\pi_1(\sem{e,e'})(A) := \sem{e,e'}(A \times E_2)$ for every measurable $A \in \Sigma_1$
		\item $\pi_2(\sem{e,e'})(B) := \sem{e,e'}(E_1 \times B)$ for every measurable $B \in \Sigma_2$
	\end{enumerate}
	Then, $\pi_1(\sem{e,e'})$ is a measure on $(E_1, \Sigma_1)$ and $\pi_2(\sem{e,e'})$ is a measure on $(E_2, \Sigma_2)$.
\end{definition}

\begin{lemma}[Small-step projection]\label{lem:small-step-proj}
	Let $\sem{e : \tau, e' : \tau'} \in \mathcal{D}(\relation_{\tau} + \bot)$. Then:
	\begin{enumerate}
		\item $\pi_1(\sem{e : \tau, e' : \tau'}) = \sem{e : \tau}$
		\item $\pi_2(\sem{e : \tau, e' : \tau'}) = \sem{e' : \tau'}$
	\end{enumerate}
\end{lemma}

\begin{proof}
	By induction on the structure of the typing derivation $e : \tau$. We prove the first equality, as the proof for the second is identical.

	\textbf{Case} float constant $(v : \tau, \discretize{v : \tau} : \discretize{\tau}) \in \relation_{\tau}$, where $\tau = \floattype{B}{V}$. 

	From the semantics, $\sem{v : \tau, \discretize{v : \tau} : \discretize{\tau}} = \delta_{v : \tau, \discretize{v : \tau} : \discretize{\tau}}$.

	\begin{align*}
		&\pi_1(\delta_{(v : \tau, \discretize{v : \tau} : \discretize{\tau})})(A) \\
		&= \delta_{(v : \tau, \discretize{v : \tau} : \discretize{\tau})}(A \times E_2) \\
		&= \delta_{v : \tau} \\
		&= \sem{v : \tau}
	\end{align*}

	\textbf{Case} float less than $(e_1 < e_2 : \bool, \discretize{e_1} < \discretize{e_2} : \discretize{\bool}) \in \relation_{\bool}$. We have the following possibilities:
	\begin{itemize}
		\item If $e_1 = v_1$ and $e_2 = v_2$ are values, then by Lemma~\ref{lem:form-preservation}, $\discretize{e_1}$ and $\discretize{e_2}$ are also values. 
		
		From the semantics, 
		$\sem{e_1 < e_2} = \begin{cases}
			\delta_{(\true:\bool, \true:\bool)} &\text{ if } v_1 < v_2 \\
			\delta_{(\false:\bool, \false:\bool)} &\text{ if } v_1 \geq v_2
		\end{cases}$
		
		Thus, either:
		\begin{align*}
			&\pi_1(\delta_{(\true:\bool, \true:\bool)})(A) \\
			&= \delta_{(\true:\bool, \true:\bool)}(A \times E_2) \\
			&= \delta_{\true:\bool} \\
			&= \sem{e_1 < e_2 : \bool} \text{ when } e_1 = v_1, e_2 = v_2, v_1 < v_2
		\end{align*}

		\begin{align*}
			&\pi_1(\delta_{(\false:\bool, \false:\bool)})(A) \\
			&= \delta_{(\false:\bool, \false:\bool)}(A \times E_2) \\
			&= \delta_{\false:\bool} \\
			&= \sem{e_1 < e_2 : \bool} \text{ when } e_1 = v_1, e_2 = v_2, v_1 \geq v_2
		\end{align*}

		\item At least $e_1$ or $e_2$ is an expression. Suppose $e_1$ is an expression. 
		
		From the semantics, 
		\begin{align*}
			\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} : \discretize{\tau_1}} \gg\!= \lambda (f,f'). \sem{ f < e_2 : \tau_2 : \bool, \\
			f' < \discretize{e_2 : \tau_2} : \discretize{\tau_2} : \discretize{\bool}}
		\end{align*}
		By Lemma~\ref{lem:bind-proj}, we get:
		\begin{align*}
			&\pi_1(\sem{ e_1 : \tau_1, \discretize{e_1 : \tau_1} : \discretize{\tau_1}}) \gg\!= \lambda f. \pi_1(\sem{ f < e_2 : \tau_2 : \bool, \\
			&\quad f' < \discretize{e_2 : \tau_2} : \discretize{\tau_2} : \discretize{\bool}}) \\
			&= \sem{ e_1 : \tau_1} \gg\!= \lambda f. \sem{ f < e_2 : \tau_2 : \bool} \\
			&= \sem{e_1 < e_2 : \bool} \text{ otherwise } 
		\end{align*}
		
		Suppose $e_2$ is an expression. Analogous to when $e_1$ is an expression.
	\end{itemize}

	\textbf{Case} uniform with $B \neq \top$, $\uniform(e_1 : \tau_1, e_2 : \tau_2) : \tau,
	\dc(\discretize{e_1 : \tau_1} : \discretize{\tau_1}, \discretize{e_2 : \tau_2} : \discretize{\tau_2}) : \discretize{\tau}$. (we detail the $\pi_2$ case because it differs from $\pi_1$ in the construction below). We have the following possibilities:
	\begin{itemize}
		\item If $e_1 = v$ and $e_2 = w$ are values, then by Lemma~\ref{lem:form-preservation}, $\discretize{e_1}$ and $\discretize{e_2}$ are also values. 
		
		From the semantics,
		\begin{align*}
			\sem{ \uniform(v : \tau_1, w : \tau_2) : \tau, \discretize{\uniform(v_i,w_j) : \floattype{B}{\top}} : \discretize{\tau}} \\
			\text{s.t. } \discretize{v_i : \floattype{B_1}{\{v_i\}}} = \discretize{v : \floattype{B_1}{\{v\}}} \\
			\text{ and } \discretize{w_j : \floattype{B_2}{\{w_j\}}} = \discretize{w : \floattype{B_2}{\{w\}}}
		\end{align*}

		If $e_1 = v$ and $e_2 = w$, then by Lemma~\ref{lem:form-preservation}, $\discretize{v} = v'$ and $\discretize{w} = w'$ are also values, namely discretized values where $v'$ and $w'$ are indices of $\intervals{B_1}$ and $\intervals{B_2}$, by definition of the discretization function (float constant case). 
		
		By Definition~\ref{def:distinguishes} and the definition of the discretization function (float constant case), the existence of unique values $v_i \in V_1$ and $w_j \in V_2$ such that $\discretize{v_i : \floattype{B_1}{V_1}} = v'$ and $\discretize{w_j : \floattype{B_2}{V_2}} = w'$ is guaranteed. In particular, $v_i = v$ and $w_j = w$ by construction.

		Since $v_i$ and $w_j$ are values and $B \neq \top$, by definition of the discretization function (uniform case where $B \neq \top$ and $e_1,e_2 \in \R$):
		\[
			\discretize{\uniform(v_i,w_j) : \floattype{B}{\top}} = \discrete(p_0, \ldots, p_{|B|})
		\]
		where $p_k$ denotes the difference in the CDF of $\uniform(v_i,w_j)$ evaluated at consecutive cuts of $B$. 

		So,

		\begin{align*}
			\sem{ \uniform(v : \tau_1, w : \tau_2) : \tau, \discrete(p_0, \ldots, p_{|B|}) : \discretize{\tau}}
		\end{align*}

		\item At least $e_1$ or $e_2$ is an expression. Analogous to float less than case when $e_1$ or $e_2$ is an expression.

	\end{itemize}
\end{proof}

\subsection{Big-step semantics}
For every $e : \tau \in Expr_{\tau}$ and all $n \in \mathbb{N}$, we define the distribution $\bigsemstep{e : \tau}{n} \in \mathcal{D}(Expr_{\tau} + \bot)$ recursively as
%
\[
	\bigsemstep{e : \tau}{n} = 
	\begin{cases}
		\delta_{e : \tau} & \text{if $n = 0$} \\
		%
		\bigsemstep{e : \tau}{n-1} \monbind  \lambda e'. \sem{e' : \tau} & \text{if $n > 0$}~.
	\end{cases}
\]

\begin{lemma}[Basic Properties of $\bigsemstep{\cdot}{n}$]
	\label{lem:bigstepsem}
	Let $e : \tau \in Expr_{\tau}$ and $n \in \mathbb{N}$. Let $V$ be the measurable set of values. We have:
	\begin{enumerate}
		\item\label{lem:bigstepsem1} $\bigsemstep{e : \tau}{n}(V) \leq \bigsemstep{e : \tau}{n+1}(V)$
		\item\label{lem:bigstepsem2} $\bigsemstep{e : \tau}{n}(\bot) \leq \bigsemstep{e : \tau}{n+1}(\bot)$
	\end{enumerate}
\end{lemma}
%
\begin{proof}
	By induction on $n$.

	\underline{Base case}: $n=0$.

	We must show $\bigsemstep{e : \tau}{0}(V) \leq \bigsemstep{e : \tau}{1}(V)$.

	By definition,
	\[
		\bigsemstep{e : \tau}{0} = \delta_{e : \tau}.
	\]

	Therefore,
	\[
		\bigsemstep{e : \tau}{0}(V) =
		\begin{cases}
			1 &\text{ if } e \in V \\
			0 &\text{ if } e \notin V
		\end{cases}
	\]

	Now, for $n=1$, by definition,
	\[
		\bigsemstep{e : \tau}{1} = \delta_{e : \tau} \gg\!= \lambda e'.\sem{e'} = \sem{e : \tau}.
	\]

	We analyze the two cases for $e$:

	\textbf{Case 1:} $e$ is a value $v$. Then, $\bigsemstep{e : \tau}{0}(V) = 1$.

	For $n=1$:
	\[
		\bigsemstep{e : \tau}{1} = \delta_{v : \tau} \gg\!= \lambda e'.\sem{e'} = \sem{v : \tau}.
	\]
	From the semantics, $\sem{v : \tau} = \delta_{v : \tau}$, which means $\bigsemstep{e : \tau}{1} = \delta_{v : \tau}$, and thus
	\[
		\bigsemstep{e : \tau}{1}(V) = 1,
	\]
	and thus $\bigsemstep{e : \tau}{0}(V) \leq \bigsemstep{e : \tau}{1}(V)$.

	\textbf{Case 2:} $e$ is an expression. Then, $\bigsemstep{e : \tau}{0}(V) = 0$.

	For $n=1$:
	\[
		\bigsemstep{e : \tau}{1} = \delta_{e : \tau} \gg\!= \lambda e'.\sem{e'} = \sem{e : \tau}.
	\]
	From the semantics, $\sem{e : \tau}$ may step to either a value, which means $\sem{e : \tau}(V) \geq 0$, or an expression, which means $\sem{e : \tau}(V) = 0$. In either case, 
	\[
		\bigsemstep{e : \tau}{1}(V) \geq 0,
	\]
	and thus $\bigsemstep{e : \tau}{0}(V) \leq \bigsemstep{e : \tau}{1}(V)$.

	\underline{Inductive step}:
	We want to show that for $n = k+1$:
	\begin{align*}
		\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V).
	\end{align*}

	By the recursive definition:
	\begin{align*}
		&\bigsemstep{e : \tau}{k}, \\
		&\bigsemstep{e : \tau}{k+1} = \bigsemstep{e : \tau}{k} \gg\!= \lambda e'.\sem{e'}, \\
		&\bigsemstep{e : \tau}{k+2} = \bigsemstep{e : \tau}{k+1} \gg\!= \lambda e''.\sem{e''} = (\bigsemstep{e : \tau}{k} \gg\!= \lambda e'.\sem{e'}) \gg\!= \lambda e''.\sem{e''}
	\end{align*}

	By the IH,
	\begin{align*}
		\bigsemstep{e : \tau}{k}(V) \leq \bigsemstep{e : \tau}{k+1}(V).
	\end{align*}

	We do a case analysis on $e'$ and $e''$, whether they are in $N$ or $V$.

	\textbf{Case} $e' \in V$, $e'' \in V$.

	Then, from the semantics, $\sem{e'}$ and $\sem{e''}$ are both Diracs over a value. Combined with the IH and the fact that $\sem{\cdot}$ does not change the probability mass of $\cdot$ if it is a value, 
	\[
		\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V).
	\]

	\textbf{Case} $e' \in V$, $e'' \in N$.
	
	Then, from the semantics, $\sem{e'}$ is a Dirac over a value and $\sem{e''}$ is a distribution over a non-value expression. Combined with the IH and the fact that $\sem{\cdot}$ does not change the probability mass of $\cdot$ if it is a value and that $\sem{\cdot}$ does not add any probability mass to $V$ if $\cdot$ is a non-value expression,
	\[
		\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V).
	\]

	\textbf{Case} $e' \in V$, $e'' \in N$.

	Analogous to the previous case, but with $e'$ and $e''$ switched.

	\textbf{Case} $e' \in N$, $e'' \in N$.

	Then, from the semantics,  $\sem{e'}$ and $\sem{e''}$ are both distributions over a non-value expression. Combined with the IH and the fact that $\sem{\cdot}$ does not add any probability mass to $V$ if $\cdot$ is a non-value expression, 
	\[
		\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V).
	\]

	In all cases, $\bigsemstep{e : \tau}{k+1}(V) \leq \bigsemstep{e : \tau}{k+2}(V)$.
\end{proof}

Given $e : \tau \in Expr_{\tau}$ and $n\in\Nats$, define $\bigsemstepterm{e : \tau}{n} = \lambda \measset. \bigsemstep{e : \tau}{n}(\measset \setminus N)$. 

The above statement states that among all possible results after $n$ steps, only count the ones that are values or error.

\begin{definition}[Big-Step Semantics of Expressions]
	Let $e : \tau \in Expr_{\tau}$. The \emph{big-step semantics of $\expr$} is 
	%
	\[
		\bigsem{e : \tau} = \lim_{n \to \infty} \lambda \measset.
		%
		\bigsemstepterm{e : \tau}{n}(A).
	\]
\end{definition}

\begin{lemma}[Existence of the big-step limit]
	The pointwise limit for $\bigsem{e : \tau}$ exists.
\end{lemma}

\begin{proof}
	By Lemmas~\ref{lem:bigstepsem1} and~\ref{lem:bigstepsem2}, the sequence $\bigsemstep{e : \tau}{n}$ is monotonically increasing for values and error. Moreover, $\bigsemstep{e : \tau}{n}$ defines a subprobability measure, and thus $\bigsemstep{e : \tau}{n} \leq 1$ for every $n \in \Nats$. Thus, by the monotone convergence theorem, the pointwise limit exists.
\end{proof}

\begin{definition}[Big-step pairing semantics]\label{def:big-step-2}
	For every $(e : \tau, e' : \tau') \in \relation_{\tau}(e,e')$ and all $n \in \Nats$, define the distribution $\bigsemstep{e : \tau, e' : \tau'}{n} \in \mathcal{D}(\relation_{\tau} + \bot)$ recursively as
	%
	\[
		\bigsemstep{e : \tau, e' : \tau'}{n} = 
		\begin{cases}
			\delta_{(e : \tau. e' : \tau')} & \text{if $n = 0$} \\
			%
			\bigsemstep{e : \tau, e' : \tau'}{n-1} \monbind  \lambda (f,f'). \sem{f : \tau, f' : \tau'} & \text{if $n > 0$}~.
		\end{cases}
	\]
	Given $(e : \tau, e' : \tau') \in \relation_{\tau}(e,e')$ and $n \in \Nats$, define $\bigsemstepterm{e : \tau, e' : \tau'}{n} = \lambda (\measset \times \measset). \bigsemstep{e : \tau}{n}((A \times A) \setminus (\nonvals \times \nonvals))$. 

	Let $(e : \tau, e' : \tau')$ such that $\relation_{\tau}(e,e')$. The big-step semantics of $(e,e')$ is:
	\[
	\bigsem{e : \tau, e' : \tau'} = \lim_{n \to \infty} \lambda (\measset \times \measset).\bigsemstepterm{e : \tau, e' : \tau'}{n}(\measset \times \measset).
	\]
\end{definition}

\begin{lemma}[Basic properties of $\bigsemstep{\cdot, \cdot}{n}$]
	Let $(e : \tau, e' : \tau') \in \relation_{\tau}$ and $n \in \mathbb{N}$. Let $V \times V$ be the measurable set of value pairs. We have:
	\begin{enumerate}
		\item\label{lem:bigstepsem1-pairing} $\bigsemstep{e : \tau, e' : \tau'}{n}(V \times V) \leq \bigsemstep{e : \tau, e' : \tau'}{n+1}(V \times V)$
		\item\label{lem:bigstepsem2-pairing} $\bigsemstep{e : \tau, e' : \tau'}{n}(\bot) \leq \bigsemstep{e : \tau, e' : \tau'}{n+1}(\bot)$
	\end{enumerate}
\end{lemma}

\begin{proof}
	By induction on $n$. Similar to proof for single case.
\end{proof}

\begin{lemma}[Existence of the big-step limit]\label{lem:limit-exists-2}
	The pointwise limit for $\bigsem{e : \tau, e' : \tau'}$ exists.
\end{lemma}

\begin{proof}
	By monotone convergence theorem, via Lemmas~\ref{lem:bigstepsem1-pairing} and~\ref{lem:bigstepsem2-pairing}. Similar to proof for single case.
\end{proof}

\begin{definition}[Projection functions for big-step]
	Let $(E_1, \Sigma_1)$ and $(E_2, \Sigma_2)$ be measurable spaces, and let $\bigsem{e,e'} \in \mathcal{D}(E_1 \times E_2)$. We define the first and second projection measures as:
	\begin{enumerate}
		\item $\pi_1(\bigsem{e,e'})(A) := \bigsem{e,e'}(A \times E_2)$ for every measurable $A \in \Sigma_1$
		\item $\pi_2(\bigsem{e,e'})(B) := \bigsem{e,e'}(E_1 \times B)$ for every measurable $B \in \Sigma_2$
	\end{enumerate}
	Then, $\pi_1(\bigsem{e,e'})$ is a measure on $(E_1, \Sigma_1)$ and $\pi_2(\bigsem{e,e'})$ is a measure on $(E_2, \Sigma_2)$.
\end{definition}

\begin{lemma}[Continunity of projections]
	For all $n \in \Nats$,
	\begin{enumerate}
		\item $\pi_1(\lim_{n \to \infty} \bigsemstep{e,e'}{n}) = \lim_{n \to \infty} \pi_1(\bigsemstep{e,e'}{n})$
		\item $\pi_2(\lim_{n \to \infty} \bigsemstep{e,e'}{n}) = \lim_{n \to \infty} \pi_2(\bigsemstep{e,e'}{n})$ 
	\end{enumerate}
\end{lemma}

\begin{proof}
	We prove part i) in detail; the proof of part ii) is analogous. Fix an arbitrary measurable $A \in \Sigma_1$:
	\begin{align*}
		&\pi_1(\lim_{n \to \infty} \bigsemstep{e,e'}{n})(A) \\
		&= \lim_{n \to \infty} \bigsemstep{e,e'}{n}(A \times X_2) \quad \text{(by defintion of $\pi_1$)}\\
		&= \lim_{n \to \infty} \pi_1(\bigsemstep{e,e'}{n})(A) \quad \text{(by defintion of $\pi_1$)}
	\end{align*}
	Since $A$ was arbitrary, this implies:
	\[
		\pi_1(\lim_{n \to \infty} \bigsemstep{e,e'}{n}) =  \lim_{n \to \infty} \pi_1(\bigsemstep{e,e'}{n}),
	\]
	as desired.
\end{proof}

\begin{lemma}[n-step projection]
	For all $n \in \Nats$, 
	\begin{enumerate}
		\item $\pi_1(\bigsemstep{e : \tau,e' : \tau'}{n}) = \bigsemstep{e : \tau}{n}$
		\item $\pi_2(\bigsemstep{e : \tau,e' : \tau'}{n}) = \bigsemstep{e' : \tau'}{n}$
	\end{enumerate}
\end{lemma}

\begin{proof}
	By induction on $n$. We prove part i) in detail; the proof of part ii) is analogous.

	\underline{Base case:} $n=0$:

	By definition of the big-step semantics, 
	\[
		\bigsemstep{e : \tau, e' : \tau'}{0} = \delta_{(e : \tau, e' : \tau')}
	\]

	Projecting the first component: $\pi_1(\delta_{(e : \tau, e' : \tau')}) = \delta_{e : \tau} = \bigsemstep{e : \tau}{0}$.

	\underline{Inductive step:} Assume for $n-1$. For $n > 0$:
	\[
		\bigsemstep{e : \tau, e' : \tau'}{n} = \bigsemstep{e : \tau, e' : \tau'}{n-1} \monbind  \lambda (f,f'). \sem{f : \tau, f' : \tau'}
	\]

	By Lemma~\ref{lem:bind-proj},
	\[
		\pi_1(\bigsemstep{e : \tau, e' : \tau'}{n}) = \pi_1(\bigsemstep{e : \tau, e' : \tau'}{n-1}) \monbind  \lambda f. \pi_1(\sem{f : \tau, f' : \tau'})
	\]

	By the IH, $\pi_1(\bigsemstep{e : \tau, e' : \tau'}{n-1}) = \bigsemstep{e : \tau}{n-1}$.

	By Lemma~\ref{lem:small-step-proj}, $\pi_1(\sem{f : \tau, f' : \tau'}) = \sem{f : \tau}$.

	So,
	\[
		\pi_1(\bigsemstep{e : \tau, e' : \tau'}{n}) = \bigsemstep{e : \tau}{n-1} \monbind \lambda f.\sem{f : \tau} = \bigsemstep{e : \tau}{n}.
	\]
\end{proof}

\begin{lemma}[Coupling projection for big-step]\label{lem:big-step-proj}
	Let $\bigsem{e : \tau, e' : \tau'} \in \mathcal{D}(\relation_{\tau} + \bot)$. Then,
	\begin{enumerate}
		\item $\pi_1(\bigsem{e : \tau, e' : \tau'}) = \bigsem{e : \tau}$
		\item $\pi_2(\bigsem{e : \tau, e' : \tau'}) = \bigsem{e' : \tau'}$
	\end{enumerate}
\end{lemma}

\begin{proof}
	\begin{align*}
		&\pi_1(\bigsem{e : \tau, e' : \tau'}) \\
		&= \pi_1(\lim_{n \to \infty} \bigsemstep{e : \tau, e' : \tau'}{n}) \quad \text{(by defintion of big-step pairing semantics)} \\
		&= \lim_{n \to \infty} \pi_1(\bigsemstep{e : \tau, e' : \tau'}{n}) \quad \text{(by communitivity lemma)} \\
		&= \lim_{n \to \infty} \bigsemstep{e : \tau}{n} \quad \text{(by definition of n-step projection lemma)} \\
		&= \bigsem{e : \tau} \quad \text{(by definition of big-step semantics)}
	\end{align*}
\end{proof}

\begin{theorem}[Contextual Equivalence]
	For all $e : \tau$, we have $\bigsem{e : \tau} = \bigsem{\discretize{e : \tau} : \discretize{\tau}}$.
\end{theorem}

\begin{proof}
	Follows from Defintion~\ref{def:big-step-2}, Lemma~\ref{lem:limit-exists-2}, and Lemma~\ref{lem:big-step-proj}.
\end{proof}
